8.5 math.h และฟังก์ชันคณิตศาสตร์ที่ใช้บ่อย

ภาษา C มีฟังก์ชันคณิตศาสตร์สำเร็จรูปให้ใน math.h เช่น รากที่สองและยกกำลัง ใช้คำนวณระยะทาง พื้นที่ ดอกเบี้ย ซึ่งออกในโจทย์บ่อย แต่มีกับดักใหญ่เรื่องจำนวนเต็มที่ทำให้คนเสียคะแนนกันมาก ครึ่งหลังของหน้าเป็นฟังก์ชันที่ไม่มีในไลบรารี แต่น้องจะได้เขียนเองซ้ำแล้วซ้ำอีกในโจทย์แข่งขัน คือตรวจจำนวนเฉพาะและหา ห.ร.ม.

sqrt รากที่สอง

บ้านอยู่ที่พิกัด (0, 0) โรงเรียนอยู่ที่ (3, 5) ระยะทางเป็นเส้นตรงหาได้จากทฤษฎีบทพีทาโกรัส คือรากที่สองของ dx2 + dy2 ฟังก์ชัน sqrt (มาจาก square root) รับ double และส่ง double กลับ

#include <stdio.h>
#include <math.h> // (1)!

double distance(double x1, double y1, double x2, double y2)
{
    double dx = x2 - x1;
    double dy = y2 - y1;
    return sqrt(dx * dx + dy * dy);
}

int main()
{
    double home_x, home_y, school_x, school_y, tutor_x, tutor_y;
    scanf("%lf %lf", &home_x, &home_y);
    scanf("%lf %lf", &school_x, &school_y);
    scanf("%lf %lf", &tutor_x, &tutor_y);
    double first = distance(home_x, home_y, school_x, school_y);
    double second = distance(school_x, school_y, tutor_x, tutor_y);
    printf("home to school %.2f\n", first);
    printf("school to tutor %.2f\n", second);
    printf("total %.2f\n", first + second);
    return 0;
}

เทียบระยะไม่ต้องใช้ sqrt

ถ้าโจทย์แค่ถามว่าจุดไหนใกล้กว่า ให้เทียบ dx2 + dy2 ตรงๆ เพราะระยะที่มากกว่าก็มีกำลังสองมากกว่าเสมอ ถ้าพิกัดเป็นจำนวนเต็ม ค่านี้เป็นจำนวนเต็มด้วย เทียบได้แม่นยำ ไม่มีปัญหาทศนิยมคลาดแบบหน้า 1.4

pow ยกกำลัง

pow(x, y) คือ x ยกกำลัง y รับ double สองตัว ส่ง double กลับ เหมาะกับงานที่เป็นทศนิยมอยู่แล้ว เช่น ฝากเงินได้ดอกเบี้ยทบต้นปีละ r เปอร์เซ็นต์ ผ่านไป y ปีจะมีเงิน เงินต้น × (1 + r/100)y

#include <stdio.h>
#include <math.h>

int main()
{
    double money, rate;
    int years;
    scanf("%lf %lf %d", &money, &rate, &years);
    double result = money * pow(1 + rate / 100, years);
    printf("%.2f\n", result);
    printf("%f\n", pow(2, 10));
    return 0;
}

กับดัก pow กับจำนวนเต็มใหญ่

เพราะ pow คำนวณเป็น double ซึ่งแม่นได้ราว 15 ถึง 16 หลักตามหน้า 1.4 ถ้าคำตอบเป็นจำนวนเต็มที่ยาวกว่านั้น หลักท้ายๆ จะเพี้ยน ลองเทียบกับฟังก์ชันยกกำลังที่เขียนเองด้วยการคูณซ้ำแบบ long long

#include <stdio.h>
#include <math.h>

long long power(long long base, int exponent)
{
    long long result = 1;
    for (int i = 0; i < exponent; i++) {
        result *= base;
    }
    return result;
}

int main()
{
    int base, exponent;
    scanf("%d %d", &base, &exponent);
    long long by_pow = (long long)pow(base, exponent);
    long long by_loop = power(base, exponent);
    printf("pow  = %lld\n", by_pow);
    printf("loop = %lld\n", by_loop);
    printf("%.1f\n", pow(base, exponent));
    return 0;
}

abs กับ fabs ค่าสัมบูรณ์

ค่าสัมบูรณ์คือระยะห่างจาก 0 โดยไม่สนเครื่องหมาย ใช้หาว่าสองค่าห่างกันเท่าไร โดยไม่ต้องรู้ว่าตัวไหนมากกว่า มีสองฟังก์ชันที่ชื่อคล้ายกันแต่ใช้คนละชนิด

is_prime ตรวจจำนวนเฉพาะ

ในหน้า 4.5 เราตรวจจำนวนเฉพาะด้วยลูปที่หยุดที่ √n โจทย์จริงมักถามต่อจากนั้น เช่น นับจำนวนเฉพาะในช่วง หาจำนวนเฉพาะคู่แฝด หรือตรวจว่าเลขกลับหลังยังเป็นจำนวนเฉพาะไหม ทุกแบบต้องตรวจซ้ำหลายครั้ง จึงควรเป็นฟังก์ชัน

#include <stdio.h>

int is_prime(int n)
{
    if (n < 2) { // (1)!
        return 0;
    }
    for (long long i = 2; i * i <= n; i++) {
        if (n % i == 0) {
            return 0; // (2)!
        }
    }
    return 1;
}

int main()
{
    int limit;
    scanf("%d", &limit);
    for (int p = 2; p + 2 <= limit; p++) {
        if (is_prime(p) && is_prime(p + 2)) { // (3)!
            printf("%d %d\n", p, p + 2);
        }
    }
    return 0;
}
  1. 0 กับ 1 ไม่ใช่จำนวนเฉพาะ และลูปข้างล่างจะไม่ทำงานเลยกับสองค่านี้ ต้องแยกไว้ตั้งแต่ต้น
  2. เจอตัวหารตัวแรกก็ตอบได้เลย return จบฟังก์ชันและออกจากลูปไปพร้อมกัน ไม่ต้องใช้ break กับตัวแปร flag แบบหน้า 4.5
  3. ฟังก์ชันคืน 1 เมื่อจริง และ 0 เมื่อเท็จ ตามหน้า 3.1 จึงใช้ในเงื่อนไขได้ตรงๆ อ่านออกเสียงได้ว่า "ถ้า p เป็นจำนวนเฉพาะ และ p + 2 เป็นจำนวนเฉพาะ"

gcd หา ห.ร.ม.

ห.ร.ม. (ตัวหารร่วมมาก) หรือ gcd (greatest common divisor) ใช้ทำเศษส่วนให้เป็นอย่างต่ำ หา ค.ร.น. และโจทย์แบ่งของเป็นกองเท่าๆ กัน วิธีลองหารทุกตัวก็ได้ แต่มีวิธีที่เร็วกว่ามากเรียกว่า วิธีของยุคลิด (Euclid) ซึ่งใช้แค่ %

ความคิดคือ ตัวเลขใดที่หารทั้ง a และ b ลงตัว จะหาร a % b ลงตัวด้วย เพราะ a % b คือ a ลบด้วย b หลายๆ เท่า และกลับกันก็จริง ห.ร.ม. ของ a กับ b จึงเท่ากับ ห.ร.ม. ของ b กับ a % b ซึ่งเป็นเลขที่เล็กลง ทำซ้ำไปจนตัวหลังเป็น 0 ตัวหน้าคือคำตอบ ลองกับ 84 และ 36

#include <stdio.h>

int gcd(int a, int b)
{
    while (b != 0) {
        int remainder = a % b;
        a = b; // (1)!
        b = remainder;
    }
    return a;
}

int main()
{
    int top, bottom;
    scanf("%d %d", &top, &bottom);
    int g = gcd(top, bottom);
    printf("gcd = %d\n", g);
    printf("fraction = %d/%d\n", top / g, bottom / g);
    printf("lcm = %d\n", top / g * bottom); // (2)!
    return 0;
}
  1. แก้ค่า a กับ b ได้ตามสบาย เพราะเป็นสำเนาตามหน้า 8.2 top กับ bottom ใน main ยังอยู่ครบ
  2. ค.ร.น. คือ a × b ÷ ห.ร.ม. แต่หารก่อนแล้วค่อยคูณ ผลระหว่างทางจะได้ไม่ใหญ่เกิน ดูแบบที่ 3 ในส่วนลองทำให้พัง

#include <stdio.h>

int gcd(int a, int b)
{
    while (b != 0) {
        int remainder = a % b;
        a = b;
        b = remainder;

    return a;
}

int main()
{

#include <stdio.h>

int is_prime(int n)
{
    for (long long i = 2; i * i <= n; i++) {
        if (n % i == 0) {
            return 0;
        }
    }
    return 1;
}

int main()
{
    for (int n = 0; n <= 5; n++) {
        printf("%d %d\n", n, is_prime(n));
    }
    return 0;
}

สรุป

  • math.h มี sqrt pow fabs ทุกตัวรับและส่ง double ส่วน abs สำหรับ int อยู่ใน stdlib.h
  • pow ใช้กับงานทศนิยม คำตอบเป็นจำนวนเต็มใหญ่ให้เขียน power เองด้วย long long
  • เทียบระยะทางใช้กำลังสองของระยะ ไม่ต้อง sqrt
  • is_prime ต้องจัดการ n < 2 แยก และลูปถึงแค่ i * i <= n
  • gcd แบบยุคลิดคือแทน (a, b) ด้วย (b, a % b) จน b เป็น 0 ค.ร.น. คือ a / gcd(a, b) * b หารก่อนคูณ
  • ฟังก์ชันที่ตอบใช่หรือไม่ ตั้งชื่อขึ้นต้นด้วย is_ แล้วใช้ในเงื่อนไขได้ตรงๆ
