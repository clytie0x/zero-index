5.3 นับความถี่ด้วยอาเรย์นับ

ครูตรวจข้อสอบย่อยคะแนนเต็ม 10 เสร็จ อยากรู้ว่าคะแนนแต่ละค่ามีกี่คน ได้ 10 กี่คน ได้ 9 กี่คน ไล่ลงไปถึง 0 คำถามแบบ "แต่ละค่าเกิดขึ้นกี่ครั้ง" เรียกว่าหา ความถี่ (frequency) เป็นโจทย์ที่เจอบ่อยมาก และมีวิธีที่เร็วมากถ้าค่าที่ต้องนับเป็นจำนวนเต็มช่วงไม่กว้าง หน้านี้คือวิธีนั้น

ให้ค่าเป็นเลขช่อง

ในหน้าก่อน เราใช้ index บอกว่า "คนที่เท่าไร" แล้วเก็บคะแนนไว้ในช่อง ครั้งนี้กลับด้าน ให้ คะแนนเป็น index แล้วเก็บ จำนวนคน ไว้ในช่อง

  • มีอาเรย์ count ที่ช่อง count[s] เก็บว่ามีกี่คนที่ได้คะแนน s เริ่มต้นทุกช่องเป็น 0
  • อ่านคะแนนมาทีละคน ได้ score เท่าไร ก็ไปเพิ่มช่องนั้นหนึ่ง count[score]++

อาเรย์ที่ใช้นับแบบนี้เรียกว่า อาเรย์นับ (counting array) คะแนนมีได้ 0 ถึง 10 รวม 11 ค่า จึงต้องมี 11 ช่อง

#include <stdio.h>

int main()
{
    int count[11] = {0}; // (1)!
    int n;
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        int score;
        scanf("%d", &score);
        count[score]++; // (2)!
    }
    for (int s = 0; s <= 10; s++) { // (3)!
        printf("%d: %d\n", s, count[s]);
    }
    return 0;
}
  1. 11 ช่องสำหรับคะแนน 0 ถึง 10 และ = {0} ให้ทุกช่องเริ่มที่ 0 ห้ามลืม
  2. count[score] เป็นตัวแปร int ธรรมดาตัวหนึ่ง ++ จึงใช้ได้เหมือนเพิ่มค่าตัวนับในบทที่ 4
  3. วนตามค่าคะแนน ไม่ใช่ตามจำนวนคน ช่องสุดท้ายคือ 10 จึงใช้ <= ได้ เพราะอาเรย์มี 11 ช่อง

คะแนนที่มีคนได้มากที่สุด

พอมีอาเรย์นับแล้ว คำถามอื่นตอบได้ด้วยเครื่องมือจากหน้า 5.2 เช่น คะแนนไหนมีคนได้มากที่สุด ก็คือหา index ของช่องที่มีค่ามากสุดในอาเรย์ count ตัวอย่างนี้คะแนนเต็ม 100 จึงต้องมี 101 ช่อง และพิมพ์เฉพาะคะแนนที่มีคนได้

#include <stdio.h>

int main()
{
    int count[101] = {0};
    int n;
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        int score;
        scanf("%d", &score);
        count[score]++;
    }
    for (int s = 0; s <= 100; s++) {
        if (count[s] > 0) {
            printf("%d %d\n", s, count[s]);
        }
    }
    int mode = 0;
    for (int s = 1; s <= 100; s++) {
        if (count[s] > count[mode]) {
            mode = s;
        }
    }
    printf("most common = %d (%d students)\n", mode, count[mode]);
    return 0;
}

เช็กว่าเคยเห็นหรือยัง

บางโจทย์ไม่ได้ต้องการจำนวนครั้ง แค่อยากรู้ว่ามีหรือไม่มี เช่น ห้องหนึ่งมีนักเรียน N คน เลขที่ 1 ถึง N ครูเก็บรายชื่อการส่งการบ้านไว้ M ครั้ง บางคนส่งซ้ำ ครูอยากรู้ว่ามีกี่คนที่ส่งแล้ว และเลขที่ไหนยังไม่ส่ง

#include <stdio.h>

int main()
{
    int sent[41] = {0}; // (1)!
    int n, m;
    scanf("%d %d", &n, &m);
    for (int i = 0; i < m; i++) {
        int number;
        scanf("%d", &number);
        sent[number] = 1; // (2)!
    }
    int done = 0;
    for (int number = 1; number <= n; number++) {
        if (sent[number] == 1) {
            done++;
        }
    }
    printf("sent = %d\n", done);
    printf("missing:");
    for (int number = 1; number <= n; number++) {
        if (sent[number] == 0) {
            printf(" %d", number);
        }
    }
    printf("\n");
    return 0;
}
  1. เลขที่มีได้ 1 ถึง 40 ขอ 41 ช่อง แล้วไม่ใช้ช่อง 0 ยอมเสียหนึ่งช่อง แลกกับการที่เลขที่ตรงกับ index พอดี ไม่ต้องลบหนึ่งทุกครั้ง
  2. ใส่ 1 แทน ++ ส่งกี่ครั้งก็ยังเป็น 1 จึงนับคนที่ส่งซ้ำแค่ครั้งเดียว

ค่าที่ติดลบ

index ติดลบไม่ได้ ถ้าค่าที่จะนับมีติดลบ ให้ เลื่อนทุกค่าด้วยค่าคงที่ ก่อนใช้เป็น index เช่น อุณหภูมิต่ำสุดของแต่ละวันบนยอดดอยอยู่ระหว่าง -10 ถึง 30 องศา รวม 41 ค่า บวก 10 เข้าไปทุกค่า -10 จะกลายเป็นช่อง 0 และ 30 กลายเป็นช่อง 40 ตอนพิมพ์ก็ลบ 10 กลับ

#include <stdio.h>

int main()
{
    int count[41] = {0};
    int n;
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        int temp;
        scanf("%d", &temp);
        count[temp + 10]++;
    }
    for (int k = 0; k <= 40; k++) {
        if (count[k] > 0) {
            printf("%d degrees: %d days\n", k - 10, count[k]);
        }
    }
    return 0;
}

ใช้ได้เมื่อไร

ลองนึกถึงวิธีที่ไม่ใช้อาเรย์นับ ถ้าจะหาว่าคะแนนแต่ละค่ามีกี่คน ต้องวนคะแนนทุกค่า 0 ถึง 100 แล้วในแต่ละค่าวนดูนักเรียนทุกคน นักเรียนแสนคนก็ต้องเทียบ 101 × 100000 ประมาณสิบล้านครั้ง ส่วนอาเรย์นับดูนักเรียนแต่ละคนครั้งเดียว แค่แสนครั้ง ยิ่งช่วงของค่ากว้าง ความต่างก็ยิ่งมาก

เงื่อนไขของวิธีนี้คือ

  • ค่าที่นับต้องเป็นจำนวนเต็ม เพราะต้องใช้เป็น index
  • ช่วงของค่าต้องไม่กว้างเกินจำนวนช่องที่ประกาศได้ ค่าถึง 106 ยังพอไหว แต่ต้องประกาศอาเรย์นอก main ซึ่งอธิบายในหน้า 5.5 ถ้าค่าถึง 109 ใช้วิธีนี้ไม่ได้ ต้องเรียงข้อมูลก่อน ซึ่งอยู่ในบทที่ 13

แบบที่ 2 ขอช่องขาดไปหนึ่ง

คะแนนเต็ม 100 แต่ประกาศ int count[100] = {0}; ช่องมีแค่ 0 ถึง 99 ไม่มีช่อง 100

#include <stdio.h>

int main()
{
    int count[100] = {0};
    int n;
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        int score;
        scanf("%d", &score);
        count[score]++;
    }
    printf("score 100: %d students\n", count[100]);
    printf("score 99: %d students\n", count[99]);
    return 0;
}

กฎคือ ค่าตั้งแต่ 0 ถึง K ต้องมี K + 1 ช่อง ทำไมช่องที่ไม่มีอยู่จริงถึงให้ค่าแปลกแบบนี้ และอาการอื่นที่หนักกว่านี้ อยู่ในหน้า 5.5

สรุป

  • อาเรย์นับใช้ค่าเป็น index แล้วเก็บจำนวนครั้งไว้ในช่อง count[x]++
  • ต้องเริ่มทุกช่องที่ 0 ด้วย = {0} ลืมแล้วได้ค่าขยะโดยไม่มีคำเตือน
  • ค่า 0 ถึง K ต้องมี K + 1 ช่อง
  • วน index จากน้อยไปมาก ผลจะเรียงตามค่าให้เอง
  • อยากรู้แค่มีหรือไม่มี ใช้อาเรย์ seen ใส่ 1
  • ค่าติดลบให้บวกค่าคงที่ก่อนใช้เป็น index ค่าที่ช่วงกว้างมากใช้วิธีนี้ไม่ได้
