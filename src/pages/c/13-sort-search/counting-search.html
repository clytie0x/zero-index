13.5 counting sort และการค้นหา

หน้านี้มีสองเรื่อง เรื่องแรกคือวิธีเรียงที่เร็วกว่า qsort อีก แต่ใช้ได้เฉพาะเมื่อค่าอยู่ในช่วงแคบ เช่น คะแนน 0 ถึง 100 ซึ่งน้องเกือบเขียนเป็นแล้วตั้งแต่หน้า 5.3 เรื่องที่สองคือการค้นหา ทั้งค้นหาตัวเลขในอาเรย์ ค้นหาชื่อในอาเรย์ของ struct และค้นหาคำในประโยค ซึ่งเป็นโจทย์ที่เจอทั้งใน TOI Zero และในข้อสอบค่าย

counting sort เรียงด้วยการนับ

ในหน้า 5.3 เรานับว่าคะแนนแต่ละค่ามีกี่คนด้วยอาเรย์นับ แล้ววนตามค่าคะแนนจาก 0 ถึง 100 เพื่อพิมพ์ผล ถ้าตอนพิมพ์ เราพิมพ์คะแนน s ซ้ำ count[s] ครั้ง ก็จะได้คะแนนทุกคนเรียงจากน้อยไปมาก โดยไม่ได้เทียบค่าสองค่าเลยสักครั้ง วิธีนี้เรียกว่า counting sort

#include <stdio.h>

int count[101];

int main()
{
    int n;
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        int score;
        scanf("%d", &score);
        count[score]++; // (1)!
    }
    int printed = 0;
    for (int value = 0; value <= 100; value++) { // (2)!
        for (int c = 0; c < count[value]; c++) {
            if (printed > 0) {
                printf(" ");
            }
            printf("%d", value);
            printed++;
        }
    }
    printf("\n");
    return 0;
}
  1. รอบแรกอ่านแล้วนับ ไม่ต้องเก็บคะแนนของแต่ละคนไว้เลย
  2. รอบสองวนตามค่า จากน้อยไปมาก พิมพ์ค่านั้นซ้ำตามจำนวนที่นับได้ ถ้าอยากได้จากมากไปน้อย ให้วน value จาก 100 ลงมาถึง 0

counting sort ใช้ได้เมื่อไร

  • ค่าเป็นจำนวนเต็มและช่วงแคบ อาเรย์นับต้องมีหนึ่งช่องต่อหนึ่งค่าที่เป็นไปได้ คะแนน 0 ถึง 100 ใช้ 101 ช่อง อายุ 0 ถึง 150 ใช้ 151 ช่อง แต่ถ้าค่าไปถึง 109 ต้องใช้พันล้านช่อง ทั้งประกาศไม่ได้และวนไม่ทัน ให้ใช้ qsort
  • ค่าติดลบก็ได้ ถ้าเลื่อนก่อน แบบหน้า 5.3 เช่น อุณหภูมิ -50 ถึง 50 องศา ใช้ช่อง t + 50 ตอนพิมพ์ค่อยลบ 50 กลับ
  • เรียงแค่ตัวเลข ถ้าต้องเรียงข้อมูลที่มีอย่างอื่นติดมาด้วย เช่น ชื่อกับคะแนน counting sort แบบนี้ทำไม่ได้ เพราะเหลือแค่จำนวน ไม่รู้ว่าใครได้คะแนนนั้น ใช้ qsort กับอาเรย์ของ struct แทน
#include <stdio.h>

int count[101];

int main()
{
    int n;
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        int t;
        scanf("%d", &t);
        count[t + 50]++;
    }
    int printed = 0;
    for (int index = 0; index <= 100; index++) {
        for (int c = 0; c < count[index]; c++) {
            if (printed > 0) {
                printf(" ");
            }
            printf("%d", index - 50);
            printed++;
        }
    }
    printf("\n");
    return 0;
}

ค้นหาแบบไล่

ลานจอดรถจดเลขบัตรจอดของรถทุกคันไว้ตามลำดับที่เข้ามา เจ้าของรถยื่นบัตรมา อยากรู้ว่ารถคันนั้นอยู่ในรายการไหม วิธีที่ตรงที่สุดคือไล่ดูทีละช่องตั้งแต่ช่องแรก เจอเมื่อไรก็ตอบเลย ไล่จนสุดแล้วไม่เจอก็คือไม่มี วิธีนี้เรียกว่า ค้นหาแบบไล่ (linear search) ใช้ได้กับข้อมูลที่ไม่ได้เรียง

#include <stdio.h>

int ticket[1000];

int linear_search(int a[], int n, int target)
{
    for (int i = 0; i < n; i++) {
        if (a[i] == target) {
            return i; // (1)!
        }
    }
    return -1; // (2)!
}

int main()
{
    int n;
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        scanf("%d", &ticket[i]);
    }
    int target;
    while (scanf("%d", &target) == 1) {
        int index = linear_search(ticket, n, target);
        if (index == -1) {
            printf("%d not found\n", target);
        } else {
            printf("%d found at index %d\n", target, index);
        }
    }
    return 0;
}
  1. เจอแล้ว return ออกจากฟังก์ชันทันที ช่องที่เหลือไม่ต้องดู จึงได้ตำแหน่งแรกที่เจอเสมอ
  2. มาถึงบรรทัดนี้ได้แปลว่าไล่ครบทุกช่องแล้วไม่เจอ คืน -1 เพราะไม่มีช่องไหนที่ index ติดลบ คนเรียกจึงแยกออกว่าไม่เจอ

    printf("count = %d, last = %d\n", count, last);
    return 0;
}

ค้นหาชื่อในอาเรย์ของ struct

ค้นหาด้วยข้อความก็ไล่แบบเดียวกัน แค่เทียบด้วย strcmp แทน == ตัวอย่างนี้คือเมนูของโรงอาหาร ค้นหาราคาจากชื่อเมนู

#include <stdio.h>
#include <string.h>

struct dish {
    char name[21];
    int price;
};

struct dish menu[100];

int find_dish(struct dish list[], int n, const char *name)
{
    for (int i = 0; i < n; i++) {
        if (strcmp(list[i].name, name) == 0) {
            return i;
        }
    }
    return -1;
}

int main()
{
    int n;
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        scanf("%s %d", menu[i].name, &menu[i].price);
    }
    char order[21];
    while (scanf("%s", order) == 1) {
        int index = find_dish(menu, n, order);
        if (index == -1) {
            printf("%s: no such dish\n", order);
        } else {
            printf("%s: %d baht\n", order, menu[index].price);
        }
    }
    return 0;
}

ค้นหาแบบไล่ใช้เวลาเท่าไร

หนึ่งคำถาม ไล่ไม่เกิน n ช่อง ถ้า n เป็นแสนก็ยังเร็วมาก ปัญหาอยู่ที่โจทย์ที่มีหลายคำถาม เช่น รายการ n = 100,000 ตัว และคำถาม Q = 100,000 ข้อ กรณีที่แย่ที่สุดคือทุกคำถามไม่เจอ ต้องไล่ครบทุกครั้ง รวม n × Q = 1010 ครั้ง ตามหลักในหน้า 4.5 ไม่มีทางทัน 1 วินาที ทางออกมีสองแบบ

  • ถ้าค่าเป็นจำนวนเต็มช่วงแคบ ใช้อาเรย์ seen หรืออาเรย์นับแบบหน้า 5.3 ตอบแต่ละคำถามได้ทันทีโดยไม่ต้องไล่
  • ถ้าค่าใหญ่ เรียงข้อมูลด้วย qsort ก่อนครั้งเดียว แล้วใช้ binary search ซึ่งค้นในข้อมูลที่เรียงแล้วโดยตัดครึ่งไปทีละรอบ ข้อมูลแสนตัวใช้ไม่เกิน 17 รอบต่อคำถาม จะเรียนในบทที่ 19 นี่คืออีกเหตุผลหนึ่งที่ต้องเรียง

ค้นหาคำในประโยค

อยากรู้ว่าคำว่า the อยู่ตรงไหนบ้างในประโยค ให้ลองวางคำไว้ใต้ประโยคที่ตำแหน่งเริ่ม i = 0 เทียบทีละตัวอักษร ถ้าตรงครบทุกตัวก็คือเจอ แล้วเลื่อนคำไปทางขวาหนึ่งช่อง ทำซ้ำจนคำเลยท้ายประโยค ประโยคยาว n ตัว คำยาว m ตัว ตำแหน่งเริ่มสุดท้ายที่เป็นไปได้คือ n - m

#include <stdio.h>
#include <string.h>

char text[1005], word[105];

void strip_newline(char s[])
{
    for (int i = 0; s[i] != '\0'; i++) {
        if (s[i] == '\n' || s[i] == '\r') {
            s[i] = '\0';
        }
    }
}

int main()
{
    fgets(text, sizeof text, stdin);
    fgets(word, sizeof word, stdin);
    strip_newline(text); // (1)!
    strip_newline(word);
    int n = strlen(text);
    int m = strlen(word);
    int count = 0;
    for (int i = 0; i + m <= n; i++) { // (2)!
        int j = 0;
        while (j < m && text[i + j] == word[j]) { // (3)!
            j++;
        }
        if (j == m) {
            printf("found at %d\n", i);
            count++;
        }
    }
    printf("count = %d\n", count);
    return 0;
}
  1. ตัด '\n' กับ '\r' ที่ fgets ติดมาด้วยแบบหน้า 6.3 ทั้งประโยคและคำ ลืมตัดของคำแล้วจะไม่เจออะไรเลย ดูในส่วนลองทำให้พัง
  2. เริ่มได้ถึง i = n - m เขียนเป็น i + m <= n จะเห็นชัดว่าคำทั้งคำยังอยู่ในประโยค ความยาวเก็บไว้ในตัวแปรก่อน ไม่เรียก strlen ในเงื่อนไขลูปตามหน้า 6.4
  3. เทียบไปเรื่อยๆ จนกว่าจะเจอตัวที่ไม่ตรง ถ้าหลุดลูปตอน j == m แปลว่าตรงครบทุกตัว

#include <stdio.h>
#include <string.h>

void strip_newline(char s[])
{
    for (int i = 0; s[i] != '\0'; i++) {
        if (s[i] == '\n' || s[i] == '\r') {
            s[i] = '\0';
        }
    }
}

int main()
{
    fgets(text, sizeof text, stdin);
    fgets(word, sizeof word, stdin);
    strip_newline(text);
    strip_newline(word);
    int n = strlen(text);
    int m = strlen(word);
    int count = 0;
    for (int i = 0; i + m <= n; i++) {
        int j = 0;
        while (j < m && text[i + j] == word[j]) {
            j++;
        }
        int left_ok = (i == 0 || text[i - 1] == ' ');
        int right_ok = (i + m == n || text[i + m] == ' ');
        if (j == m && left_ok && right_ok) {
            printf("found at %d\n", i);
            count++;
        }
    }
    printf("count = %d\n", count);
    return 0;
}

นับซ้อนกันหรือไม่ซ้อน

หา ana ใน banana เจอที่ตำแหน่ง 1 และ 3 สองครั้งนี้ใช้ตัว a ตรงกลางร่วมกัน ถ้าโจทย์บอกว่า "นับโดยไม่ให้ซ้อนกัน" เช่น ตัดคำออกจากแถบกระดาษ ตัดแล้วตัวอักษรนั้นหายไป พอเจอแล้วต้องกระโดดข้ามไปทั้งคำ คือเพิ่ม i อีก m แทน 1

#include <stdio.h>
#include <string.h>

char text[1005], word[105];

int matches_at(int i, int m)
{
    for (int j = 0; j < m; j++) {
        if (text[i + j] != word[j]) {
            return 0;
        }
    }
    return 1;
}

int main()
{
    while (scanf("%s %s", text, word) == 2) {
        int n = strlen(text);
        int m = strlen(word);
        int overlapping = 0;
        for (int i = 0; i + m <= n; i++) {
            if (matches_at(i, m)) {
                overlapping++;
            }
        }
        int separate = 0;
        int i = 0;
        while (i + m <= n) {
            if (matches_at(i, m)) {
                separate++;
                i += m; // (1)!
            } else {
                i++;
            }
        }
        printf("%s in %s: overlapping %d, separate %d\n", word, text, overlapping, separate);
    }
    return 0;
}
  1. เจอแล้วข้ามทั้งคำ ไม่เจอเลื่อนทีละช่อง การนับแบบไม่ซ้อนจึงต้องใช้ while ที่เลือกได้ว่าจะเพิ่ม i เท่าไร

strstr ใน string.h

string.h มีฟังก์ชันหาข้อความให้ชื่อ strstr(text, word) คืนที่อยู่ของตำแหน่งแรกที่เจอ หรือ NULL ถ้าไม่เจอ เอาที่อยู่ที่ได้ลบกับที่อยู่ของช่องแรก ก็ได้เป็นตำแหน่ง แบบการลบ pointer ในบทที่ 10 ถ้าจะหาต่อ ให้เริ่มหาใหม่จากตำแหน่งถัดไป

#include <stdio.h>
#include <string.h>

char text[1005], word[105];

int main()
{
    scanf("%s %s", text, word);
    int count = 0;
    char *found = strstr(text, word);
    while (found != NULL) {
        printf("found at %d\n", (int)(found - text));
        count++;
        found = strstr(found + 1, word);
    }
    printf("count = %d\n", count);
    return 0;
}

แบบที่ 2 ลืมตัด '\n' ของคำ

ในโปรแกรม find_word ลบบรรทัด strip_newline(word); ออก

#include <stdio.h>
#include <string.h>

char text[1005], word[105];

void strip_newline(char s[])
{
    for (int i = 0; s[i] != '\0'; i++) {
        if (s[i] == '\n' || s[i] == '\r') {
            s[i] = '\0';
        }
    }
}

int main()
{
    fgets(text, sizeof text, stdin);
    fgets(word, sizeof word, stdin);
    strip_newline(text);
    int n = strlen(text);
    int m = strlen(word);
    int count = 0;
    for (int i = 0; i + m <= n; i++) {
        int j = 0;
        while (j < m && text[i + j] == word[j]) {
            j++;
        }
        if (j == m) {
            printf("found at %d\n", i);
            count++;
        }
    }
    printf("count = %d\n", count);
    return 0;
}
