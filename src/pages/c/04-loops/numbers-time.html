4.5 ลูปกับตัวเลขและเวลาที่ใช้

ในบทที่ 1 เราแยกหลักได้แค่เลขที่รู้จำนวนหลักล่วงหน้า มีลูปแล้วทำได้กับเลขกี่หลักก็ได้ หน้านี้รวมงานตัวเลขที่ออกสอบบ่อย คือ แยกหลัก กลับเลข หาตัวหาร และตรวจจำนวนเฉพาะ แล้วปิดท้ายด้วยเรื่องที่สำคัญพอๆ กับเขียนให้ถูก คือเขียนให้ทันเวลา เราจะจับเวลาลูปจริงบนเครื่อง แล้วใช้ตัวเลขนั้นตัดสินว่าวิธีไหนผ่านโจทย์ที่จำกัดเวลา 1 วินาที

แยกหลักทีละหลัก

หลักการเดิมจากหน้า 1.2 % 10 หยิบหลักขวาสุด / 10 ตัดหลักขวาสุดทิ้ง ทำซ้ำไปเรื่อยๆ จนเลขเหลือ 0 ก็ได้ครบทุกหลัก เราไม่รู้ล่วงหน้าว่าต้องวนกี่รอบ จึงใช้ while

#include <stdio.h>

int main()
{
    int n;
    scanf("%d", &n);
    int digits = 0, digit_sum = 0;
    while (n > 0) {
        digit_sum += n % 10;
        digits++;
        n /= 10;
    }
    printf("digits = %d\n", digits);
    printf("sum = %d\n", digit_sum);
    return 0;
}

ลูปได้หลักออกมาจากขวาไปซ้าย และทำให้ n กลายเป็น 0 ตอนจบ ถ้ายังต้องใช้เลขเดิมหลังลูป ให้เก็บสำเนาไว้ในตัวแปรอื่นก่อน ถ้าเลขใหญ่ถึง 1018 ก็ใช้วิธีเดียวกัน แค่เปลี่ยน n เป็น long long และรับด้วย %lld

เลข 0 มีกี่หลัก

ป้อน 0 ให้โปรแกรมข้างบน จะได้ digits = 0 เพราะเงื่อนไข 0 > 0 เป็นเท็จตั้งแต่แรก ลูปไม่ทำเลยสักรอบ แต่ 0 เขียนได้ด้วยหลักเดียว คำตอบที่ถูกคือ 1 นี่คืองานที่ต้องทำอย่างน้อยหนึ่งครั้ง ตรงกับ do while ในหน้า 4.1

#include <stdio.h>

int main()
{
    int n;
    scanf("%d", &n);
    int digits = 0;
    do {
        digits++;
        n /= 10;
    } while (n > 0);
    printf("digits = %d\n", digits);
    return 0;
}

เลขอื่นที่ไม่ใช่ 0 ได้คำตอบเท่าเดิมทั้งสองแบบ ต่างกันแค่ 0 ตัวเดียว ชุดทดสอบชอบใส่ 0 ไว้ดักตรงนี้

กลับเลข

สร้างเลขใหม่จากหลักที่หยิบออกมา ทุกครั้งที่ได้หลักใหม่ เลื่อนเลขเดิมไปทางซ้ายหนึ่งหลักด้วยการคูณ 10 แล้วบวกหลักใหม่ต่อท้าย เลขที่อ่านกลับหลังแล้วได้เลขเดิม เช่น 12321 เรียกว่า palindrome

#include <stdio.h>

int main()
{
    int n;
    scanf("%d", &n);
    int original = n;
    int reversed = 0;
    while (n > 0) {
        reversed = reversed * 10 + n % 10;
        n /= 10;
    }
    printf("reversed = %d\n", reversed);
    if (reversed == original) {
        printf("palindrome\n");
    } else {
        printf("not palindrome\n");
    }
    return 0;
}

หาตัวหาร

ตัวหารของ n คือเลขที่หาร n ลงตัว วิธีตรงไปตรงมาคือลองทุกเลขตั้งแต่ 1 ถึง n แล้วดูว่า n % i เป็น 0 ไหม

#include <stdio.h>

int main()
{
    int n;
    scanf("%d", &n);
    int count = 0;
    for (int i = 1; i <= n; i++) {
        if (n % i == 0) {
            if (count > 0) {
                printf(" ");
            }
            printf("%d", i);
            count++;
        }
    }
    printf("\n");
    printf("count = %d\n", count);
    return 0;
}

ตรวจจำนวนเฉพาะ

จำนวนเฉพาะ (prime) คือจำนวนเต็มที่มากกว่า 1 และมีตัวหารแค่ 1 กับตัวมันเอง เช่น 2 3 5 7 11 ตรวจได้ด้วยการลองหารด้วย 2 ถึง n - 1 ถ้ามีตัวไหนหารลงตัวก็ไม่ใช่จำนวนเฉพาะ และเจอตัวเดียวก็พอ ใช้ break ได้เลย

#include <stdio.h>

int main()
{
    int n;
    scanf("%d", &n);
    int smallest = 0;
    for (int i = 2; i < n; i++) {
        if (n % i == 0) {
            smallest = i;
            break;
        }
    }
    if (n < 2) {
        printf("%d is not prime\n", n);
    } else if (smallest == 0) {
        printf("%d is prime\n", n);
    } else {
        printf("%d is not prime, divisible by %d\n", n, smallest);
    }
    return 0;
}

91

91 is not prime, divisible by 7

smallest เป็นทั้ง flag และเก็บตัวหารที่เจอ ถ้ายังเป็น 0 แปลว่าไม่มีอะไรหารลงตัว ส่วนเงื่อนไข n < 2 แยกไว้เพราะ 0 กับ 1 ไม่ใช่จำนวนเฉพาะ แต่ลูปไม่ทำงานเลยสำหรับสองค่านี้ ถ้าไม่แยกจะตอบผิด

โปรแกรมนี้ถูก แต่ถ้า n เป็นจำนวนเฉพาะขนาดใหญ่ ลูปต้องวนเกือบ n รอบ เร็วพอไหม ต้องวัดดู

ลูปหนึ่งรอบใช้เวลาเท่าไร

โจทย์แข่งขันทุกข้อมีเวลาจำกัด ส่วนใหญ่ 1 วินาที ถ้าโปรแกรมทำงานนานกว่านั้น grader จะตัดจบแล้วให้ผล Time Limit Exceeded แม้คำตอบจะถูก เราจึงต้องรู้ว่าใน 1 วินาทีคอมพิวเตอร์ทำได้กี่รอบ วิธีที่ดีที่สุดคือจับเวลาเอง

ไลบรารี time.h มีฟังก์ชัน clock() ทำหน้าที่เหมือนนาฬิกาจับเวลา เรียกก่อนและหลังส่วนที่อยากวัด แล้วเอามาลบกัน หารด้วย CLOCKS_PER_SEC ก็ได้เป็นวินาที ชนิด clock_t คือชนิดของค่าที่ clock() ส่งกลับ ตอนนี้ใช้ตามแบบไปก่อนได้

#include <stdio.h>
#include <time.h>

int main()
{
    int n;
    scanf("%d", &n);
    clock_t start = clock();
    long long sum = 0;
    for (int i = 1; i <= n; i++) {
        sum += i;
    }
    clock_t end = clock();
    printf("sum = %lld\n", sum);
    printf("time = %.3f seconds\n", (double)(end - start) / CLOCKS_PER_SEC);
    return 0;
}

หาตัวหารให้เร็วขึ้น

ตัวหารมาเป็นคู่เสมอ ถ้า i หาร n ลงตัว n / i ก็หาร n ลงตัวด้วย เช่น 36 มีคู่ 1 กับ 36, 2 กับ 18, 3 กับ 12, 4 กับ 9 และ 6 กับ 6 ในแต่ละคู่ ตัวที่เล็กกว่าไม่มีทางเกิน √n เพราะถ้าทั้งสองตัวมากกว่า √n ผลคูณจะเกิน n จึงลองแค่ i ที่ i * i <= n แล้วนับทั้ง i และคู่ของมัน ระวังกรณีที่ i กับ n / i เป็นเลขเดียวกัน อย่างคู่ 6 กับ 6 ต้องนับครั้งเดียว

#include <stdio.h>

int main()
{
    long long n;
    scanf("%lld", &n);
    int count = 0;
    for (long long i = 1; i * i <= n; i++) {
        if (n % i == 0) {
            count++;
            if (i != n / i) {
                count++;
            }
        }
    }
    printf("count = %d\n", count);
    return 0;
}

  • บนเครื่องที่เขียนเอกสารนี้ บวก 108 ครั้งใช้ 0.039 วินาที แต่ลูป 109 รอบที่มีการหารใช้ 1.2 วินาที
