6.1 ตัวอักษรคือตัวเลข

ในหน้า 1.4 น้องเห็นแล้วว่า char เก็บรหัส ASCII ของตัวอักษร 'B' + 1 ได้ 'C' และ '7' - '0' ได้ 7 หน้านี้เอาความจริงข้อเดียวนี้ไปทำงานจริง คือดูว่าตัวอักษรเป็นชนิดไหน เปลี่ยนตัวพิมพ์ใหญ่เล็ก แปลงตัวอักษรตัวเลขเป็นค่าจริง และเลื่อนตัวอักษรแบบวนรอบ ทั้งหมดนี้คือชิ้นส่วนที่โจทย์สตริงทุกข้อในหน้าต่อๆ ไปต้องใช้

char เป็นตัวเลขขนาดเล็ก

char คือจำนวนเต็มขนาด 1 byte ที่ printf แสดงเป็นตัวอักษรได้เมื่อใช้ %c เพราะเป็นจำนวนเต็ม จึงบวก ลบ เปรียบเทียบ และใช้เป็นตัวนับในลูปได้เหมือน int

#include <stdio.h>

int main()
{
    for (char letter = 'a'; letter <= 'z'; letter++) {
        printf("%c", letter);
    }
    printf("\n");
    for (char letter = 'Z'; letter >= 'A'; letter--) {
        printf("%c", letter);
    }
    printf("\n");
    return 0;
}

ดูว่าเป็นตัวอักษรชนิดไหน

หน้า 3.3 แยกตัวพิมพ์ใหญ่ ตัวพิมพ์เล็ก และตัวเลขด้วยเงื่อนไขช่วงอย่าง c >= 'a' && c <= 'z' ไปแล้ว ตอนนี้รู้เหตุผลแล้วว่าทำไมใช้ได้ คือรหัสของ 'a' ถึง 'z' เรียงติดกัน ตัวพิมพ์ใหญ่กับตัวเลขก็เช่นกัน โปรแกรมนี้ใช้เงื่อนไขชุดเดิม แต่อ่านทีละตัวไปจนหมดข้อมูล

#include <stdio.h>

int main()
{
    char c;
    while (scanf(" %c", &c) == 1) {
        if (c >= 'a' && c <= 'z') {
            printf("%c lowercase\n", c);
        } else if (c >= 'A' && c <= 'Z') {
            printf("%c uppercase\n", c);
        } else if (c >= '0' && c <= '9') {
            printf("%c digit\n", c);
        } else {
            printf("%c other\n", c);
        }
    }
    return 0;
}

ตัวอักษรตัวเลขกับค่าจริง

'7' - '0' ได้ 7 ทำให้อ่านตัวเลขทีละหลักแล้วประกอบเป็นค่าเองได้ วิธีคือทุกครั้งที่ได้หลักใหม่ ให้คูณค่าเดิมด้วย 10 แล้วบวกหลักใหม่ต่อท้าย

#include <stdio.h>

int main()
{
    char c;
    int year = 0;
    while (scanf("%c", &c) == 1 && c != '\n') {
        year = year * 10 + (c - '0');
    }
    printf("%d\n", year + 1);
    return 0;
}

เลื่อนตัวอักษรแบบวนรอบ

โจทย์รหัสลับชอบให้เลื่อนตัวอักษรไปข้างหน้า k ตัว โดยเลยจาก z แล้ววนกลับมา a ถ้าบวกตรงๆ จะเลยออกไปนอกตัวอักษร ต้องแปลงเป็นลำดับ 0 ถึง 25 ก่อน บวก k แล้วใช้ % 26 ให้วนกลับ แล้วค่อยแปลงกลับเป็นตัวอักษร

#include <stdio.h>

int main()
{
    char letter;
    int steps;
    scanf(" %c %d", &letter, &steps);
    printf("%c\n", letter + steps);
    int position = letter - 'a';
    position = (position + steps) % 26;
    printf("%c\n", 'a' + position);
    return 0;
}

ctype.h ทำให้ไม่ต้องเขียนช่วงเอง

การเช็กช่วงและเปลี่ยนตัวพิมพ์ใช้บ่อยจนภาษา C มีให้ใน ctype.h แล้ว ใส่ #include <ctype.h> ไว้บนโปรแกรมก็ใช้ได้

เรียกได้อะไรเขียนเองได้ว่า
isdigit(c)จริงถ้าเป็นตัวเลขc >= '0' && c <= '9'
isalpha(c)จริงถ้าเป็นตัวอักษรภาษาอังกฤษ ใหญ่หรือเล็กตัวพิมพ์เล็กหรือตัวพิมพ์ใหญ่
isupper(c)จริงถ้าเป็นตัวพิมพ์ใหญ่c >= 'A' && c <= 'Z'
islower(c)จริงถ้าเป็นตัวพิมพ์เล็กc >= 'a' && c <= 'z'
isspace(c)จริงถ้าเป็นช่องว่าง '\n' หรือตัวเว้นวรรคอื่นหลายตัว เขียนเองยาว
toupper(c)ตัวพิมพ์ใหญ่ของ c ถ้า c ไม่ใช่ตัวพิมพ์เล็กได้ c คืนมาเหมือนเดิมหัวข้อที่แล้ว
tolower(c)ตัวพิมพ์เล็กของ c ถ้า c ไม่ใช่ตัวพิมพ์ใหญ่ได้ c คืนมาเหมือนเดิมหัวข้อที่แล้ว
 
#include <stdio.h>
#include <ctype.h>

int main()
{
    char c;
    int letters = 0, digits = 0;
    while (scanf("%c", &c) == 1 && c != '\n') {
        if (isalpha(c)) {
            letters++;
        } else if (isdigit(c)) {
            digits++;
        }
        printf("%c", toupper(c));
    }
    printf("\n%d letters %d digits\n", letters, digits);
    return 0;
}

สรุป

  • char เป็นจำนวนเต็ม บวก ลบ เปรียบเทียบ และใช้เป็นตัวนับในลูปได้
  • เช็กชนิดด้วยช่วง เช่น c >= 'a' && c <= 'z' อย่ารวมสองช่วงเป็น 'A' ถึง 'z'
  • c - 'a' คือลำดับ 0 ถึง 25 บวก 'A' ได้ตัวพิมพ์ใหญ่ บวก k แล้ว % 26 ได้การเลื่อนแบบวนรอบ
  • c - '0' แปลงตัวอักษรตัวเลขเป็นค่า และ value * 10 + digit ประกอบหลักเป็นตัวเลข
  • ctype.h มี isdigit isalpha isupper islower isspace toupper tolower ใช้ใน if ตรงๆ อย่าเทียบกับ 1
