14.2 แทรก ลบ และกลับลำดับ

หน้าที่แล้วสร้างขบวนรถไฟได้ เพิ่มตู้ได้ทั้งหัวและท้าย หน้านี้ทำงานของลานสับเปลี่ยนตู้ คือพ่วงตู้ใหม่ไว้หลังตู้ที่กำหนด ปลดตู้ออกจากกลางขบวน และกลับหัวขบวน ทั้งสามงานใช้หลักเดียวกัน คือเปลี่ยนแค่ next ไม่กี่ตัว ไม่ย้ายข้อมูล ส่วนที่ยากคือลำดับการเปลี่ยน เปลี่ยนผิดลำดับนิดเดียว ตู้ครึ่งขบวนหายได้

ทุกโปรแกรมในหน้านี้ใช้ฟังก์ชันช่วยสามตัวเหมือนกัน read_list อ่าน N ตัวแล้วสร้างลิสต์ตามลำดับด้วยวิธีเพิ่มท้ายจากหน้า 14.1 print_list พิมพ์ทั้งลิสต์โดยไม่มีช่องว่างเกินท้ายบรรทัด และ free_list ตัวเดิม เวลาอ่านให้ข้ามสามตัวนี้ไปดูฟังก์ชันใหม่ของแต่ละหัวข้อได้เลย

แทรกหลังตู้ที่กำหนด

มีตู้ที่ p ชี้อยู่ จะพ่วงตู้ใหม่ไว้ข้างหลังมัน ต้องเปลี่ยนสองเส้น และลำดับสำคัญ

  1. ให้ตู้ใหม่เกี่ยวตู้ที่อยู่หลัง p ตอนนี้ คือ node->next = p->next;
  2. ให้ p เกี่ยวตู้ใหม่ คือ p->next = node;

เส้นประคือเส้นเดิมที่ถูกแทนที่ ถ้าทำเส้นที่ 2 ก่อน p->next จะเป็นตู้ใหม่ไปแล้ว และไม่มีใครจำที่อยู่ของตู้ 2 อีก ดูผลได้ในส่วนลองทำให้พัง

โปรแกรมนี้รับขบวน แล้วรับคู่ x y แปลว่าพ่วงตู้หมายเลข y ไว้หลังตู้หมายเลข x ใช้ find จากหน้า 14.1 หาตู้ x ก่อน

#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int value;
    struct Node *next;
} Node;

Node *read_list(int n)
{
    Node *head = NULL;
    Node *tail = NULL;
    for (int i = 0; i < n; i++) {
        Node *node = malloc(sizeof(Node));
        scanf("%d", &node->value);
        node->next = NULL;
        if (head == NULL) {
            head = node;
        } else {
            tail->next = node;
        }
        tail = node;
    }
    return head;
}

void print_list(Node *head)
{
    for (Node *p = head; p != NULL; p = p->next) {
        printf("%d", p->value);
        if (p->next != NULL) { // (1)!
            printf(" ");
        }
    }
    printf("\n");
}

void free_list(Node *head)
{
    while (head != NULL) {
        Node *next = head->next;
        free(head);
        head = next;
    }
}

Node *find(Node *head, int value)
{
    for (Node *p = head; p != NULL; p = p->next) {
        if (p->value == value) {
            return p;
        }
    }
    return NULL;
}

void insert_after(Node *p, int value) // (2)!
{
    Node *node = malloc(sizeof(Node));
    node->value = value;
    node->next = p->next;
    p->next = node;
}

int main()
{
    int n, q;
    scanf("%d %d", &n, &q);
    Node *head = read_list(n);
    for (int i = 0; i < q; i++) {
        int x, y;
        scanf("%d %d", &x, &y);
        Node *p = find(head, x);
        if (p != NULL) {
            insert_after(p, y);
        }
        print_list(head);
    }
    free_list(head);
    return 0;
}
  1. เว้นวรรคเฉพาะเมื่อยังมีตู้ต่อ บรรทัดจึงไม่มีช่องว่างเกินท้าย grader บางระบบเทียบทุกตัวอักษร ช่องว่างเกินตัวเดียวก็ผิดได้
  2. ฟังก์ชันนี้ไม่ต้องคืนหัวใหม่ เพราะแทรกหลัง p ไม่มีทางเปลี่ยนตู้หัว ถ้าอยากแทรกเป็นตู้แรกให้ใช้ push_front

ปลดตู้ออก

จะปลดตู้ที่มีหมายเลข x ออก ต้องให้ตู้ก่อนหน้ามันเกี่ยวข้ามไปหาตู้ที่อยู่หลังมันแทน ปัญหาคือ next ชี้ไปข้างหน้าทางเดียว ยืนอยู่ที่ตู้ x แล้วย้อนกลับไปหาตู้ก่อนหน้าไม่ได้ จึงต้องเดินด้วย prev ที่หยุดก่อนถึงตู้ x หนึ่งตู้ แล้วดู prev->next แทน

ตู้แรกเป็นกรณีพิเศษ เพราะไม่มีตู้ก่อนหน้า ถ้าตู้ที่จะปลดคือหัว ต้องย้าย head ไปตู้ที่สองแทน ฟังก์ชันจึงต้องคืนหัวใหม่แบบ push_front

#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int value;
    struct Node *next;
} Node;

Node *read_list(int n)
{
    Node *head = NULL;
    Node *tail = NULL;
    for (int i = 0; i < n; i++) {
        Node *node = malloc(sizeof(Node));
        scanf("%d", &node->value);
        node->next = NULL;
        if (head == NULL) {
            head = node;
        } else {
            tail->next = node;
        }
        tail = node;
    }
    return head;
}

void print_list(Node *head)
{
    if (head == NULL) {
        printf("empty\n");
        return;
    }
    for (Node *p = head; p != NULL; p = p->next) {
        printf("%d", p->value);
        if (p->next != NULL) {
            printf(" ");
        }
    }
    printf("\n");
}

void free_list(Node *head)
{
    while (head != NULL) {
        Node *next = head->next;
        free(head);
        head = next;
    }
}

Node *delete_value(Node *head, int value)
{
    if (head == NULL) { // (1)!
        return NULL;
    }
    if (head->value == value) { // (2)!
        Node *rest = head->next;
        free(head);
        return rest;
    }
    Node *prev = head;
    while (prev->next != NULL && prev->next->value != value) { // (3)!
        prev = prev->next;
    }
    if (prev->next != NULL) { // (4)!
        Node *target = prev->next;
        prev->next = target->next;
        free(target);
    }
    return head;
}

int main()
{
    int n, q;
    scanf("%d %d", &n, &q);
    Node *head = read_list(n);
    for (int i = 0; i < q; i++) {
        int x;
        scanf("%d", &x);
        head = delete_value(head, x); // (5)!
        print_list(head);
    }
    free_list(head);
    return 0;
}
  1. ลิสต์ว่างไม่มีอะไรให้ลบ และใช้ head->value ไม่ได้
  2. ตู้หัวคือตู้ที่จะลบ จำตู้ที่สองไว้ก่อน free แล้วคืนตู้ที่สองเป็นหัวใหม่
  3. เดินจนกว่าตู้ถัดจาก prev คือตู้ที่ต้องการ หรือหมดขบวน ลำดับของสองเงื่อนไขสำคัญ ถ้า prev->next เป็น NULL เงื่อนไขแรกเป็นเท็จ && จะไม่ประเมินเงื่อนไขที่สองเลยตามหน้า 3.3 จึงไม่ไปแตะ NULL->value
  4. ถ้าออกจากลูปเพราะหมดขบวน แปลว่าไม่มีตู้ x ไม่ต้องทำอะไร
  5. ห้ามลืม head = เพราะการลบตู้หัวทำให้หัวเปลี่ยน

กลับลำดับ

อยากกลับขบวน 1 2 3 4 เป็น 4 3 2 1 วิธีที่ง่ายที่สุดคือสร้างลิสต์ใหม่ด้วย push_front ทีละตัวตามหน้า 14.1 แต่ต้องขอ node ใหม่ทั้งชุด วิธีที่นิยมกว่าคือกลับทิศลูกศรของลิสต์เดิมทุกเส้น ไม่ต้องสร้าง node ใหม่เลย

เดินจากหัวด้วยสามตัวแปร prev คือตู้ที่กลับทิศแล้ว cur คือตู้ที่กำลังจะกลับ และ next จำตู้ถัดไปไว้ก่อนเส้นจะถูกเปลี่ยน แต่ละรอบทำสี่บรรทัด

Node *next = cur->next; // (1)!
cur->next = prev; // (2)!
prev = cur; // (3)!
cur = next;
  1. จำทางไปตู้ถัดไปไว้ก่อน บรรทัดต่อไปจะเขียนทับ cur->next
  2. กลับทิศ ให้ตู้นี้ชี้ย้อนไปหาตู้ก่อนหน้า
  3. เลื่อนทั้งสองตัวไปข้างหน้าหนึ่งตู้

ภาพนี้คือหลังรอบที่สอง ตู้ 1 กับ 2 กลับทิศแล้ว (สีส้ม) prev อยู่ที่ตู้ 2 ซึ่งเป็นหัวของส่วนที่กลับแล้ว cur อยู่ที่ตู้ 3 ซึ่งยังชี้ไปข้างหน้าตามเดิม พอ cur เป็น NULL ทุกตู้กลับครบ และ prev คือหัวใหม่

Node *reverse(Node *head)
{
    Node *prev = NULL; // (1)!
    Node *cur = head;
    while (cur != NULL) {
        Node *next = cur->next;
        cur->next = prev;
        prev = cur;
        cur = next;
    }
    return prev;
}

int main()
{

    Node *head = read_list(n);
    head = reverse(head);
    print_list(head);
    head = reverse(head); // (2)!
    print_list(head);
    free_list(head);
    return 0;
}
