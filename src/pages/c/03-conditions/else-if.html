3.2 else if และ if ซ้อน

if else เลือกได้สองทาง แต่โจทย์จริงมักมีมากกว่านั้น เกรดมี 8 ระดับ ค่าไฟคิดหลายอัตรา ตั๋วเครื่องเล่นขึ้นกับทั้งส่วนสูงและอายุ หน้านี้ต่อ if เป็นสายยาวด้วย else if และวาง if ไว้ใน if อีกชั้น แล้วใช้สองอย่างนี้ทำโจทย์แนวที่ออกสอบบ่อย คือตัดเกรด คิดค่าบริการแบบขั้นบันได และหาค่ามากสุดของสามตัว

เลือกหนึ่งจากหลายทาง

บอกสภาพอากาศจากอุณหภูมิ ต่ำกว่า 20 องศาคือ cold ตั้งแต่ 20 แต่ต่ำกว่า 30 คือ warm ตั้งแต่ 30 ขึ้นไปคือ hot

#include <stdio.h>

int main()
{
    int temp;
    scanf("%d", &temp);
    if (temp < 20) {
        printf("cold\n");
    } else if (temp < 30) {
        printf("warm\n");
    } else {
        printf("hot\n");
    }
    return 0;
}

else if ไม่ใช่คำสั่งใหม่ มันคือ else ที่ตามด้วย if อีกตัว จะต่อกี่ข้อก็ได้ ส่วน else ตัวท้ายจะมีหรือไม่มีก็ได้ ถ้าไม่มีแล้วไม่มีข้อไหนจริง ก็ไม่ทำอะไรเลย

ตัดเกรด

เกณฑ์ที่โรงเรียนส่วนใหญ่ใช้ คือ 80 ขึ้นไปได้ A ลดลงช่วงละ 5 คะแนน จนต่ำกว่า 50 ได้ F

#include <stdio.h>

int main()
{
    int score;
    scanf("%d", &score);
    if (score >= 80) {
        printf("A\n");
    } else if (score >= 75) {
        printf("B+\n");
    } else if (score >= 70) {
        printf("B\n");
    } else if (score >= 65) {
        printf("C+\n");
    } else if (score >= 60) {
        printf("C\n");
    } else if (score >= 55) {
        printf("D+\n");
    } else if (score >= 50) {
        printf("D\n");
    } else {
        printf("F\n");
    }
    return 0;
}

หลักการคือ เรียงจากเกณฑ์สูงลงต่ำ แต่ละข้อเขียนแค่ขอบล่าง เพราะขอบบนถูกข้อก่อนหน้ากันไว้แล้ว 77 ไม่ผ่าน >= 80 แต่ผ่าน >= 75 จึงได้ B+ แล้วหยุด ไม่ไปถึง >= 70 ทั้งที่ 77 ก็มากกว่า 70

เวลาทดสอบโปรแกรมแบบนี้ ให้ป้อนค่าที่ขอบของทุกช่วง คือ 80 กับ 79 ได้ A กับ B+ และ 50 กับ 49 ได้ D กับ F เกรดผิดส่วนใหญ่เกิดตรงขอบ เช่นเผลอเขียน > แทน >= ซึ่งทำให้ 80 พอดีได้ B+

if ซ้อน

block ของ if ใส่คำสั่งอะไรก็ได้ รวมถึง if อีกตัว เรียกว่า if ซ้อน (nested if) ใช้เมื่อต้องตัดสินเป็นขั้นๆ เครื่องเล่นในสวนสนุกให้ขึ้นได้เมื่อสูงตั้งแต่ 120 เซนติเมตร คนที่ขึ้นได้ ถ้าอายุต่ำกว่า 12 ปี ต้องมีผู้ใหญ่นั่งด้วย

#include <stdio.h>

int main()
{
    int age, height;
    scanf("%d %d", &age, &height);
    if (height >= 120) {
        if (age >= 12) {
            printf("ride alone\n");
        } else {
            printf("ride with adult\n");
        }
    } else {
        printf("too short\n");
    }
    return 0;
}

ใช้ไป 250 หน่วย ไม่ได้ แปลว่าจ่ายหน่วยละ 5 บาททั้ง 250 หน่วย แต่ 100 หน่วยแรกจ่ายหน่วยละ 3 อีก 100 หน่วยถัดมาจ่ายหน่วยละ 4 และ 50 หน่วยที่เหลือจ่ายหน่วยละ 5

#include <stdio.h>

int main()
{
    int units;
    scanf("%d", &units);
    int bill;
    if (units <= 100) {
        bill = units * 3;
    } else if (units <= 200) {
        bill = 100 * 3 + (units - 100) * 4;
    } else {
        bill = 100 * 3 + 100 * 4 + (units - 200) * 5;
    }
    printf("%d\n", bill);
    return 0;
}

หาค่ามากสุดของสามตัว

หาค่ามากสุดของสองตัวใช้ if else ตรงๆ ได้ แต่พอเป็นสามตัว ถ้าเทียบทุกคู่จะต้องเขียน if ซ้อนหลายชั้น วิธีที่ง่ายกว่าคือ สมมติให้ตัวแรกเป็นค่ามากสุดไปก่อน แล้วเอาตัวที่เหลือมาท้าชิงทีละตัว ตัวไหนมากกว่าแชมป์ปัจจุบัน ก็ขึ้นเป็นแชมป์แทน

#include <stdio.h>

int main()
{
    int a, b, c;
    scanf("%d %d %d", &a, &b, &c);
    int largest = a;
    if (b > largest) {
        largest = b;
    }
    if (c > largest) {
        largest = c;
    }
    printf("%d\n", largest);
    return 0;
}

แบบที่ 3 else ห้อยผิดตัว

โปรแกรม ride ที่เขียนโดยไม่ใส่ปีกกา และตั้งใจให้ else เป็นของเงื่อนไขส่วนสูง

#include <stdio.h>

int main()
{
    int age, height;
    scanf("%d %d", &age, &height);
    if (height >= 120)
        if (age >= 12)
            printf("ride alone\n");
    else
        printf("too short\n");
    return 0;
}

สรุป

  • if ... else if ... else ตรวจจากบนลงล่าง ทำแค่ block แรกที่จริง แล้วข้ามที่เหลือ
  • ตัดเกรดหรือแบ่งช่วง ให้เรียงเกณฑ์ตามลำดับ แต่ละข้อเขียนแค่ขอบเดียว อีกขอบถูกข้อก่อนหน้ากันไว้แล้ว
  • if ซ้อนใช้ตัดสินเป็นขั้น ใส่ปีกกาทุกชั้น เพราะ else จับคู่กับ if ที่ใกล้ที่สุด
  • ค่าบริการขั้นบันไดคิดแต่ละขั้นด้วยราคาของขั้นนั้น ตรวจด้วยค่าสองฝั่งของทุกขอบ
  • ค่ามากสุดของหลายตัว ตั้งตัวแรกเป็นแชมป์ แล้วใช้ if แยกให้ตัวที่เหลือท้าชิงทีละตัว
  • ทดสอบให้ครบทุกกิ่ง และทุกค่าตรงขอบของแต่ละช่วง
