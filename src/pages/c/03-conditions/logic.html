3.3 ตรรกะ และ หรือ ไม่

เงื่อนไขในชีวิตจริงมักมีหลายข้อพร้อมกัน สมัครค่ายได้ถ้าอายุไม่เกิน 18 และ สอบคัดเลือกผ่าน ตอบ y หรือ Y ก็นับว่าตกลง คะแนนที่ ไม่ อยู่ในช่วง 0 ถึง 100 ถือว่าผิด หน้านี้รู้จักตัวดำเนินการตรรกะสามตัวที่ใช้ต่อเงื่อนไขเข้าด้วยกัน วิธีเขียนช่วงของตัวเลขให้ถูก และใช้ทั้งหมดตรวจปีอธิกสุรทิน ซึ่งเป็นโจทย์คลาสสิกของเรื่องนี้

&& และ

A && B เป็นจริงเมื่อ ทั้ง A และ B จริง ข้อใดข้อหนึ่งเท็จก็เท็จทั้งหมด

#include <stdio.h>

int main()
{
    int age, score;
    scanf("%d %d", &age, &score);
    if (age <= 18 && score >= 60) {
        printf("accepted\n");
    } else {
        printf("rejected\n");
    }
    return 0;
}

ถ้าป้อน 19 75 หรือ 16 40 จะได้ rejected ทั้งคู่ เพราะเงื่อนไขข้อหนึ่งไม่ผ่าน ทั้งหมดนี้เขียนด้วย if ซ้อนแบบหน้า 3.2 ก็ได้ แต่ && สั้นกว่าเมื่อทั้งสองข้อนำไปสู่ผลเดียวกัน

|| หรือ

A || B เป็นจริงเมื่อ อย่างน้อยหนึ่ง ใน A หรือ B จริง จะเท็จก็ต่อเมื่อเท็จทั้งคู่ เครื่องหมาย | อยู่บนปุ่มเดียวกับ \ กด Shift+\ สองครั้ง

#include <stdio.h>

int main()
{
    char answer;
    scanf(" %c", &answer);
    if (answer == 'y' || answer == 'Y') {
        printf("yes\n");
    } else {
        printf("no\n");
    }
    return 0;
}

#include <stdio.h>

หยุดเมื่อรู้คำตอบแล้ว

C ประเมิน && กับ || จากซ้ายไปขวา และ หยุดทันทีที่รู้คำตอบ เรียกว่า short-circuit evaluation

  • A && B ถ้า A เท็จ ผลเป็นเท็จแน่นอน ไม่ประเมิน B เลย
  • A || B ถ้า A จริง ผลเป็นจริงแน่นอน ไม่ประเมิน B เลย

นิสัยนี้ใช้เป็นด่านกันพังได้ แบ่งลูกอมให้เพื่อนเท่าๆ กันได้ไหม ต้องหารลงตัว แต่ถ้าจำนวนคนเป็น 0 การใช้ % จะพังตามที่เห็นในหน้า 1.2

#include <stdio.h>

int main()
{
    int total, people;
    scanf("%d %d", &total, &people);
    if (people != 0 && total % people == 0) {
        printf("fair\n");
    } else {
        printf("not fair\n");
    }
    return 0;
}

ปีอธิกสุรทิน

ปีอธิกสุรทินคือปีที่เดือนกุมภาพันธ์มี 29 วัน ทั้งปีจึงมี 366 วัน กฎตามปฏิทินสากลมีสามข้อ

  1. ปีที่หารด้วย 4 ลงตัว เป็นปีอธิกสุรทิน
  2. ยกเว้นปีที่หารด้วย 100 ลงตัว ไม่เป็น
  3. ยกเว้นอีกชั้น ปีที่หารด้วย 400 ลงตัว เป็น

รวบเป็นประโยคเดียวได้ว่า หาร 4 ลงตัว และ หาร 100 ไม่ลงตัว หรือ หาร 400 ลงตัว

#include <stdio.h>

int main()
{

    scanf("%d", &year);
    if ((year % 4 == 0 && year % 100 != 0) || year % 400 == 0) {
        printf("leap\n");
    } else {
        printf("common\n");
    }
    return 0;
}

ค่ามากสุดคือ 5 แต่ได้ 3 เพราะ a กับ b เท่ากัน ไม่มีใคร มากกว่า อีกตัว ทั้งสองข้อแรกจึงเท็จ ตกไปที่ else ซึ่งเดาว่าเป็น c ทั้งที่ c น้อยที่สุด ชุดทดสอบที่ตัวเลขไม่ซ้ำกันเลยจะผ่านหมด แต่ตกชุดที่มีค่าเท่ากัน แก้ได้ด้วยการเปลี่ยนเป็น >= แต่วิธีตั้งแชมป์ในหน้า 3.2 ไม่มีปัญหานี้ตั้งแต่ต้น

สรุป

  • && จริงเมื่อจริงทั้งคู่ || จริงเมื่อจริงอย่างน้อยหนึ่ง ! กลับค่าความจริง
  • ช่วงเขียนเป็น low <= x && x <= high นอกช่วงเขียนเป็น x < low || x > high ห้ามเขียน low <= x <= high
  • ทั้งสองข้างของ && และ || ต้องเป็นเงื่อนไขเต็ม c == 'y' || c == 'Y'
  • ! ทำก่อนการเปรียบเทียบ และ && ทำก่อน || ผสมกันเมื่อไรใส่วงเล็บ
  • && กับ || หยุดเมื่อรู้คำตอบ วางเงื่อนไขกันพังไว้ซ้าย เช่น b != 0 && a % b == 0
  • ปีอธิกสุรทินคือ (y % 4 == 0 && y % 100 != 0) || y % 400 == 0 ทดสอบด้วย 2024 2023 1900 2000
  • ทศนิยมห้ามเทียบด้วย == ให้ดูว่าผลต่างอยู่ในช่วงแคบรอบ 0 หรือเลี่ยงไปใช้จำนวนเต็ม
