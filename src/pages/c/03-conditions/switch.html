3.4 switch และ ? :

เมื่อต้องเอาตัวแปรตัวเดียวไปเทียบกับค่าตายตัวหลายค่า เช่น เลขวันเป็นชื่อวัน เลขเดือนเป็นไตรมาส ตัวอักษรเกรดเป็นแต้ม else if ทำได้ แต่ต้องเขียน day == ซ้ำทุกบรรทัด ภาษา C มีคำสั่ง switch ที่เขียนเรื่องแบบนี้ได้เป็นระเบียบกว่า ท้ายหน้ามีตัวดำเนินการ ? : สำหรับเลือกค่าหนึ่งในสองค่าให้จบในบรรทัดเดียว

คำสั่ง switch

#include <stdio.h>

int main()
{
    int day;
    scanf("%d", &day);
    switch (day) { // (1)!
        case 1: // (2)!
            printf("Monday\n");
            break; // (3)!
        case 2:
            printf("Tuesday\n");
            break;
        case 3:
            printf("Wednesday\n");
            break;
        case 4:
            printf("Thursday\n");
            break;
        case 5:
            printf("Friday\n");
            break;
        case 6:
            printf("Saturday\n");
            break;
        case 7:
            printf("Sunday\n");
            break;
        default: // (4)!
            printf("invalid\n");
    }
    return 0;
}
  1. ค่าในวงเล็บคือค่าที่จะเอาไปเทียบ ต้องเป็นจำนวนเต็มหรือ char
  2. case ตามด้วยค่าคงที่หนึ่งค่า และเครื่องหมาย : ถ้า day เท่ากับค่านี้ โปรแกรมจะกระโดดมาเริ่มทำตรงนี้
  3. break คือออกจาก switch ไปทำบรรทัดถัดจากปีกกาปิดของมัน
  4. default ทำเมื่อไม่ตรงกับ case ไหนเลย เหมือน else ตัวท้าย จะมีหรือไม่มีก็ได้

ไม่มี break แล้วไหลต่อ

สิ่งที่ switch ต่างจาก else if มากที่สุด คือ case เป็นแค่ จุดเริ่มต้น ไม่ใช่ block ที่มีขอบ พอกระโดดมาที่ case ที่ตรงแล้ว โปรแกรมจะทำคำสั่งไล่ลงไปเรื่อยๆ ผ่าน case ตัวถัดไปด้วย จนกว่าจะเจอ break หรือถึงปีกกาปิดของ switch การไหลต่อแบบนี้เรียกว่า fall through

ส่วนใหญ่มันคือบั๊ก ดูแบบที่ 1 ในส่วนลองทำให้พัง แต่มีกรณีหนึ่งที่ใช้ประโยชน์ได้ คือให้หลายค่าทำสิ่งเดียวกัน โดยวาง case ติดกันไว้โดยไม่มีคำสั่งคั่น

switch กับตัวอักษร

char คือตัวเลข จึงใช้กับ switch ได้ ใส่ตัวอักษรในเครื่องหมายคำพูดเดี่ยวหลัง case ตัวอย่างนี้แปลงเกรดเป็นแต้ม และรับได้ทั้งตัวพิมพ์เล็กและพิมพ์ใหญ่ด้วยการวาง case คู่กัน

#include <stdio.h>

int main()
{

    scanf(" %c", &grade);

        case 'A':

            break;
        case 'B':
        case 'b':
            printf("3\n");
            break;
        case 'C':
        case 'c':
            printf("2\n");
            break;
        case 'D':
        case 'd':
            printf("1\n");
            break;
        case 'F':
        case 'f':
            printf("0\n");
            break;
        default:
            printf("unknown\n");
    }
    return 0;
}

switch ทำอะไรไม่ได้

switch เทียบได้แค่ว่า เท่ากับค่าคงที่ไหม เท่านั้น จึงมีข้อจำกัดที่ต้องรู้

  • เทียบช่วงไม่ได้ ตัดเกรดจากคะแนนด้วย case score >= 80: ไม่ได้ ต้องใช้ else if แบบหน้า 3.2
  • ใช้กับ double ไม่ได้ ค่าในวงเล็บต้องเป็นจำนวนเต็มหรือ char
  • ค่าหลัง case ต้องเป็นค่าคงที่ เช่น 5 หรือ 'A' ใส่ชื่อตัวแปรไม่ได้
  • ค่าหลัง case ห้ามซ้ำกัน

ถ้าลองฝ่าฝืน จะได้ error ตอนคอมไพล์ ดูได้ในส่วนลองทำให้พัง กฎง่ายๆ ในการเลือกคือ ค่าตายตัวไม่กี่ค่าใช้ switch ช่วงหรือเงื่อนไขซับซ้อนใช้ if

? : เลือกค่าแบบสั้น

บางครั้งเราแค่อยากได้ ค่า หนึ่งในสองค่าตามเงื่อนไข เช่น ค่าที่มากกว่าของสองตัว ถ้าใช้ if else ต้องเขียนห้าบรรทัด ตัวดำเนินการ ? : ทำได้ในนิพจน์เดียว

เงื่อนไข ? ค่าเมื่อจริง : ค่าเมื่อเท็จ

#include <stdio.h>

int main()
{
    int a, b;
    scanf("%d %d", &a, &b);
    int larger = a > b ? a : b;
    printf("larger = %d\n", larger);
    return 0;
}

เพราะ ? : ให้ผลเป็นค่า จึงวางไว้ใน printf ได้เลย ใช้บ่อยกับการเลือกคำให้ถูกกับจำนวน

#include <stdio.h>

int main()
{
    int count;
    scanf("%d", &count);
    printf("%d %s\n", count, count == 1 ? "item" : "items");
    return 0;
}

{

    scanf("%d", &day);

        case 1:
            printf("Monday\n");
        case 2:
            printf("Tuesday\n");
        case 3:
            printf("Wednesday\n");
            break;
        default:
            printf("other\n");
    }
    return 0;
}

สรุป

  • switch เทียบตัวแปรหนึ่งตัวกับค่าคงที่หลายค่า ใช้ได้กับจำนวนเต็มและ char
  • ทุก case ต้องจบด้วย break ไม่อย่างนั้นจะไหลต่อไป case ถัดไป และ -Wall ไม่เตือน
  • วาง case ติดกันโดยไม่มีคำสั่งคั่น เพื่อให้หลายค่าทำสิ่งเดียวกัน
  • default ทำเมื่อไม่ตรงกับ case ไหน เหมือน else ตัวท้าย
  • ช่วงตัวเลข ทศนิยม และเงื่อนไขที่มี && || ใช้ if
  • เงื่อนไข ? ก : ข ให้ค่า ก เมื่อจริง ข เมื่อเท็จ ใช้กับการเลือกค่าง่ายๆ เท่านั้น
