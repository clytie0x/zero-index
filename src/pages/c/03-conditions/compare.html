3.1 การเปรียบเทียบและ if

ทุกโปรแกรมในบทที่ 1 และ 2 ทำงานครบทุกบรรทัดจากบนลงล่าง ข้อมูลเข้าเป็นอะไรก็เดินทางเดิม แต่โจทย์ส่วนใหญ่ต้องการคำตอบที่ต่างกันตามข้อมูล เช่น คะแนนถึง 50 ให้พิมพ์ pass ไม่ถึงให้พิมพ์ fail หน้านี้มีสองเรื่องที่ทำให้โปรแกรมเลือกทางได้ คือการ เปรียบเทียบ ค่าสองค่า และคำสั่ง if ที่ใช้ผลเปรียบเทียบนั้นตัดสินว่าจะทำคำสั่งไหน

เปรียบเทียบสองค่า

ภาษา C มีตัวดำเนินการเปรียบเทียบ 6 ตัว

ตัวดำเนินการความหมายตัวอย่างที่เป็นจริง
==เท่ากับ ใช้เครื่องหมายเท่ากับสองตัว5 == 5
!=ไม่เท่ากับ5 != 3
<น้อยกว่า3 < 5
>มากกว่า5 > 3
<=น้อยกว่าหรือเท่ากับ5 <= 5
>=มากกว่าหรือเท่ากับ5 >= 5

ตัวที่มีสองเครื่องหมาย เขียนติดกันห้ามเว้นวรรค และ = อยู่หลังเสมอ อ่านตามลำดับที่เขียนว่า "น้อยกว่า หรือเท่ากับ" จะไม่สลับกัน

ผลของการเปรียบเทียบคือตัวเลข ลองพิมพ์ออกมาดู

#include <stdio.h>

int main()
{
    int score = 72;
    printf("%d\n", score > 50);
    printf("%d\n", score < 50);
    printf("%d\n", score >= 72);
    printf("%d\n", score <= 71);
    printf("%d\n", score == 72);
    printf("%d\n", score != 72);
    return 0;
}

= ตัวเดียว กับ == สองตัว

= ตัวเดียวคือ กำหนดค่า เอาค่าทางขวาใส่กล่องทางซ้าย ตามหน้า 1.1 ส่วน == สองตัวคือ ถามว่าเท่ากันไหม ไม่เปลี่ยนค่าอะไรเลย สองตัวนี้สลับกันแล้วยังคอมไพล์ผ่าน แต่ได้ผลผิด ดูได้ในส่วนลองทำให้พัง

คำสั่ง if

if แปลว่า "ถ้า" รูปแบบคือ

if (เงื่อนไข) {
    คำสั่งที่ทำเมื่อเงื่อนไขเป็นจริง
}

ถ้าเงื่อนไขเป็นจริง ทำคำสั่งในปีกกา ถ้าเป็นเท็จ ข้ามปีกกาทั้งก้อนไปเลย แล้วทำบรรทัดถัดจากปีกกาปิดต่อ

#include <stdio.h>

int main()
{
    int score;
    scanf("%d", &score);
    if (score >= 50) { // (1)!
        printf("pass\n"); // (2)!
    }
    printf("done\n"); // (3)!
    return 0;
}
  1. เงื่อนไขต้องอยู่ในวงเล็บเสมอ และ ไม่มี ; หลังวงเล็บปิด เพราะ if ยังไม่จบ ต้องมีคำสั่งตามมา
  2. คำสั่งในปีกกาเยื้องเข้าไป 4 ช่อง ให้เห็นว่าอยู่ใต้ if compiler ไม่สนการเยื้อง แต่คนอ่านสน
  3. บรรทัดนี้อยู่นอกปีกกา ทำทุกครั้งไม่ว่าเงื่อนไขจะเป็นอะไร

หลายคำสั่งในปีกกาเดียว

ของที่อยู่ในปีกกาเรียกว่า block (ก้อนคำสั่ง) ใส่กี่คำสั่งก็ได้ ทั้งหมดจะทำหรือไม่ทำไปพร้อมกัน ร้านค้าแห่งหนึ่งลดให้ 100 บาทเมื่อซื้อครบ 1000 บาท

#include <stdio.h>

int main()
{
    int price;
    scanf("%d", &price);
    if (price >= 1000) {
        printf("discount 100\n");
        price = price - 100;
    }
    printf("pay %d\n", price);
    return 0;
}

ถ้า block มีคำสั่งเดียว ภาษา C ยอมให้ไม่ใส่ปีกกา เช่น if (score >= 50) printf("pass\n"); แต่เอกสารนี้ ใส่ปีกกาทุกครั้ง เพราะวันหนึ่งที่เพิ่มคำสั่งที่สองเข้าไป คำสั่งนั้นจะไม่อยู่ใต้ if อย่างที่ตาเห็น ดูแบบที่ 3 ในส่วนลองทำให้พัง

if else ทางเลือกสองทาง

บ่อยครั้งเราต้องการให้ทำอย่างหนึ่งเมื่อจริง และอีกอย่างหนึ่งเมื่อเท็จ ใช้ else (มิฉะนั้น) ต่อท้าย if โปรแกรมจะเดินทางใดทางหนึ่งเท่านั้น ไม่มีทางทำทั้งสองก้อน และไม่มีทางข้ามทั้งสองก้อน

ตัวอย่างที่ใช้บ่อยที่สุด คือบอกว่าเป็นเลขคู่หรือเลขคี่ เลขคู่คือเลขที่หารด้วย 2 ลงตัว หรือ n % 2 == 0 ตามที่เห็นในหน้า 1.2

#include <stdio.h>

int main()
{
    int n;
    scanf("%d", &n);
    if (n % 2 == 0) {
        printf("even\n");
    } else {
        printf("odd\n");
    }
    return 0;
}

else ไม่มีเงื่อนไขของตัวเอง มันคือ "กรณีที่เหลือทั้งหมด" ตำแหน่งของ else ต้องต่อจากปีกกาปิดของ if ทันที เอกสารนี้เขียน } else { ไว้บรรทัดเดียวกัน

ตัวอย่าง ส่วนต่างที่ไม่ติดลบ

น้องสองคนสูง 150 กับ 163 เซนติเมตร สูงต่างกันกี่เซนติเมตร ถ้าเอาคนแรกลบคนที่สองจะได้ -13 แต่ส่วนต่างควรเป็น 13 ไม่ว่าจะป้อนใครก่อน วิธีคือลบไปก่อน แล้วถ้าผลติดลบ ค่อยกลับเครื่องหมาย ค่าที่ได้เรียกว่า ค่าสัมบูรณ์ (absolute value)

#include <stdio.h>

int main()
{
    int first, second;
    scanf("%d %d", &first, &second);
    int diff = first - second;
    if (diff < 0) {
        diff = -diff;
    }
    printf("%d\n", diff);
    return 0;
}

เปรียบเทียบตัวอักษร

จากหน้า 1.4 char คือตัวเลขรหัส ASCII จึงเปรียบเทียบด้วยตัวดำเนินการชุดเดียวกันได้เลย

#include <stdio.h>

int main()
{
    char answer;
    scanf(" %c", &answer);
    if (answer == 'y') {
        printf("start\n");
    } else {
        printf("stop\n");
    }
    return 0;
}

start
  • ตัวอักษรต้องอยู่ในเครื่องหมายคำพูดเดี่ยว 'y' เสมอ ถ้าเขียน "y" จะเป็นข้อความ เทียบกับ char ไม่ได้ compiler จะเตือนว่า comparison between pointer and integer
  • ตัวพิมพ์ใหญ่กับพิมพ์เล็กมีรหัสต่างกัน 'Y' คือ 89 ส่วน 'y' คือ 121 ป้อน Y โปรแกรมนี้จึงตอบ stop วิธีรับได้ทั้งสองแบบอยู่ในหน้า 3.3
  • < กับ > ใช้กับตัวอักษรได้ด้วย 'a' < 'b' เป็นจริงเพราะ 97 น้อยกว่า 98 เรื่องนี้ใช้ตรวจว่าเป็นตัวพิมพ์ใหญ่หรือไม่ในหน้า 3.3

ค่าอะไรนับว่าจริง

if ไม่ได้รับแค่ผลเปรียบเทียบ รับตัวเลขอะไรก็ได้ กฎคือ 0 เป็นเท็จ ค่าอื่นทุกค่าเป็นจริง รวมถึงเลขติดลบด้วย

int main()
{

แบบที่ 4 ตรวจเลขคี่ด้วย == 1

เลขคี่หาร 2 เหลือเศษ 1 จึงน่าจะเขียนแบบนี้ได้

#include <stdio.h>

int main()
{
    int n;
    scanf("%d", &n);
    if (n % 2 == 1) {
        printf("odd\n");
    } else {
        printf("even\n");
    }
    return 0;
}

-7 เป็นเลขคี่ แต่โปรแกรมบอกว่าคู่ จำจากหน้า 1.2 ได้ไหมว่าเศษจาก % มีเครื่องหมายตามตัวตั้ง -7 % 2 จึงได้ -1 ไม่ใช่ 1 ไม่มีคำเตือนอะไรเลย ชุดทดสอบที่มีแต่เลขบวกก็ผ่านหมด ทางที่ปลอดภัยคือเทียบกับ 0 เสมอ n % 2 == 0 เป็นคู่ n % 2 != 0 เป็นคี่ ใช้ได้ทั้งเลขบวกและลบ

compiler อ่าน = เป็นการกำหนดค่า แล้วเจอ < ตามมาโดยไม่มีค่าอยู่ข้างหน้า ข้อความไม่ได้บอกว่าสลับเครื่องหมาย แต่ ^ ชี้ตรงจุดนั้นพอดี จำแค่ว่า = อยู่หลังเสมอ

สรุป

  • เปรียบเทียบด้วย == != < > <= >= ผลเป็น int จริงได้ 1 เท็จได้ 0
  • if (เงื่อนไข) { ... } ทำ block เมื่อเงื่อนไขจริง ถ้ามี else จะทำ block ใด block หนึ่งเสมอ
  • ใน if ค่า 0 คือเท็จ ค่าอื่นทุกค่าคือจริง
  • = คือกำหนดค่า == คือถามว่าเท่ากันไหม สลับกันแล้วได้แค่คำเตือน
  • ห้ามมี ; หลังวงเล็บเงื่อนไข และใส่ปีกกาทุกครั้ง
  • ตรวจคู่คี่ด้วย n % 2 == 0 หรือ != 0 ไม่ใช่ == 1
  • ทดสอบทุก if ด้วยค่าที่ทำให้จริง ค่าที่ทำให้เท็จ และค่าตรงขอบ
