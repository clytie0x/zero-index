12.1 mod และคำตอบที่ใหญ่เกิน

น้องใช้ % มาตั้งแต่บทที่ 1 ทั้งแยกหลัก ตรวจเลขคู่ และแปลงเวลา หน้านี้ต่อจากตรงนั้นสามเรื่อง เรื่องแรกคือ % กับเลขติดลบที่ให้ผลไม่เหมือนในห้องเรียนคณิตศาสตร์ เรื่องที่สองคือโจทย์ที่คำตอบใหญ่เกิน long long แล้วสั่งว่า "ตอบเป็นเศษจากการหารด้วย 1000000007" ซึ่งเจอบ่อยมากในโจทย์นับจำนวนวิธี และเรื่องสุดท้ายคือยกกำลังเลขชี้กำลังระดับ 1018 ให้เสร็จในเสี้ยววินาที

หารและ mod เมื่อมีเลขติดลบ

ในหน้า 1.2 เราเห็นแล้วว่าเลขติดลบกับ % ต้องระวัง คราวนี้ดูให้ครบทุกแบบของเครื่องหมาย

#include <stdio.h>

int main()
{
    printf("%d %d\n", 7 / 3, 7 % 3);
    printf("%d %d\n", -7 / 3, -7 % 3);
    printf("%d %d\n", 7 / -3, 7 % -3);
    printf("%d %d\n", -7 / -3, -7 % -3);
    return 0;
}

2 1
-2 -1
-2 1
2 -1

ภาษา C ตั้งแต่มาตรฐาน C99 เป็นต้นมา ซึ่งรวมถึง GCC ที่เราใช้ ทำตามกฎสองข้อนี้ทุกเครื่อง

  1. ผลหารตัดทศนิยมทิ้งไปทาง 0 -7 / 3 คือ -2.33 ตัดเหลือ -2 ไม่ใช่ -3
  2. เศษมีเครื่องหมายตามตัวตั้ง ตัวตั้งคือเลขที่อยู่หน้า % ตัวตั้งติดลบ เศษก็ติดลบหรือเป็น 0

สองข้อนี้มาคู่กันเพื่อให้ความสัมพันธ์ a == (a / b) * b + a % b เป็นจริงเสมอ ลองแทน a = -7 กับ b = 3 จะได้ (-2) × 3 + (-1) = -7 พอดี

แต่ในวิชาคณิตศาสตร์ เศษต้องอยู่ระหว่าง 0 ถึง b - 1 เสมอ -7 หารด้วย 3 จึงเขียนเป็น -7 = 3 × (-3) + 2 เศษคือ 2 ไม่ใช่ -1 โจทย์ที่ถามเรื่องวนรอบ เช่น วันในสัปดาห์ ชั่วโมงบนหน้าปัดนาฬิกา หรือที่นั่งรอบโต๊ะกลม ต้องการเศษแบบคณิตศาสตร์ ถ้าใช้ % ตรงๆ กับเลขติดลบจะได้ตำแหน่งที่ไม่มีอยู่จริง

ทำให้ mod เป็นบวก

ให้วันอาทิตย์เป็นวันที่ 0 จันทร์เป็น 1 ไปจนถึงเสาร์เป็น 6 วันนี้วันพุธ (3) ถามว่า 12 วันก่อนเป็นวันอะไร

#include <stdio.h>

int main()
{
    int today, shift;
    scanf("%d %d", &today, &shift);
    printf("plain %%  : %d\n", (today + shift) % 7);
    printf("fixed     : %d\n", ((today + shift) % 7 + 7) % 7);
    return 0;
}

จังหวะค่าเริ่ม -9ค่าเริ่ม 15
x % 7 ได้ค่าระหว่าง -6 ถึง 6-21
+ 7 ได้ค่าระหว่าง 1 ถึง 1358
% 7 อีกรอบ ได้ค่าระหว่าง 0 ถึง 651

% 7 รอบสองมีไว้สำหรับค่าที่เป็นบวกอยู่แล้ว ถ้าไม่มี 15 จะกลายเป็น 8 ซึ่งเกิน 6 ไป วันที่ 5 คือวันศุกร์ ตรวจด้วยมือได้ว่า 14 วันก่อนเป็นวันพุธ อีกสองวันหลังจากนั้นคือวันศุกร์ จำรูปนี้ไว้ใช้ทุกครั้งที่ค่าอาจติดลบ

((x % m) + m) % m    ได้ค่าระหว่าง 0 ถึง m - 1 เสมอ

ทำไมโจทย์ให้ตอบ mod 1000000007

ห้องหนึ่งมีนักเรียน N คน ครูจะเลือกกลุ่มหนึ่งไปแข่งขัน จะเลือกกี่คนก็ได้ นักเรียนแต่ละคนมีสองทาง ไปหรือไม่ไป จำนวนแบบทั้งหมดจึงเป็น 2 × 2 × ... × 2 = 2N แบบ ถ้า N = 100 คำตอบคือ 1267650600228229401496703205376 ยาว 31 หลัก ส่วน long long เก็บได้แค่ 19 หลัก

โจทย์นับจำนวนวิธีแทบทุกข้อเจอปัญหานี้ คนตั้งโจทย์จึงสั่งว่า "ตอบเป็นเศษจากการหารด้วย 1000000007" (ก็คือ 109 + 7 ซึ่งโจทย์มักเขียนแบบนี้) ตัวเลขนี้ถูกเลือกเพราะ

  • ค่าที่ mod แล้วอยู่ระหว่าง 0 ถึง 1000000006 เก็บใน int ได้
  • สองค่านั้นบวกกัน ได้ไม่เกิน 2000000012 ยังไม่เกิน int ที่เก็บได้ถึง 2147483647
  • สองค่านั้นคูณกัน ได้ไม่เกินราว 1018 ยังไม่เกิน long long ที่เก็บได้ราว 9.2 × 1018
  • เป็นจำนวนเฉพาะ ซึ่งมีประโยชน์กับเทคนิคขั้นสูงที่อยู่นอกหลักสูตรนี้ ตอนนี้ไม่ต้องใช้

บางโจทย์ใช้ 998244353 แทน ทุกอย่างในหน้านี้ใช้ได้เหมือนกัน

ตั้งชื่อให้ค่าคงที่ด้วย #define

เลข 1000000007 จะโผล่หลายที่ในโปรแกรม พิมพ์ผิดศูนย์เดียวก็ได้คำตอบผิดทั้งข้อ จึงตั้งชื่อให้มันครั้งเดียวที่บรรทัดบนสุด

#define MOD 1000000007

#define สั่งให้ compiler เอา 1000000007 ไปแทนคำว่า MOD ทุกที่ก่อนคอมไพล์ ไม่มีเครื่องหมาย = และไม่มี ; ท้ายบรรทัด ชื่อแบบนี้นิยมเขียนเป็นตัวพิมพ์ใหญ่ทั้งหมด จะได้รู้ว่าเป็นค่าคงที่ ไม่ใช่ตัวแปร

บวก ลบ คูณ แบบ mod

ความลับที่ทำให้ไม่ต้องเก็บเลข 31 หลักคือ ถ้าสุดท้ายต้องการแค่เศษ ระหว่างทางก็เก็บแค่เศษได้ เหมือนเข็มสั้นของนาฬิกา เดินไปกี่ชั่วโมงก็ตาม สิ่งที่บอกตำแหน่งเข็มคือจำนวนชั่วโมงหารด้วย 12 เหลือเศษเท่าไร ไม่ต้องจำว่าเดินมาแล้วกี่รอบ เขียนเป็นกฎได้แบบนี้ เมื่อ m คือ 1000000007

(a + b) % m  ==  ((a % m) + (b % m)) % m
(a * b) % m  ==  ((a % m) * (b % m)) % m
(a - b) % m  ==  ((a % m) - (b % m) + m) % m

ในโค้ดแปลว่า mod ทันทีหลังบวกหรือคูณทุกครั้ง ค่าในตัวแปรจะอยู่ระหว่าง 0 ถึง m - 1 ตลอดเวลา ผลคูณของสองค่าแบบนี้ไม่เกิน 1018 เก็บใน long long ได้เสมอ นับแบบเลือกนักเรียนไปแข่งได้แบบนี้

#include <stdio.h>
#define MOD 1000000007

int main()
{
    int n;
    scanf("%d", &n);
    long long ways = 1;
    for (int i = 0; i < n; i++) {
        ways = ways * 2 % MOD;
    }
    printf("%lld\n", ways);
    return 0;
}

ลบแล้วต้องบวก MOD กลับ

รหัสผ่านยาว N หลัก แต่ละหลักเป็น 0 1 หรือ 2 ต้องมีเลข 2 อย่างน้อยหนึ่งตัว มีได้กี่แบบ วิธีคิดคือนับทั้งหมด 3N แบบ แล้วลบแบบที่ไม่มีเลข 2 เลย ซึ่งมี 2N แบบ

#include <stdio.h>
#define MOD 1000000007

int main()
{
    int n;
    scanf("%d", &n);
    long long all = 1, no_two = 1;
    for (int i = 0; i < n; i++) {
        all = all * 3 % MOD;
        no_two = no_two * 2 % MOD;
    }
    printf("all     = %lld\n", all);
    printf("no two  = %lld\n", no_two);
    printf("wrong   = %lld\n", (all - no_two) % MOD);
    printf("correct = %lld\n", (all - no_two + MOD) % MOD);
    return 0;
}

ของจริง 329 มากกว่า 229 แน่นอน แต่หลัง mod แล้ว เศษของ 329 บังเอิญน้อยกว่า ลบกันจึงติดลบ เป็นปัญหาเดียวกับวันในสัปดาห์ข้างบน บวก MOD เข้าไปก่อน mod ทำให้ผลกลับมาอยู่ระหว่าง 0 ถึง MOD - 1 เพราะทั้งสองค่าอยู่ระหว่าง 0 ถึง MOD - 1 อยู่แล้ว ผลลบจึงไม่ต่ำกว่า -(MOD - 1) บวก MOD ครั้งเดียวก็พอ ชุดทดสอบเล็กๆ มักไม่ติดลบ โปรแกรมที่ลืมบวก MOD จึงผ่านตัวอย่างแต่ตกชุดใหญ่

หารไม่มีกฎแบบนี้

(10 / 2) % 3 ได้ 2 แต่ (10 % 3) / (2 % 3) คือ 1 / 2 ได้ 0 การหารใช้กฎ mod ระหว่างทางไม่ได้ ถ้าสูตรมีการหาร ให้หารให้เสร็จตอนตัวเลขยังเป็นค่าจริง ก่อนจะ mod วิธีทำดูได้ในหน้า 12.4

ยกกำลังแบบเร็ว

ถ้าโจทย์ถาม AN mod 1000000007 เมื่อ N ใหญ่ถึง 1018 ลูปคูณ N รอบแบบโปรแกรม pow2 ไม่มีทางทัน ตามหลักในหน้า 4.5 หนึ่งวินาทีทำได้ราว 108 รอบเท่านั้น

หน้า 9.2 เขียนยกกำลังแบบแบ่งครึ่งไปแล้ว ทบทวนสั้นๆ คือหา AN/2 ครั้งเดียวแล้วยกกำลังสอง ถ้า N เป็นเลขคี่คูณ A เพิ่มอีกตัว เลขชี้กำลังลดลงครึ่งหนึ่งทุกครั้งที่เรียก สิ่งที่หน้านี้เพิ่มมีสองอย่าง คือ mod ทุกครั้งที่คูณตามกฎในหัวข้อที่แล้ว และเลขชี้กำลังเป็น long long เพราะใหญ่ถึง 1018

#include <stdio.h>
#define MOD 1000000007

long long power(long long base, long long exp)
{
    if (exp == 0) { // (1)!
        return 1;
    }
    long long half = power(base, exp / 2); // (2)!
    long long result = half * half % MOD;
    if (exp % 2 == 1) {
        result = result * base % MOD;
    }
    return result;
}

int main()
{
    long long base, exp;
    scanf("%lld %lld", &base, &exp);
    printf("%lld\n", power(base % MOD, exp)); // (3)!
    return 0;
}

  2. เรียกตัวเองครั้งเดียวแล้วใช้ half ซ้ำ เหตุผลเดียวกับแบบที่ 1 ในส่วนลองทำให้พังของหน้า 9.2
  3. mod ฐานก่อน ฐานที่รับเข้ามาอาจใหญ่ถึง 1018 ถ้าไม่ลดก่อน result * base จะล้น long long

ลองทำให้พัง

#include <stdio.h>

int main()
{

            odd++;
        }

{

        ways = ways * 2;

    printf("%lld\n", ways % MOD);
    return 0;

ค่าที่ mod แล้วไม่เกิน 1000000006 เก็บใน int ได้ หลายคนจึงประกาศ answer ในโปรแกรม factorial เป็น int

#include <stdio.h>
#define MOD 1000000007

int main()
{
    int n;
    scanf("%d", &n);
    int answer = 1;
    for (int i = 1; i <= n; i++) {
        answer = answer * i % MOD;
    }
    printf("%d\n", answer);
    return 0;
}

  • อยากได้เศษแบบคณิตศาสตร์ระหว่าง 0 ถึง m - 1 ใช้ ((x % m) + m) % m ตรวจเลขคี่ใช้ x % 2 != 0
  • คำตอบใหญ่เกินให้ตอบ mod 1000000007 ใช้ #define MOD 1000000007 แล้ว mod ทุกครั้งหลังบวกหรือคูณ
