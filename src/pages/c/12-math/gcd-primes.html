12.2 ห.ร.ม. ค.ร.น. และจำนวนเฉพาะ

ห.ร.ม. กับ ค.ร.น. ใช้กับโจทย์แบ่งของให้ลงตัว และโจทย์หาเวลาที่หลายเหตุการณ์มาตรงกัน ส่วนจำนวนเฉพาะเป็นฐานของโจทย์ตัวเลขแทบทุกแบบ หน้านี้ใช้ ห.ร.ม. แบบยุคลิดที่น้องเขียนเป็นแล้วในบทที่ 9 มาหา ค.ร.น. ที่ไม่ล้น แล้วต่อจากการตรวจจำนวนเฉพาะในหน้า 4.5 ไปสู่ตะแกรงที่หาจำนวนเฉพาะทั้งหมดถึงสิบล้านได้ในเวลาไม่ถึงหนึ่งในสามสิบวินาที ปิดด้วยการแยกตัวประกอบ

ห.ร.ม. แบบยุคลิด

ทบทวนสั้นๆ ห.ร.ม. แบบยุคลิดเขียนเป็นลูปในหน้า 8.5 และเป็น recursion ในหน้า 9.2 หน้านี้ใช้แบบ recursion ที่รับ long long ตามโปรแกรม lights ข้างล่าง เลขขนาด 1018 ก็ยังใช้ไม่ถึงร้อยรอบ เพราะทุกสองรอบตัวเลขเหลือไม่ถึงครึ่ง

ถ้ามีหลายตัว ให้หา ห.ร.ม. ทีละคู่ไปเรื่อยๆ g = gcd(g, x) เพราะ ห.ร.ม. ของสามตัวเท่ากับ ห.ร.ม. ของ (ห.ร.ม. ของสองตัวแรก) กับตัวที่สาม

ค.ร.น. ต้องหารก่อนคูณ

ไฟสองดวงเปิดพร้อมกันที่วินาทีที่ 0 ดวงแรกกะพริบทุก a วินาที ดวงที่สองทุก b วินาที จะกะพริบพร้อมกันอีกครั้งตอนไหน คำตอบคือ ค.ร.น. (ตัวคูณร่วมน้อย least common multiple หรือ lcm) ของ a กับ b

มีความจริงข้อหนึ่งว่า ห.ร.ม. × ค.ร.น. = a × b เช่น 12 กับ 18 มี ห.ร.ม. 6 และ ค.ร.น. 36 ซึ่ง 6 × 36 = 216 = 12 × 18 ค.ร.น. จึงเท่ากับ a × b / ห.ร.ม. แต่ในโปรแกรมให้เขียนเป็น a / g * b คือหารก่อนแล้วค่อยคูณ เหตุผลอยู่ในส่วนลองทำให้พัง

#include <stdio.h>

long long gcd(long long a, long long b)
{
    if (b == 0) {
        return a;
    }
    return gcd(b, a % b);
}

int main()
{
    long long a, b;
    scanf("%lld %lld", &a, &b);
    long long g = gcd(a, b);
    printf("gcd = %lld\n", g);
    printf("lcm = %lld\n", a / g * b);
    return 0;
}

ค.ร.น. ของหลายตัวแบบสะสม

ค.ร.น. ของหลายตัวทำแบบสะสมเหมือน ห.ร.ม. เริ่มจาก l = 1 แล้วรวมเลขเข้าไปทีละตัวด้วย l = l / gcd(l, x) * x ซึ่งก็คือ ค.ร.น. ของ l กับ x นั่นเอง ลองกับ 6 10 15 4

ถ้าต้องรู้ว่าทุกเลขเป็นจำนวนเฉพาะไหม

หน้า 4.5 ตรวจเลขตัวเดียวด้วยการลองหาร 2 ถึง √n ตัวละไม่เกิน √n รอบ ถ้าโจทย์ถามว่าเลขตั้งแต่ 1 ถึง 107 มีจำนวนเฉพาะกี่ตัว หรือถามหลายแสนคำถามว่าเลขนั้นเลขนี้เป็นจำนวนเฉพาะไหม การตรวจทีละตัวต้องทำซ้ำเป็นสิบล้านครั้ง ข้างล่างจะวัดให้เห็นว่าช้าแค่ไหน ทางออกคือตะแกรงเอราทอสเทนีส (sieve of Eratosthenes) ซึ่งหาคำตอบของทุกเลขพร้อมกันในรอบเดียว

  • เริ่มขีดที่ i × i พหุคูณที่เล็กกว่านั้น เช่น 3 × 2 = 6 ถูกจำนวนเฉพาะที่เล็กกว่าขีดไปแล้ว
  • หยุดเมื่อ i × i เกิน n เพราะจุดเริ่มขีดเกิน n แล้ว ไม่มีอะไรให้ขีด เหตุผลเดียวกับการลองหารถึง √n
#include <stdio.h>

char is_composite[10000001]; // (1)!

int main()
{
    int n;
    scanf("%d", &n);
    for (int i = 2; i * i <= n; i++) {
        if (!is_composite[i]) { // (2)!
            for (int j = i * i; j <= n; j += i) {
                is_composite[j] = 1;
            }
        }
    }
    int count = 0;
    for (int i = 2; i <= n; i++) {
        if (!is_composite[i]) {
            count++;
        }
    }
    printf("%d\n", count);
    return 0;
}
  1. ช่องที่ i เป็น 1 แปลว่า i ถูกขีดฆ่าแล้ว คือเป็นจำนวนประกอบ (composite) อาเรย์อยู่นอก main ตามหน้า 5.5 จึงเริ่มต้นเป็น 0 ทุกช่อง ใช้ char เพราะเก็บแค่ 0 กับ 1 ช่องละ 1 byte สิบล้านช่องใช้ราว 10 MB ถ้าเป็น int จะใช้ 40 MB
  2. ขีดเฉพาะพหุคูณของเลขที่ยังรอดอยู่ ถ้า i ถูกขีดไปแล้ว พหุคูณของ i ก็ถูกตัวประกอบของ i ขีดไปหมดแล้วเหมือนกัน

ตรวจเลขไม่กี่ตัว หรือเลขใหญ่เกินกว่าจะสร้างอาเรย์ เช่น 1012 ใช้ไล่หารถึง √n ตรวจเลขจำนวนมากที่ไม่เกินราว 107 สร้างตะแกรงครั้งเดียวแล้วตอบทุกคำถามจากอาเรย์

แยกตัวประกอบ

ทุกจำนวนเต็มที่มากกว่า 1 เขียนเป็นผลคูณของจำนวนเฉพาะได้แบบเดียว เช่น 360 = 23 × 32 × 5 วิธีแยกคือลองหารด้วย p = 2, 3, 4, ... ตัวไหนหารลงตัวก็หารซ้ำจนหารไม่ลง แล้วนับว่าหารไปกี่ครั้ง

#include <stdio.h>

int main()
{
    long long n;
    scanf("%lld", &n);
    for (long long p = 2; p * p <= n; p++) {
        if (n % p == 0) {
            int power = 0;
            while (n % p == 0) {
                n /= p;
                power++;
            }
            printf("%lld^%d\n", p, power);
        }
    }
    if (n > 1) {
        printf("%lld^1\n", n);
    }
    return 0;
}

คำถามที่น้องน่าจะสงสัยมีสามข้อ

  1. p = 4 หรือ 6 ไม่ใช่จำนวนเฉพาะ จะพิมพ์ออกมาไหม ไม่ เพราะตอนถึง p = 4 ตัวประกอบ 2 ถูกหารออกจาก n หมดแล้ว 4 จึงหาร n ไม่ลงตัว ตัวที่หารลงตัวได้จึงเป็นจำนวนเฉพาะเสมอ
  2. เงื่อนไข p * p <= n ใช้ n ที่เล็กลงเรื่อยๆ ได้หรือ ได้ และยิ่งดี n ที่เหลือไม่มีตัวประกอบที่เล็กกว่า p แล้ว ถ้ามันยังมีตัวประกอบสองตัวขึ้นไป ตัวที่เล็กสุดต้องไม่เกิน √n ลูปจึงหยุดเร็วขึ้นเมื่อ n ลดลง
  3. บรรทัดที่ 17 มีไว้ทำไม พอลูปจบ ถ้า n ยังเหลือมากกว่า 1 ที่เหลือคือจำนวนเฉพาะตัวใหญ่ที่เกิน √ ของ n ตั้งต้น มีได้อย่างมากตัวเดียว เช่น 2000006 = 2 × 1000003 ลูปหาร 2 ออกแล้วเหลือ 1000003 ซึ่ง 1001 × 1001 เกินแล้ว ลูปจึงจบโดยไม่เจอมัน

เลข 1012 วนไม่เกิน 106 รอบ เสร็จทันที ลองป้อน 999999999989 ซึ่งเป็นจำนวนเฉพาะ ลูปจะวนจนสุดทาง ได้ 999999999989^1 และป้อน 1000000000000 ได้ 2^12 กับ 5^12

ลองทำให้พัง

แบบที่ 1 คูณก่อนหารใน ค.ร.น.

เขียนโปรแกรมไฟกะพริบด้วย int เพราะโจทย์บอกว่า a กับ b ไม่เกิน 105 และคำตอบรับประกันว่าไม่เกิน int แล้วคำนวณ ค.ร.น. ทั้งสองแบบ

#include <stdio.h>

int gcd(int a, int b)
{
    if (b == 0) {
        return a;
    }
    return gcd(b, a % b);
}

int main()
{
    int a, b;
    scanf("%d %d", &a, &b);
    int g = gcd(a, b);
    printf("gcd = %d\n", g);
    printf("multiply first = %d\n", a * b / g);
    printf("divide first   = %d\n", a / g * b);
    return 0;
}

แบบที่ 3 ลืมตัวประกอบที่เหลือ

ตัดสามบรรทัด if (n > 1) ท้ายโปรแกรม factorize ออก แล้วป้อน 2000006

#include <stdio.h>

int main()
{
    long long n;
    scanf("%lld", &n);
    for (long long p = 2; p * p <= n; p++) {
        if (n % p == 0) {
            int power = 0;
            while (n % p == 0) {
                n /= p;
                power++;
            }
            printf("%lld^%d\n", p, power);
        }
    }
    return 0;
}
