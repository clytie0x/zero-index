10.4 สตริงกับ char *

สตริงในบทที่ 6 คืออาเรย์ของ char ที่ปิดท้ายด้วย '\0' พอรู้ว่าชื่ออาเรย์คือที่อยู่ของช่องแรก ก็จะเห็นว่าฟังก์ชันสตริงทุกตัว ตั้งแต่ printf("%s") จนถึง strlen ทำงานกับ char * คือที่อยู่ของตัวอักษรตัวหนึ่ง แล้วเดินไปเรื่อยๆ จนเจอ '\0' หน้านี้ใช้ความรู้นี้ตัดสตริงเป็นท่อน ลบตัวอักษรออกจากสตริงโดยไม่ใช้อาเรย์ที่สอง และอธิบายคำเตือนกับผลแปลกๆ ของสตริงที่ค้างไว้ตั้งแต่บทที่ 2 และบทที่ 6

%s รับที่อยู่ แล้วพิมพ์จนเจอ \0

printf("%s", word) ไม่ได้รับทั้งอาเรย์ มันได้แค่ที่อยู่ของ word[0] ซึ่งมีชนิด char * แล้วพิมพ์ทีละตัวไปจนเจอ '\0' ถ้าส่งที่อยู่ของช่องอื่นไป มันก็เริ่มพิมพ์จากช่องนั้น

#include <stdio.h>

int main()
{
    char word[] = "programming";
    printf("%s\n", word);
    printf("%s\n", word + 3);
    printf("%c\n", *(word + 3));
    for (char *p = word + 7; *p != '\0'; p++) {
        printf("%s\n", p);
    }
    return 0;
}

  • บรรทัดที่ 8 *(word + 3) คือตัวอักษรในช่องนั้นหนึ่งตัว ชนิด char จึงพิมพ์ด้วย %c ความต่างระหว่าง char * กับ char คือความต่างระหว่าง "ข้อความที่เริ่มตรงนี้" กับ "ตัวอักษรตัวนี้"
  • ลูปบรรทัดที่ 9 เลื่อน p ทีละตัว ทุกครั้งที่พิมพ์ %s จึงได้ส่วนท้ายที่สั้นลงทีละตัว ลูปหยุดเมื่อ *p เป็น '\0' แบบเดียวกับที่ printf ทำข้างใน

scanf("%s", word) ก็เหมือนกัน มันต้องการที่อยู่ของช่องแรกที่จะเริ่มเขียน ชื่อ word ให้ที่อยู่นั้นอยู่แล้ว จึงไม่ต้องมี & ตามที่หน้า 2.3 บอกไว้

&word คือที่อยู่ของทั้งอาเรย์

หน้า 2.3 และหน้า 6.2 ลองใส่ & หน้าชื่ออาเรย์ใน scanf แล้วได้คำเตือนว่า '%s' expects argument of type 'char *', but argument 2 has type 'char (*)[105]' ทั้งที่โปรแกรมยังทำงานถูก ลองพิมพ์ที่อยู่ดู

#include <stdio.h>

int main()
{
    char word[105] = "pointer";
    printf("word      = %p\n", word);
    printf("&word     = %p\n", &word);
    printf("word + 1  = %p\n", word + 1);
    printf("&word + 1 = %p\n", &word + 1);
    return 0;
}

word กับ &word เป็นเลขเดียวกัน เพราะอาเรย์เริ่มที่ช่องแรก แต่ชนิดต่างกัน word คือที่อยู่ของ char หนึ่งตัว (char *) ส่วน &word คือที่อยู่ของทั้งอาเรย์ 105 ช่อง ชนิดของมันเขียนว่า char (*)[105] อ่านว่า pointer ไปยังอาเรย์ของ char 105 ช่อง ความต่างโผล่ตอนบวก 1 word + 1 ขยับ 1 byte ไปตัวอักษรถัดไป แต่ &word + 1 ขยับไปทั้งก้อน จาก BC0 เป็น C29 ซึ่งห่างกัน 69 ในฐานสิบหก คือ 105 พอดี

scanf เขียนตัวอักษรทีละตัวเริ่มจากที่อยู่ที่ได้ เลขที่อยู่ตรงกันจึงบังเอิญทำงานถูก แต่ชนิดผิด compiler จึงเตือน และถ้าเอาไปบวกเลขแบบในตัวอย่างข้างบน ความต่างของชนิดจะได้ที่อยู่ผิดไปทันที เขียนชื่ออาเรย์เปล่าๆ ทุกครั้ง

เดินบนสตริงด้วย pointer

ฟังก์ชันที่รับสตริงจึงรับเป็น char * ได้ แล้วเดินด้วย pointer จนเจอ '\0' ไม่ต้องรับความยาวเพราะสตริงบอกจุดจบเอง ตัวอย่างนี้เขียน strlen เอง และนับตัวพิมพ์ใหญ่

#include <stdio.h>

int length(char *s)
{
    char *p = s;
    while (*p != '\0') {
        p++;
    }
    return (int)(p - s); // (1)!
}

int count_upper(char *s)
{
    int count = 0;
    for (char *p = s; *p != '\0'; p++) {
        if (*p >= 'A' && *p <= 'Z') {
            count++;
        }
    }
    return count;
}

int main()
{
    char name[105];
    scanf("%s", name);
    printf("length %d, upper %d\n", length(name), count_upper(name));
    return 0;
}
  1. p หยุดที่ช่องของ '\0' ระยะห่างจากช่องแรกคือจำนวนตัวอักษรก่อน '\0' พอดี ใช้การลบ pointer จากหน้า 10.3

ฟังก์ชันที่คืนที่อยู่ในสตริง

อีเมลแบ่งเป็นชื่อผู้ใช้กับโดเมน คั่นด้วย @ ฟังก์ชัน find_char คืนที่อยู่ของตัวที่หาเจอตัวแรก หรือคืน NULL ถ้าไม่มี เมื่อได้ที่อยู่ของ @ แล้ว เขียน '\0' ทับตรงนั้น สตริงเดียวก็กลายเป็นสองท่อนทันที

#include <stdio.h>

char *find_char(char *s, char c) // (1)!
{
    for (char *p = s; *p != '\0'; p++) {
        if (*p == c) {
            return p;
        }
    }
    return NULL; // (2)!
}

int main()
{
    int n;
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        char email[105];
        scanf("%s", email);
        char *at = find_char(email, '@');
        if (at == NULL) {
            printf("%s is not an email\n", email);
        } else {
            *at = '\0'; // (3)!
            printf("user %s, domain %s\n", email, at + 1); // (4)!
        }
    }
    return 0;
}
  1. ชนิดที่คืนคือ char * ที่อยู่ที่คืนเป็นของช่องในอาเรย์ email ของ main ซึ่งยังอยู่หลังฟังก์ชันจบ จึงคืนได้ ต่างจากแบบที่ 4 ของหน้า 10.2
  2. เดินจนจบแล้วไม่เจอ คืน NULL ตามความหมายในหน้า 10.1
  3. แก้ช่องที่ at ชี้ จาก '@' เป็น '\0' สตริงที่เริ่มจาก email จึงจบตรงนี้
  4. โดเมนเริ่มที่ตัวถัดจาก @ และจบที่ '\0' เดิมท้ายอีเมล

pointer อ่านกับเขียนบนสตริงเดียวกัน

เบอร์โทร 081-234-5678 ต้องการเอาขีดออกให้เหลือแต่ตัวเลข วิธีหนึ่งคือคัดลอกตัวที่ไม่ใช่ขีดไปอีกอาเรย์ แต่ทำในอาเรย์เดิมได้ด้วย pointer สองตัว read เดินอ่านทุกตัว write ชี้ช่องถัดไปที่จะเขียน ตัวที่จะเก็บไว้ถูกคัดลอกจาก read ไป write

#include <stdio.h>

void remove_char(char *s, char c)
{
    char *write = s;
    for (char *read = s; *read != '\0'; read++) {
        if (*read != c) {
            *write = *read; // (1)!
            write++;
        }
    }
    *write = '\0'; // (2)!
}

int main()
{
    char phone[105];
    scanf("%s", phone);
    remove_char(phone, '-');
    printf("%s\n", phone);
    return 0;
}
  1. write ไม่มีทางแซง read เพราะขยับได้อย่างมากเท่ากับ read ตัวที่เขียนทับจึงเป็นตัวที่อ่านไปแล้วเสมอ ไม่ทำให้ข้อมูลที่ยังไม่ได้อ่านเสีย
  2. สตริงใหม่สั้นกว่าเดิม ต้องปิดท้ายตรงที่ write หยุด ไม่งั้น '\0' ยังอยู่ที่เดิมท้ายสตริงยาว ดูแบบที่ 2 ในส่วนลองทำให้พัง

ข้อความในเครื่องหมายคำพูด

ข้อความอย่าง "Monday" ในโค้ดก็เป็นอาเรย์ของ char ที่ compiler เตรียมไว้ให้ในหน่วยความจำส่วนที่อ่านได้อย่างเดียว เมื่อใช้เป็นค่าจะได้ที่อยู่ของตัวแรก เก็บที่อยู่นี้ไว้ใน const char * ได้ คำว่า const หน้าชนิดคือสัญญาว่าจะไม่แก้ตัวอักษรผ่าน pointer นี้ ถ้าเผลอแก้ compiler จะฟ้อง อาเรย์ของ const char * จึงเป็นตารางข้อความที่สะดวกมาก

#include <stdio.h>

int main()
{
    const char *days[7] = {"Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"}; // (1)!
    int today, after;
    scanf("%d %d", &today, &after);
    printf("%s\n", days[(today + after) % 7]);
    return 0;
}
  1. อาเรย์เจ็ดช่อง แต่ละช่องเก็บ ที่อยู่ ของข้อความหนึ่งข้อความ ไม่ได้เก็บตัวอักษรเอง ข้อความจึงยาวไม่เท่ากันได้ ต่างจากอาเรย์สองมิติของ char ในบทที่ 7 ที่ทุกแถวยาวเท่ากัน

word[0] คือตัว p ซึ่งมีรหัส 112 ไม่ใช่ที่อยู่ (compiler บอกว่าเป็น int เพราะ char ถูกแปลงเป็น int ตอนส่งให้ printf) แต่ %s จะเอา 112 ไปใช้เป็นที่อยู่แล้วตามไปอ่าน บนเครื่องที่เขียนเอกสารนี้โปรแกรมพังด้วย 0xC0000005 ทุกครั้ง อยากพิมพ์ตัวอักษรตัวเดียวใช้ %c อยากพิมพ์ข้อความตั้งแต่ตัวนั้นใช้ %s กับ word + 0 หรือ &word[0]

สรุป

  • ชื่ออาเรย์ char ที่ใช้เป็นค่าคือ char * ที่อยู่ของตัวแรก %s รับที่อยู่แล้วพิมพ์ไปจนเจอ '\0' ส่ง word + k ก็ได้ข้อความตั้งแต่ตัวที่ k
  • &word มีเลขที่อยู่เท่ากับ word แต่ชนิดเป็น char (*)[105] ใน scanf("%s") ใช้ชื่ออาเรย์เปล่าๆ
  • เดินสตริงด้วย for (char *p = s; *p != '\0'; p++) และความยาวคือ p - s ตอนเจอ '\0'
  • ฟังก์ชันคืน char * ชี้เข้าไปในสตริงได้ และคืน NULL เมื่อไม่เจอ คนเรียกต้องตรวจ NULL ก่อนใช้
  • เขียน '\0' ทับตรงไหน สตริงก็จบตรงนั้น ใช้ตัดสตริงเป็นท่อน และต้องปิดท้ายทุกครั้งที่เขียนสตริงใหม่ด้วย pointer
  • ข้อความในเครื่องหมายคำพูดแก้ไม่ได้ เก็บด้วย const char * และ == กับสตริงเทียบที่อยู่ ไม่ได้เทียบตัวอักษร
