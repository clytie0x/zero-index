16.2 vector สองมิติและความเร็ว

ตารางในบทที่ 7 ต้องจองขนาดสูงสุดไว้ก่อน เช่น int grid[1000][1000]; ถ้าโจทย์บอกแค่ว่าจำนวนช่องทั้งหมดไม่เกินล้าน ตารางอาจเป็น 1 × 1000000 หรือ 1000000 × 1 ก็ได้ อาเรย์สองมิติที่รับได้ทั้งสองแบบใหญ่เกินจะจองไหว หน้านี้สร้างตารางจาก vector ที่ขนาดมาจากข้อมูลเข้า ส่ง vector ให้ฟังก์ชันโดยไม่คัดลอกทิ้งเปล่าๆ แล้ววัดดูว่า push_back ช้ากว่าอาเรย์จริงไหม

ตารางที่ขนาดมาจากข้อมูลเข้า

vector สองมิติคือ vector ที่แต่ละช่องเป็น vector อีกตัว ชนิดจึงเขียนซ้อนกันเป็น vector<vector<int>> ตัวอย่างนี้รับคะแนนของนักเรียน R คน คนละ C วิชา แล้วพิมพ์คะแนนรวมของแต่ละคน

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int students, subjects;
    cin >> students >> subjects;
    vector<vector<int>> score(students, vector<int>(subjects)); // (1)!
    for (int i = 0; i < students; i++) {
        for (int j = 0; j < subjects; j++) {
            cin >> score[i][j]; // (2)!
        }
    }
    for (int i = 0; i < students; i++) {
        int total = 0;
        for (int x : score[i]) { // (3)!
            total += x;
        }
        cout << "student " << i + 1 << ": " << total << '\n';
    }
    return 0;
}
  1. สร้าง students แถว แต่ละแถวคือ vector<int>(subjects) ซึ่งมี subjects ช่องที่เป็น 0 อ่านแบบเดียวกับ vector<int> a(n, x) ในหน้า 16.1 คือ n ช่อง แต่ละช่องเป็น x เพียงแต่ x คือแถวทั้งแถว
  2. ใช้ [i][j] แถวก่อนคอลัมน์ เหมือนอาเรย์สองมิติในบทที่ 7
  3. score[i] คือแถวที่ i ซึ่งเป็น vector<int> หนึ่งตัว จึงวนด้วย range-for ได้ ส่วนจำนวนแถวคือ score.size() และจำนวนช่องในแถวที่ i คือ score[i].size()

แต่ละแถวยาวไม่เท่ากัน

แต่ละแถวเป็น vector ของตัวเอง จึงยาวไม่เท่ากันได้ และยืดทีละแถวด้วย push_back ได้ ซึ่งอาเรย์สองมิติทำไม่ได้ โรงเรียนมีชุมนุม K ชุมนุม นักเรียน N คน แต่ละคนเลือกหนึ่งชุมนุม ให้พิมพ์รายชื่อเลขที่ของสมาชิกแต่ละชุมนุม

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int k, n;
    cin >> k >> n;
    vector<vector<int>> members(k + 1);
    for (int student = 1; student <= n; student++) {
        int club;
        cin >> club;
        members[club].push_back(student);
    }
    for (int c = 1; c <= k; c++) {
        cout << "club " << c << " (" << members[c].size() << "):";
        for (int student : members[c]) {
            cout << ' ' << student;
        }
        cout << '\n';
    }
    return 0;
}

ส่ง vector ให้ฟังก์ชัน

ในหน้า 8.3 อาเรย์ที่ส่งให้ฟังก์ชันไม่รู้ขนาดของตัวเอง ต้องส่งจำนวนช่องไปด้วยเสมอ vector พกขนาดไปด้วย และฟังก์ชันคืน vector กลับมาได้ด้วย สิ่งที่ต้องเลือกคือจะส่งแบบไหน

หัวฟังก์ชันความหมายใช้เมื่อ
f(vector<int> v)คัดลอกทั้งก้อนมาใหม่ทุกครั้งที่เรียกต้องการสำเนาไปแก้เล่นโดยไม่กระทบตัวจริง ซึ่งน้อยครั้งมาก
f(const vector<int> &v)reference ไปยังตัวจริง และสัญญาว่าจะไม่แก้แค่อ่าน ใช้แบบนี้เป็นค่าเริ่มต้น
f(vector<int> &v)reference ไปยังตัวจริง แก้ได้ฟังก์ชันต้องเปลี่ยนค่าใน vector ของผู้เรียก
 
#include <bits/stdc++.h>
using namespace std;

long long total(const vector<int> &score)
{
    long long sum = 0;
    for (int s : score) {
        sum += s;
    }
    return sum;
}

void add_bonus(vector<int> &score, int bonus)
{
    for (int &s : score) {
        s = min(100, s + bonus);
    }
}

vector<int> passed(const vector<int> &score, int limit)
{
    vector<int> result;
    for (int s : score) {
        if (s >= limit) {
            result.push_back(s);
        }
    }
    return result;
}

int main()
{
    vector<int> score = {72, 45, 98, 49, 60};
    cout << "total = " << total(score) << '\n';
    add_bonus(score, 5);
    cout << "total = " << total(score) << '\n';
    vector<int> ok = passed(score, 60);
    cout << "passed = " << ok.size() << '\n';
    return 0;
}

add_bonus รับแบบ reference ที่แก้ได้ จึงบวกคะแนนเข้าไปใน score ของ main จริงๆ ส่วน min คืนค่าที่น้อยกว่าของสองค่า มีให้ใช้ใน C++ เลย ใช้กันคะแนนไม่ให้เกิน 100 คะแนน 98 จึงกลายเป็น 100 ไม่ใช่ 103 ผลรวมเลยเพิ่มขึ้น 22 ไม่ใช่ 25 ส่วน passed สร้าง vector ใหม่เก็บคะแนนที่ถึง 60 แล้วคืนกลับไปทั้งก้อน ภาษา C ทำแบบนี้กับอาเรย์ไม่ได้

push_back เร็วแค่ไหน

vector ต้องคอยยืดตัวเอง จึงน่าสงสัยว่าช้ากว่าอาเรย์ไหม วัดด้วย clock() แบบหน้า 4.5 เติมเลข 0 ถึง n - 1 ลงไปทีละตัว แล้วพิมพ์เวลาเป็นมิลลิวินาที (ms) ที่ต้องพิมพ์ช่องสุดท้ายด้วยก็เพื่อยืนยันว่าเติมครบจริง

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int n;
    cin >> n;
    clock_t start = clock();
    vector<int> v; // (1)!
    for (int i = 0; i < n; i++) {
        v.push_back(i); // (2)!
    }
    clock_t end = clock();
    cout << "n = " << n << ", last = " << v[n - 1] << ", time = " << (end - start) * 1000 / CLOCKS_PER_SEC << " ms\n"; // (3)!
    return 0;
}
  1. แบบ vector(n) เปลี่ยนเป็น vector<int> v(n); แบบ reserve เพิ่มบรรทัด v.reserve(n); ต่อจากบรรทัดนี้ ส่วนแบบอาเรย์ลบบรรทัดนี้ แล้วประกาศ int a[10000000]; ไว้นอก main
  2. แบบ vector(n) เปลี่ยนเป็น v[i] = i; แบบอาเรย์เปลี่ยนเป็น a[i] = i; และบรรทัดที่ 14 พิมพ์ a[n - 1] แทน
  3. clock() นับเป็นหน่วยย่อย CLOCKS_PER_SEC หน่วยต่อวินาที คูณ 1000 ก่อนหารจึงได้มิลลิวินาทีเป็นจำนวนเต็ม

ทำไม push_back ไม่ช้า

เบื้องหลัง vector จองที่ไว้เกินจำนวนที่ใช้เสมอ จำนวนที่จองไว้ดูได้จาก capacity() ถ้า push_back แล้วที่เต็ม vector จะจองที่ใหม่ที่ใหญ่ขึ้น ย้ายข้อมูลเดิมไปทั้งหมด แล้วคืนที่เก่า ลองพิมพ์ดูทุกครั้งที่ capacity() เปลี่ยน

#include <bits/stdc++.h>
using namespace std;

int main()
{
    vector<int> v;
    int last_capacity = 0;
    for (int i = 0; i < 100; i++) {
        v.push_back(i);
        if ((int)v.capacity() != last_capacity) {
            last_capacity = v.capacity();
            cout << "size " << v.size() << ", capacity " << v.capacity() << '\n';
        }
    }
    return 0;
}

g++ ขยายทีละเท่าตัว การย้ายข้อมูลจึงเกิดนานๆ ครั้ง และยิ่งนานขึ้นเรื่อยๆ เติม 100 ตัวย้ายแค่ 7 ครั้ง ถ้านับจำนวนค่าที่ถูกย้ายทั้งหมด จะได้ 1 + 2 + 4 + ... + 64 = 127 ค่า ไม่ถึงสองเท่าของจำนวนที่เติม ไม่ว่าจะเติมกี่ตัว ผลรวมแบบนี้จะน้อยกว่าสองเท่าของจำนวนที่เติมเสมอ โดยเฉลี่ย push_back หนึ่งครั้งจึงทำงานเพิ่มแค่ไม่กี่ขั้น compiler อื่นอาจขยายด้วยอัตราอื่น แต่ใช้หลักเดียวกัน

v.reserve(n) คือสั่งให้จองที่ n ช่องไว้ก่อนเลย size() ยังเป็น 0 เหมือนเดิม แต่ push_back n ครั้งต่อจากนั้นไม่ต้องย้ายข้อมูลอีก ต่างจาก vector<int> v(n) ที่สร้าง n ช่องจริงๆ ซึ่งถ้าตามด้วย push_back จะได้ 2n ช่องแบบในหน้า 16.1

แบบที่ 2 ส่ง vector ให้ฟังก์ชันโดยไม่ใส่ &

โปรแกรมนี้รวมค่าของช่องข้างเคียงทั้งสี่ทิศของทุกช่องในตาราง n × n แบบหน้า 7.4 โดยมีฟังก์ชัน cell คืนค่าในช่อง หรือคืน 0 ถ้าช่องนั้นอยู่นอกตาราง ทุกช่องในตารางเป็น 1

#include <bits/stdc++.h>
using namespace std;

int cell(vector<vector<int>> grid, int r, int c)
{
    if (r < 0 || r >= (int)grid.size() || c < 0 || c >= (int)grid[0].size()) {
        return 0;
    }
    return grid[r][c];
}

int main()
{
    int n;
    cin >> n;
    vector<vector<int>> grid(n, vector<int>(n, 1));
    clock_t start = clock();
    long long total = 0;
    for (int r = 0; r < n; r++) {
        for (int c = 0; c < n; c++) {
            total += cell(grid, r - 1, c) + cell(grid, r + 1, c) + cell(grid, r, c - 1) + cell(grid, r, c + 1);
        }
    }
    clock_t end = clock();
    cout << "total = " << total << ", time = " << (end - start) * 1000 / CLOCKS_PER_SEC << " ms\n";
    return 0;
}

สรุป

  • ตารางขนาด R × C สร้างด้วย vector<vector<int>> g(R, vector<int>(C)); ต้องบอกขนาดทั้งสองชั้น

  • ส่ง vector ให้ฟังก์ชันด้วย const vector<int> & ถ้าแค่อ่าน vector<int> & ถ้าต้องแก้ ลืม & แล้วอาจช้าเป็นพันเท่า
  • ฟังก์ชันคืน vector ทั้งก้อนได้
  • push_back สิบล้านครั้งใช้ไม่ถึงหนึ่งในสิบวินาที เพราะ vector ขยายทีละเท่าตัว reserve(n) จองที่ไว้ก่อนได้
