16.4 pair tuple และเรียงหลายเงื่อนไข

ข้อมูลในโจทย์มักมาเป็นคู่หรือเป็นชุด เช่น ชื่อกับคะแนน เวลาเริ่มกับเวลาจบ ตำแหน่งกับหมายเลข ในบทที่ 11 เราสร้าง struct ทุกครั้ง C++ มี pair สำหรับเก็บสองค่าคู่กัน และ tuple สำหรับสามค่าขึ้นไป ที่สำคัญคือทั้งสองเทียบกันเองได้ตามลำดับช่อง sort จึงเรียงได้หลายเงื่อนไขโดยไม่ต้องเขียนฟังก์ชันเปรียบเทียบเลย หน้านี้ปิดด้วยการเรียง struct หลายเงื่อนไขด้วย lambda เทียบกับ qsort ในหน้า 13.4 และ stable_sort สำหรับโจทย์ที่ค่าเท่ากันต้องอยู่ตามลำดับเดิม

pair เก็บสองค่าคู่กัน

#include <bits/stdc++.h>
using namespace std;

int main()
{
    pair<string, int> best = {"ploy", 92}; // (1)!
    cout << best.first << ' ' << best.second << '\n'; // (2)!
    best.second += 3;
    pair<int, int> start = make_pair(7, 30); // (3)!
    cout << best.second << ' ' << start.first << ':' << start.second << '\n';
    return 0;
}
  1. ชนิดของสองช่องเขียนในวงเล็บแหลม คั่นด้วยจุลภาค สองช่องเป็นคนละชนิดได้ กำหนดค่าด้วยปีกกาเรียงตามช่อง
  2. ช่องแรกชื่อ first ช่องที่สองชื่อ second ใช้เหมือนตัวแปรธรรมดา อ่าน แก้ บวกได้หมด
  3. make_pair(7, 30) สร้าง pair จากสองค่าโดยเดาชนิดให้ ได้ผลเหมือน {7, 30} โค้ดเก่าๆ ใช้แบบนี้บ่อย

pair ก็คือ struct ที่มีสองช่องชื่อ first กับ second ที่ C++ เตรียมไว้ให้ ไม่ต้องประกาศเอง ข้อเสียคือชื่อช่องไม่บอกความหมาย อ่านโค้ดยาวๆ แล้วต้องจำว่า first คืออะไร

pair เทียบกันอย่างไร

pair เทียบด้วย < > == ได้ โดยเทียบ first ก่อน ถ้า first เท่ากันจึงเทียบ second เหมือนเทียบคำในพจนานุกรมที่ดูตัวอักษรแรกก่อน

#include <bits/stdc++.h>
using namespace std;

int main()
{
    pair<int, int> a = {3, 9};
    pair<int, int> b = {5, 1};
    pair<int, int> c = {3, 4};
    cout << (a < b) << '\n';
    cout << (c < a) << '\n';
    cout << (a == c) << '\n';
    pair<int, int> smallest = min({a, b, c});
    cout << smallest.first << ' ' << smallest.second << '\n';
    return 0;
}

  • a < b จริง เพราะ first 3 น้อยกว่า 5 ไม่ต้องดู second เลย แม้ 9 จะมากกว่า 1
  • c < a จริง เพราะ first เท่ากัน จึงตัดสินด้วย second 4 น้อยกว่า 9
  • min ที่ใส่หลายค่าในปีกกาหาค่าน้อยสุดของทั้งหมด และใช้กับ pair ได้เพราะ pair เทียบกันได้

เรียง vector ของ pair

กติกาการเทียบนี้คือ "เรียงตาม first ถ้าเท่ากันเรียงตาม second" พอดี sort vector ของ pair จึงได้การเรียงสองเงื่อนไขทันที เคล็ดลับคือเอาค่าที่ใช้เรียงเป็นอันดับแรกไว้ใน first งานวิ่งการกุศลบันทึกชื่อกับเวลาเป็นนาทีของนักวิ่ง ให้พิมพ์จากคนที่ใช้เวลาน้อยที่สุด ถ้าเวลาเท่ากันเรียงตามชื่อ

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int n;
    cin >> n;
    vector<pair<int, string>> runner(n);
    for (auto &r : runner) {
        cin >> r.second >> r.first;
    }
    sort(runner.begin(), runner.end());
    for (const auto &r : runner) {
        cout << r.first << ' ' << r.second << '\n';
    }
    return 0;
}

เงื่อนไขหนึ่งมากไปน้อย อีกเงื่อนไขน้อยไปมาก

ถ้าโจทย์ให้เรียงคะแนนจากมากไปน้อย แต่คะแนนเท่ากันให้เรียงชื่อจากน้อยไปมาก จะใช้ greater ไม่ได้ เพราะชื่อจะกลับด้านไปด้วย เคล็ดลับคือเก็บคะแนนติดลบไว้ใน first คะแนนมากกลายเป็นค่าน้อย จึงขึ้นก่อนเมื่อเรียงจากน้อยไปมาก ตอนพิมพ์ค่อยกลับเครื่องหมาย

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int n;
    cin >> n;
    vector<pair<int, string>> list(n);
    for (auto &p : list) {
        int score;
        cin >> p.second >> score;
        p.first = -score;
    }
    sort(list.begin(), list.end());
    for (const auto &p : list) {
        cout << p.second << ' ' << -p.first << '\n';
    }
    return 0;
}

tuple เมื่อมีสามค่าขึ้นไป

tuple คือ pair ที่มีกี่ช่องก็ได้ เทียบกันแบบเดียวกัน คือดูช่องแรกก่อน เท่ากันค่อยดูช่องถัดไป ช่องของ tuple ไม่มีชื่อ หยิบด้วย get<0>(t) get<1>(t) ตามเลขช่อง หรือแตกออกเป็นตัวแปรหลายตัวในบรรทัดเดียวด้วย auto [a, b, c] = t; ซึ่งเรียกว่า structured binding ใช้กับ pair ได้ด้วย

ตัวอย่างนี้คือการแข่งตอบคำถามในหน้า 13.4 คะแนนมากก่อน คะแนนเท่ากันเวลาน้อยก่อน เท่ากันอีกเรียงตามชื่อ สามเงื่อนไขพอดีกับสามช่องของ tuple

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int n;
    cin >> n;
    vector<tuple<int, int, string>> list;
    for (int i = 0; i < n; i++) {
        string name;
        int score, minutes;
        cin >> name >> score >> minutes;
        list.push_back({-score, minutes, name}); // (1)!
    }
    sort(list.begin(), list.end());
    cout << "top: " << get<2>(list[0]) << '\n'; // (2)!
    int rank = 1;
    for (auto [neg_score, minutes, name] : list) { // (3)!
        cout << rank << ". " << name << ' ' << -neg_score << ' ' << minutes << '\n';
        rank++;
    }
    return 0;
}
  1. ช่องแรกคะแนนติดลบ ให้คะแนนมากขึ้นก่อน ช่องที่สองเวลา ช่องที่สามชื่อ ลำดับช่องคือลำดับความสำคัญของเงื่อนไข
  2. get<2> หยิบช่องที่ 2 นับจาก 0 คือชื่อ เลขในวงเล็บแหลมต้องเป็นค่าคงที่ จะใช้ตัวแปรแทนไม่ได้
  3. แตก tuple แต่ละตัวเป็นตัวแปรสามตัวตามลำดับช่อง ตั้งชื่ออะไรก็ได้ อ่านง่ายกว่า get มาก

เรียง struct ด้วย lambda

pair และ tuple สะดวกเมื่อข้อมูลมีสองสามช่อง แต่ถ้ามีหลายช่อง หรือช่องที่ใช้เรียงไม่ได้เรียงตามลำดับในข้อมูล struct ที่มีชื่อช่องชัดเจนอ่านง่ายกว่า ใน C++ ประกาศ struct แล้วใช้ชื่อเป็นชนิดได้เลย ไม่ต้องเขียนคำว่า struct นำหน้าทุกครั้งแบบภาษา C

#include <bits/stdc++.h>
using namespace std;

struct Contestant {
    string name;
    int score;
    int minutes;
};

int main()
{
    int n;
    cin >> n;
    vector<Contestant> list(n);
    for (Contestant &c : list) {
        cin >> c.name >> c.score >> c.minutes;
    }
    sort(list.begin(), list.end(), [](const Contestant &a, const Contestant &b) {
        if (a.score != b.score) {
            return a.score > b.score;
        }
        if (a.minutes != b.minutes) {
            return a.minutes < b.minutes;
        }
        return a.name < b.name;
    });
    for (int i = 0; i < n; i++) {
        cout << i + 1 << ". " << list[i].name << ' ' << list[i].score << ' ' << list[i].minutes << '\n';
    }
    return 0;
}

stable_sort รักษาลำดับของค่าที่เท่ากัน

หน้า 13.4 เห็นแล้วว่า qsort ไม่สัญญาว่าค่าที่เท่ากันจะอยู่ตามลำดับเดิม sort ก็ไม่สัญญาเช่นกัน ตัวอย่างนี้ให้นักเรียนเลขที่ 1 ถึง 20 ได้คะแนน 5 กับ 7 สลับกัน เรียงตามคะแนนจากมากไปน้อยอย่างเดียว แล้วพิมพ์เป็น เลขที่:คะแนน

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int n;
    cin >> n;
    vector<pair<int, int>> list(n);
    for (int i = 0; i < n; i++) {
        cin >> list[i].first;
        list[i].second = i + 1;
    }
    sort(list.begin(), list.end(), [](const pair<int, int> &a, const pair<int, int> &b) {
        return a.first > b.first;
    });
    for (int i = 0; i < n; i++) {
        cout << list[i].second << ':' << list[i].first << (i + 1 < n ? ' ' : '\n');
    }
    return 0;
}

เปลี่ยน sort ในบรรทัดที่ 13 เป็น stable_sort ซึ่งรับค่าแบบเดียวกันทุกอย่าง แต่สัญญาว่าค่าที่ฟังก์ชันเปรียบเทียบบอกว่าเท่ากัน จะอยู่ตามลำดับเดิมเสมอ

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int n;
    cin >> n;
    vector<pair<int, int>> list(n);
    for (int i = 0; i < n; i++) {
        cin >> list[i].first;
        list[i].second = i + 1;
    }
    stable_sort(list.begin(), list.end(), [](const pair<int, int> &a, const pair<int, int> &b) {
        return a.first > b.first;
    });
    for (int i = 0; i < n; i++) {
        cout << list[i].second << ':' << list[i].first << (i + 1 < n ? ' ' : '\n');
    }
    return 0;
}

  • pair และ tuple เทียบช่องแรกก่อน เท่ากันค่อยดูช่องถัดไป sort vector ของ pair จึงเรียงหลายเงื่อนไขได้โดยไม่ต้องเขียน lambda
  • เงื่อนไขที่ต้องการมากไปน้อย เก็บค่าติดลบไว้ในช่องนั้น
  • tuple มีกี่ช่องก็ได้ หยิบด้วย get<i>(t) หรือ auto [a, b, c] = t;
  • struct เรียงด้วย lambda ที่รับ const T & ไล่เงื่อนไขทีละข้อ เงื่อนไขแรกที่ต่างกันเป็นตัวตัดสิน
  • sort ไม่รักษาลำดับของค่าที่เท่ากัน ใช้ stable_sort หรือใส่ลำดับที่รับเข้ามาเป็นเงื่อนไขสุดท้าย และอย่าเชื่อผลที่ทดลองกับข้อมูลไม่ถึง 17 ตัว
