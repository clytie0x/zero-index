ในบทที่ 6 ข้อความคืออาเรย์ของ char ต้องจองขนาดเผื่อไว้ก่อน ต้องเหลือช่องให้ '\0' คัดลอกด้วย strcpy ต่อด้วย strcat เทียบด้วย strcmp และระวังเขียนเกินอาเรย์ตลอดเวลา C++ มีชนิด string ที่จัดการเรื่องพวกนี้ให้ทั้งหมด ยืดหดได้เอง ใช้ + ต่อ ใช้ == เทียบได้ตรงๆ และมีคำสั่งตัดและค้นหาข้อความในตัว โจทย์ข้อความที่เคยยาวครึ่งหน้าจะเหลือไม่กี่บรรทัด

เทียบข้อความด้วย == และ <

โปรแกรมเดียวกับตัวอย่าง strcmp ในหน้า 6.4 เขียนด้วย string

#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    string first, second;
    while (cin >> first >> second) {
        if (first == second) {
            cout << first << " is the same\n";
        } else if (first < second) {
            cout << first << " comes before " << second << '\n';
        } else {
            cout << first << " comes after " << second << '\n';
        }
    }
    return 0;
}

ความยาวและตัวอักษรทีละตัว

เข้าถึงตัวที่ i ด้วย name[i] เหมือนอาเรย์ เริ่มที่ช่อง 0 ตัวสุดท้ายอยู่ที่ช่อง size() - 1 และแก้ทีละตัวได้

#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    string name;
    cin >> name;
    int n = name.size(); // (1)!
    name[0] = toupper(name[0]);
    int vowels = 0;
    for (int i = 0; i < n; i++) {
        if (name[i] == 'a' || name[i] == 'e' || name[i] == 'i' || name[i] == 'o' || name[i] == 'u') {
            vowels++;
        }
    }
    cout << name << '\n';
    cout << "first " << name[0] << " last " << name[n - 1] << '\n';
    cout << vowels << " vowels\n";
    string line(n, '-'); // (2)!
    cout << line << '\n';
    return 0;
}
  1. เก็บความยาวลง int ไว้ก่อน แล้วใช้ n ทั้งในลูปและตอนหาตัวสุดท้าย เหตุผลอยู่ในส่วนลองทำให้พังแบบที่ 1 และ 4
  2. สร้าง string ที่มีตัวอักษร '-' ซ้ำกัน n ตัว ใช้ทำเส้นคั่นหรือแถวของดาวได้

substr ตัดข้อความบางส่วน

s.substr(start, count) คืนข้อความใหม่ที่เริ่มจากช่อง start ยาว count ตัว ถ้าไม่ใส่ count จะเอาไปจนสุดข้อความ ตัวอย่างคือวันที่ในรูปแบบ ปี-เดือน-วัน ซึ่งทุกส่วนอยู่ตำแหน่งตายตัว

}

find ค้นหาข้อความ

s.find(x) หาว่า x ซึ่งจะเป็น char หรือข้อความก็ได้ ปรากฏครั้งแรกที่ช่องไหนของ s ถ้าไม่เจอ จะคืนค่าพิเศษชื่อ string::npos (no position) ต้องเช็กค่านี้ก่อนเอาตำแหน่งไปใช้ทุกครั้ง

#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    string email;
    while (cin >> email) {
        size_t at = email.find('@'); // (1)!
        if (at == string::npos) {
            cout << email << " is not an email\n";
        } else {
            cout << "user " << email.substr(0, at) << ", domain " << email.substr(at + 1) << '\n';
        }
    }
    return 0;
}
  1. ตำแหน่งที่ find คืนมามีชนิด size_t ซึ่งเป็นจำนวนเต็มไม่ติดลบแบบเดียวกับที่ strlen คืนในหน้า 6.4 เก็บด้วยชนิดนี้แล้วเทียบกับ string::npos ตรงๆ ปลอดภัยที่สุด

ใส่ตำแหน่งเริ่มค้นเป็นตัวที่สองได้ s.find(x, start) จะเริ่มหาตั้งแต่ช่อง start เป็นต้นไป ใช้หาทุกตำแหน่งที่คำหนึ่งปรากฏ โดยเริ่มรอบถัดไปหลังตำแหน่งที่เพิ่งเจอ

#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    string text, word;
    cin >> text >> word;
    int found = 0;
    size_t pos = text.find(word);
    while (pos != string::npos) {
        cout << "found at " << pos << '\n';
        found++;
        pos = text.find(word, pos + 1);
    }
    cout << found << " times\n";
    return 0;
}

getline รับทั้งบรรทัด

cin >> หยุดที่ช่องว่างเหมือน %s ถ้าต้องการทั้งบรรทัดรวมช่องว่าง ใช้ getline(cin, line) ซึ่งต่างจาก fgets ในหน้า 6.3 ตรงที่ไม่เก็บ '\n' ไว้ในข้อความ จึงไม่ต้องตัดเอง และไม่ต้องกำหนดขนาด

#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    string line;
    int count = 0;
    while (getline(cin, line)) {
        count++;
        cout << count << " [" << line << "] " << line.size() << '\n';
    }
    return 0;
}

ไฟล์จาก Windows บน grader ที่เป็น Linux

หน้า 6.3 เล่าไว้ว่าไฟล์ที่สร้างบน Windows จบบรรทัดด้วย '\r' กับ '\n' และ grader บน Linux ไม่แปลงให้ getline ตัด '\n' ทิ้งแต่ไม่ตัด '\r' ถ้าโจทย์ไหนสงสัยว่าจะเจอ ให้ตัดเองด้วย if (line.size() > 0 && line[line.size() - 1] == '\r') line.pop_back(); โดย pop_back() คือลบตัวสุดท้ายทิ้งหนึ่งตัว

#include <bits/stdc++.h>

วิธีแก้คืออ่านทิ้งให้ถึง '\n' ก่อนเรียก getline แบบเดียวกับลูป getchar ในหน้า 6.3 ใน C++ ใช้ cin.ignore

#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int room;
    string name;
    cin >> room;
    cin.ignore(1000000, '\n'); // (1)!
    getline(cin, name);
    cout << "room " << room << '\n';
    cout << "name [" << name << "]\n";
    return 0;
}
  1. อ่านทิ้งไปเรื่อยๆ จนกว่าจะทิ้ง '\n' ไปแล้วหนึ่งตัว หรือจนทิ้งครบหนึ่งล้านตัว เลขหนึ่งล้านแค่เป็นเพดานที่ใหญ่พอ บรรทัดแรกของโจทย์ไม่มีทางยาวขนาดนั้น

  what():  basic_string::substr: __pos (which is 18446744073709551615) > this->size() (which is 2)

Process returned -1073740791 (0xC0000409)

มีสองเรื่องในข้อความนี้ อย่างแรก word.size() เป็นจำนวนไม่ติดลบ 2 ลบ 3 จึงไม่ได้ -1 แต่วนไปเป็นเลขที่ใหญ่ที่สุดแบบเดียวกับ npos อย่างที่สอง substr ตรวจว่าตำแหน่งเริ่มต้นเกินความยาวไหม ถ้าเกินจะหยุดโปรแกรมด้วยข้อความ out_of_range ซึ่ง grader จะนับเป็น runtime error แก้ด้วยการเก็บความยาวลง int ก่อนคำนวณ แล้วเช็กว่ายาวพอไหมก่อนตัด ส่วน word[i] ไม่ตรวจอะไรให้เลยเหมือนอาเรย์ เกินขอบแล้วได้ค่ามั่วเงียบๆ แบบหน้า 5.5

สรุป

  • string ไม่ต้องจองขนาด ต่อด้วย + += เทียบด้วย == < ได้ตรงๆ
  • s.size() คือความยาว เก็บลง int ก่อนใช้ในลูปหรือก่อนลบ
  • s.substr(start, count) ตัดข้อความ ตำแหน่งเริ่มเกินความยาวแล้วโปรแกรมพัง
  • s.find(x) คืนตำแหน่งแรกที่เจอ หรือ string::npos ถ้าไม่เจอ เทียบกับ npos เสมอ ห้ามเทียบกับ 0 หรือ -1
  • getline(cin, line) รับทั้งบรรทัด ไม่เก็บ '\n'
  • หลัง cin >> จะใช้ getline ต้องมี cin.ignore(1000000, '\n'); ก่อน
