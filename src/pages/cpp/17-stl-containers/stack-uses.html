17.3 งานที่ใช้ stack

หน้า 14.4 สรุปไว้ว่า stack ใช้เมื่อต้องจับคู่กับ "ตัวล่าสุดที่ยังค้างอยู่" หรือเมื่อต้องย้อนลำดับ หน้านี้เขียนงานเดิมสองงานใหม่ด้วย stack ของ STL คือตรวจวงเล็บกับคำนวณ postfix เพิ่มงานย้อนกลับแบบปุ่ม back ของเว็บเบราว์เซอร์ และปิดท้ายด้วยเทคนิคที่ใช้บ่อยมากในโจทย์แข่งขัน คือ monotonic stack ที่เปลี่ยนวิธีลูปซ้อนซึ่งไม่ทันเวลา ให้เหลือการเดินรอบเดียว

ตรวจวงเล็บ

กฎเดิมจากหน้า 14.4 เจอวงเล็บเปิด push เจอวงเล็บปิดต้องคู่กับตัวบนสุด และจบแล้ว stack ต้องว่าง

#include <bits/stdc++.h>
using namespace std;

char opening_of(char c)
{
    if (c == ')') {
        return '(';
    } else if (c == ']') {
        return '[';
    }
    return '{';
}

bool is_balanced(string s)
{
    stack<char> open; // (1)!
    for (char c : s) {
        if (c == '(' || c == '[' || c == '{') {
            open.push(c);
        } else {
            if (open.empty() || open.top() != opening_of(c)) { // (2)!
                return false;
            }
            open.pop();
        }
    }
    return open.empty();
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    string s;
    while (cin >> s) {
        if (is_balanced(s)) {
            cout << "YES\n";
        } else {
            cout << "NO\n";
        }
    }
    return 0;
}

ย้อนกลับด้วยปุ่ม back

เว็บเบราว์เซอร์จำหน้าที่เคยเปิด กด back แล้วกลับไปหน้าล่าสุดก่อนหน้านี้ กด back อีกครั้งก็ถอยไปอีกหน้า หน้าที่ต้องกลับไปคือหน้าที่เข้ามาล่าสุด จึงเป็นงานของ stack ทุกครั้งที่เปิดหน้าใหม่ ให้ push หน้าปัจจุบันเก็บไว้ก่อนย้าย พอกด back ก็ pop ออกมาเป็นหน้าปัจจุบัน

โปรแกรมเริ่มที่หน้า home คำสั่ง VISIT x คือเปิดหน้า x และ BACK คือกดปุ่ม back หลังทุกคำสั่งให้พิมพ์ว่าอยู่หน้าไหน

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int q;
    cin >> q;
    stack<string> history;
    string current = "home";
    for (int i = 0; i < q; i++) {
        string command;
        cin >> command;
        if (command == "VISIT") {
            string page;
            cin >> page;
            history.push(current); // (1)!
            current = page;
            cout << "at " << current << '\n';
        } else if (history.empty()) {
            cout << "no page to go back\n";
        } else {
            current = history.top(); // (2)!
            history.pop();
            cout << "at " << current << '\n';
        }
    }
    return 0;
}

คำนวณ postfix

กฎเดิมจากหน้า 14.4 เจอตัวเลข push เจอเครื่องหมาย pop สองครั้ง ตัวแรกที่ออกมาคือตัวขวา คำนวณแล้ว push ผลกลับ

#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    stack<long long> values;
    string token;
    while (cin >> token) {
        if (isdigit(token[0])) {
            long long number = 0;
            for (char c : token) {
                number = number * 10 + (c - '0');
            }
            values.push(number);
        } else {
            long long right = values.top(); // (1)!
            values.pop();
            long long left = values.top();
            values.pop();
            if (token == "+") { // (2)!
                values.push(left + right);
            } else if (token == "-") {
                values.push(left - right);
            } else if (token == "*") {
                values.push(left * right);
            } else {
                values.push(left / right);
            }
        }
    }
    cout << values.top() << '\n';
    return 0;
}
  1. เพราะ pop() ไม่คืนค่า หนึ่งครั้งของ pop() ในหน้า 14.4 จึงกลายเป็นสองบรรทัด ดูก่อนแล้วค่อยทิ้ง
  2. string เทียบกับข้อความด้วย == ได้ตรงๆ ไม่ต้องดูแค่ตัวแรกแบบภาษา C

วันแรกที่ร้อนกว่า

สถานีอากาศบันทึกอุณหภูมิสูงสุดของ N วันติดกัน สำหรับแต่ละวัน อยากรู้ว่าวันแรกหลังจากนั้นที่ร้อนกว่าวันนี้อย่างเคร่งครัดร้อนกี่องศา ถ้าไม่มีวันไหนร้อนกว่าเลยให้ตอบ -1 โจทย์แบบนี้มีชื่อเรียกว่า next greater element (ตัวถัดไปที่มากกว่า) และมีอีกหลายแบบ เช่น ตัวก่อนหน้าที่มากกว่า หรือตัวถัดไปที่น้อยกว่า

วิธีตรงไปตรงมาคือ ทุกวันเดินไปทางขวาจนเจอวันที่ร้อนกว่า

#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<int> temp(n), answer(n, -1);
    for (int i = 0; i < n; i++) {
        cin >> temp[i];
    }
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (temp[j] > temp[i]) {
                answer[i] = temp[j];
                break;
            }
        }
    }
    for (int i = 0; i < n; i++) {
        cout << answer[i] << (i + 1 < n ? ' ' : '\n');
    }
    return 0;
}

กลับมุมมอง ให้วันที่ยังไม่มีคำตอบรอ

แทนที่แต่ละวันจะเดินไปหาคำตอบเอง ให้เดินจากซ้ายไปขวารอบเดียว แล้วเก็บวันที่ยังรอคำตอบอยู่ไว้ใน stack พอเจอวันใหม่ วันที่รออยู่ซึ่งเย็นกว่าวันใหม่ ได้คำตอบเป็นวันใหม่นี้ทั้งหมด

สิ่งที่ทำให้วิธีนี้เร็วคือข้อสังเกตนี้ อุณหภูมิของวันที่รออยู่ ไล่จากก้น stack ขึ้นมา ไม่เคยเพิ่มขึ้น เพราะถ้ามีวันที่ร้อนกว่ามาทีหลัง วันที่เย็นกว่าที่รออยู่ก่อนก็ได้คำตอบแล้วถูก pop ออกไปตั้งแต่ตอนนั้น ตัวบนสุดจึงเป็นวันที่เย็นที่สุดในบรรดาวันที่รอเสมอ ถ้าตัวบนสุดไม่เย็นกว่าวันใหม่ ตัวที่อยู่ข้างล่างก็ไม่เย็นกว่าแน่นอน หยุดดูได้ทันที stack ที่ของข้างในเรียงทางเดียวตลอดแบบนี้เรียกว่า monotonic stack (monotonic แปลว่าเปลี่ยนไปทางเดียว)

#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<int> temp(n), answer(n, -1);
    for (int i = 0; i < n; i++) {
        cin >> temp[i];
    }
    stack<int> waiting; // (1)!
    for (int i = 0; i < n; i++) {
        while (!waiting.empty() && temp[waiting.top()] < temp[i]) { // (2)!
            answer[waiting.top()] = temp[i];
            waiting.pop();
        }
        waiting.push(i);
    }
    for (int i = 0; i < n; i++) {
        cout << answer[i] << (i + 1 < n ? ' ' : '\n');
    }
    return 0;
}
  1. เก็บเลขวัน ไม่ใช่อุณหภูมิ เพราะตอนได้คำตอบต้องรู้ว่าจะเขียนลง answer ช่องไหน อุณหภูมิหาได้จาก temp[เลขวัน] อยู่แล้ว
  2. pop ไปเรื่อยๆ ตราบใดที่ตัวบนสุดเย็นกว่าวันใหม่ วันที่เหลือใน stack ตอนจบไม่มีวันที่ร้อนกว่า คำตอบจึงเป็น -1 ตามค่าเริ่มต้น

ลูป while ข้างในไม่ทำให้ช้า

ดูเผินๆ โปรแกรมมีลูปซ้อนเหมือนวิธีแรก แต่ให้นับจากอีกมุมหนึ่ง ทุกวันถูก push ครั้งเดียว และถูก pop ไม่เกินครั้งเดียว เพราะ pop แล้วไม่เคยกลับเข้าไปอีก ทั้งโปรแกรมจึงมี push n ครั้ง และ pop ไม่เกิน n ครั้ง รวมไม่เกิน 2n ครั้ง ถึงบางรอบของ for จะ pop หลายตัว อย่างวันที่ 6 ที่ pop สามตัว รอบอื่นก็ pop น้อยลงชดเชยกัน งานรวมจึงโตตาม n ตรงๆ เรียกว่า O(n) ส่วนวิธีแรกเป็น O(n2)

วัดด้วยโปรแกรมที่สร้างข้อมูลเอง มีสองแบบ แบบ random ใช้ rand() แบบหน้า 13.3 และแบบ falling ที่อุณหภูมิลดลงทุกวัน ซึ่งเป็นกรณีแย่ที่สุดของวิธีแรก ทั้งสองวิธีอยู่ในโปรแกรมเดียวกัน และตรวจว่าได้คำตอบตรงกัน

#include <bits/stdc++.h>
using namespace std;

int temp[200000];
int answer_slow[200000];
int answer_fast[200000];

int main()
{
    int n;
    string pattern;
    cin >> n >> pattern;
    srand(2569);
    for (int i = 0; i < n; i++) {
        if (pattern == "falling") {
            temp[i] = n - i;
        } else {
            temp[i] = rand() % 1000;
        }
    }

    clock_t start = clock();
    for (int i = 0; i < n; i++) {
        answer_slow[i] = -1;
        for (int j = i + 1; j < n; j++) {
            if (temp[j] > temp[i]) {
                answer_slow[i] = temp[j];
                break;
            }
        }
    }
    clock_t middle = clock();
    stack<int> waiting;
    for (int i = 0; i < n; i++) {
        answer_fast[i] = -1;
        while (!waiting.empty() && temp[waiting.top()] < temp[i]) {
            answer_fast[waiting.top()] = temp[i];
            waiting.pop();
        }
        waiting.push(i);
    }
    clock_t end = clock();

    int same = 1;
    for (int i = 0; i < n; i++) {
        if (answer_slow[i] != answer_fast[i]) {
            same = 0;
        }
    }
    cout << "same answers: " << same << '\n';
    cout << fixed << setprecision(3);
    cout << "nested loops: " << (double)(middle - start) / CLOCKS_PER_SEC << " seconds\n";
    cout << "stack: " << (double)(end - middle) / CLOCKS_PER_SEC << " seconds\n";
    return 0;
}

สรุป

  • stack ใช้กับงานที่ต้องจับคู่กับตัวล่าสุดที่ค้างอยู่ เช่น วงเล็บ และงานย้อนกลับ เช่น ปุ่ม back กับ undo
  • ประกาศ stack ในฟังก์ชันหรือในลูป จะได้ stack ว่างใหม่ทุกครั้งโดยไม่ต้องเริ่มเอง
  • monotonic stack หาตัวถัดไปที่มากกว่าของทุกตัวได้ในรอบเดียว เก็บ index ไม่ใช่ค่า
  • ทุกตัว push ครั้งเดียว pop ไม่เกินครั้งเดียว งานรวม O(n) ส่วนลูปซ้อนเป็น O(n2) และช้าจริงเฉพาะข้อมูลแบบที่แย่ที่สุด
  • ระวัง < กับ <= เมื่อมีค่าซ้ำ และเช็ก empty() ไว้ซ้ายสุดของเงื่อนไข
