17.1 stack และ queue สำเร็จรูป

ในบทที่ 14 น้องเขียน stack กับ queue เองด้วยอาเรย์ ต้องเลือกขนาดอาเรย์ ต้องมีตัวแปร size หรือ head กับ tail และต้องเขียนฟังก์ชัน push pop ทุกครั้งที่เริ่มโจทย์ใหม่ C++ มีของพวกนี้ทำไว้ให้แล้วใน STL ชุดเดียวกับ vector ที่ใช้ในบทที่ 16 ของที่ใช้เก็บข้อมูลหลายตัวแบบนี้เรียกรวมว่า container (ภาชนะ) หน้านี้ใช้ container สองตัวแรก คือ stack กับ queue แล้วดูว่าโค้ดสั้นลงแค่ไหน และมีอะไรที่ยังต้องระวังเองเหมือนเดิม

stack

ประกาศ stack ด้วย stack<ชนิด> ชื่อ; เหมือนประกาศ vector ได้ stack ว่างที่ยังไม่มีของ โปรแกรมนี้ใช้ลำดับคำสั่งเดียวกับโปรแกรม plates ในหน้า 14.4 ทุกบรรทัด

#include <bits/stdc++.h>
using namespace std;

int main()
{
    stack<int> plates; // (1)!
    plates.push(11);
    plates.push(12);
    plates.push(13);
    cout << "top = " << plates.top() << '\n';
    cout << "pop = " << plates.top() << '\n'; // (2)!
    plates.pop();
    cout << "pop = " << plates.top() << '\n';
    plates.pop();
    plates.push(14);
    cout << "size = " << plates.size() << '\n';
    while (!plates.empty()) { // (3)!
        cout << "pop = " << plates.top() << '\n';
        plates.pop();
    }
    return 0;
}
  1. ไม่ต้องบอกขนาด stack ยืดออกเองตามจำนวนของที่ใส่ ของในวงเล็บ <> คือชนิดของที่เก็บ
  2. pop() ของ C++ ไม่คืนค่า มันแค่ทิ้งตัวบนสุด ถ้าอยากรู้ว่าทิ้งอะไรไป ต้องดูด้วย top() ก่อนแล้วค่อย pop() เสมอ
  3. empty() คืน true เมื่อไม่มีของเหลือ ใช้แทน size == 0 ในบทที่ 14

สิ่งที่ stack ไม่มีให้ คือการดูตัวที่อยู่ข้างล่าง ไม่มี s[0] และวนดูทุกตัวด้วย for ไม่ได้ เพราะ stack ตั้งใจให้แตะได้แค่ด้านบน ถ้างานไหนต้องดูของข้างล่างด้วย ให้ใช้ vector แทน ซึ่งใช้เป็น stack ได้ด้วย push_back pop_back และ back หรือใช้ deque ในหน้าถัดไป

queue

queue ใช้คำว่า push กับ pop เหมือน stack แต่ push ต่อท้ายคิว และ pop เอาหัวคิวออก ดูหัวคิวด้วย front() และดูท้ายคิวด้วย back() โปรแกรมนี้ใช้ลำดับคำสั่งเดียวกับโปรแกรม queue-basics ในหน้า 14.5 และเพิ่มการดูท้ายคิวอีกหนึ่งบรรทัด

#include <bits/stdc++.h>
using namespace std;

int main()
{
    queue<int> line;
    line.push(11); // (1)!
    line.push(12);
    line.push(13);
    cout << "front = " << line.front() << '\n';
    cout << "back = " << line.back() << '\n';
    cout << "dequeue = " << line.front() << '\n';
    line.pop(); // (2)!
    cout << "dequeue = " << line.front() << '\n';
    line.pop();
    line.push(14);
    cout << "size = " << line.size() << '\n';
    while (!line.empty()) {
        cout << "dequeue = " << line.front() << '\n';
        line.pop();
    }
    return 0;
}
  1. push ของ queue คือ enqueue ในบทที่ 14 ต่อท้ายคิว
  2. pop ของ queue คือ dequeue เอาหัวคิวออก และไม่คืนค่าเหมือน stack ต้องดู front() ก่อน

ระวังชื่อที่สับสนง่าย stack ใช้ top() ส่วน queue ใช้ front() เขียน line.top() กับ queue จะคอมไพล์ไม่ผ่าน และ queue ก็ไม่มี line[0] หรือวนดูทุกตัวเหมือน stack

เก็บอะไรก็ได้

stack ในบทที่ 14 เก็บ int ได้อย่างเดียว ถ้าอยากได้ stack ของ char ต้องเขียนใหม่ทั้งชุดแบบโปรแกรม brackets ของ C++ แค่เปลี่ยนชนิดในวงเล็บ <> จะเป็น string หรือ pair จากบทที่ 16 ก็ได้

ตัวอย่าง ร้านถ่ายเอกสารมีเครื่องเดียว ลูกค้าส่งงานเรียงกันมา แต่ละงานมีชื่อลูกค้ากับจำนวนหน้า เครื่องถ่ายได้นาทีละหนึ่งหน้า ทำงานตามลำดับที่ส่งมา ให้บอกว่างานของแต่ละคนเสร็จนาทีที่เท่าไร

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int n;
    cin >> n;
    queue<pair<string, int>> jobs; // (1)!
    for (int i = 0; i < n; i++) {
        string name;
        int pages;
        cin >> name >> pages;
        jobs.push({name, pages}); // (2)!
    }
    int minute = 0;
    while (!jobs.empty()) {
        auto job = jobs.front(); // (3)!
        jobs.pop();
        minute += job.second;
        cout << job.first << " done at minute " << minute << '\n';
    }
    return 0;
}

  2. สร้าง pair ด้วยวงเล็บปีกกา เขียน make_pair(name, pages) ก็ได้ผลเหมือนกัน
  3. คัดลอกหัวคิวเก็บไว้ก่อน แล้วค่อย pop ถ้า pop ก่อนจะไม่มีอะไรให้ดูแล้ว

แถวสุดท้ายสำคัญที่สุด STL ไม่ได้ป้องกันการ pop ตอนว่างให้ ดูผลได้ในส่วนลองทำให้พัง

ความเร็วต่างกันไหม ลองวัดด้วย clock() แบบหน้า 4.5 ให้ทั้งสองแบบ push 10 ล้านตัวแล้ว pop จนหมด

#include <bits/stdc++.h>
using namespace std;

int arr[10000000];

int main()
{
    int n = 10000000;
    clock_t start = clock();
    int size = 0;
    long long sum = 0;
    for (int i = 0; i < n; i++) {
        arr[size] = i;
        size++;
    }
    while (size > 0) {
        size--;
        sum += arr[size];
    }
    clock_t middle = clock();
    stack<int> st;
    for (int i = 0; i < n; i++) {
        st.push(i);
    }
    while (!st.empty()) {
        sum += st.top();
        st.pop();
    }
    clock_t end = clock();
    cout << "sum = " << sum << '\n';
    cout << fixed << setprecision(3);
    cout << "array: " << (double)(middle - start) / CLOCKS_PER_SEC << " seconds\n";
    cout << "stack: " << (double)(end - middle) / CLOCKS_PER_SEC << " seconds\n";
    return 0;
}

0xC0000005 คือรหัสของ Windows ที่แปลว่าโปรแกรมไปอ่านหน่วยความจำที่ไม่ใช่ของมัน top() ไม่ได้ตรวจว่ามีของหรือเปล่า มันแค่ไปอ่านช่องที่ควรเป็นตัวบนสุด ซึ่งไม่มีอยู่จริง ส่งให้ grader จะได้ผล runtime error

สรุป

  • stack<int> s; ใช้ push pop top empty size
  • queue<int> q; ใช้ push pop front back empty size
  • pop() ไม่คืนค่า ดูด้วย top() หรือ front() ก่อนแล้วค่อย pop
  • ทุกคำสั่งใช้เวลาคงที่ O(1) ไม่ต้องเลือกขนาด และเก็บชนิดไหนก็ได้
