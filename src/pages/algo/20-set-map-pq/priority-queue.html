20.5 priority_queue

ห้องฉุกเฉินไม่ได้รักษาตามลำดับที่มาถึง แต่รักษาคนที่อาการหนักที่สุดก่อน คนไข้ใหม่เข้ามาตลอดเวลา และทุกครั้งที่หมอว่างต้องรู้ทันทีว่าใครหนักที่สุดในตอนนั้น งานแบบ "เพิ่มของเข้าไปเรื่อยๆ แล้วหยิบตัวที่มากที่สุดออก" เจอบ่อยมากในโจทย์ เช่น จำลองเหตุการณ์ตามเวลา รวมข้อมูลหลายแถว หรือหาเส้นทางสั้นสุดในบทที่ 28 priority_queue (คิวตามลำดับความสำคัญ) ทำงานนี้อย่างเดียว และทำได้เร็วมาก

หยิบค่ามากสุดก่อน

#include <bits/stdc++.h>
using namespace std;

int main()
{
    priority_queue<int> patients;
    patients.push(3);
    patients.push(9);
    patients.push(5);
    patients.push(9);
    patients.push(1);
    cout << "waiting " << patients.size() << '\n';
    while (!patients.empty()) {
        cout << patients.top() << ' ';
        patients.pop();
    }
    cout << '\n';
    return 0;
}

  • push(x) ใส่ค่าเข้าไป ค่าซ้ำได้
  • top() ดูค่าที่มากที่สุดโดยไม่เอาออก
  • pop() เอาค่าที่มากที่สุดออก ไม่คืนค่าอะไรกลับมา จึงต้อง top() ก่อนแล้วค่อย pop() เสมอ
  • size() กับ empty() เหมือน container อื่น

ชื่อมีคำว่า queue และใช้ push pop เหมือน queue ในบทที่ 17 แต่ตัวที่ออกไม่ใช่ตัวที่เข้ามาก่อน เป็นตัวที่มากที่สุด สิ่งที่ priority_queue ทำไม่ได้คือ ดูตัวอื่นนอกจากตัวบนสุด วนดูทุกตัว ค้นหาค่า หรือลบค่าที่อยู่ตรงกลาง ถ้าต้องการสิ่งเหล่านี้ให้ใช้ multiset แทน

หยิบค่าน้อยสุดก่อน

ค่าเริ่มต้นคือมากสุดก่อน ถ้าอยากได้น้อยสุดก่อน ต้องบอกสามอย่างในวงเล็บมุม ชนิดของค่า ที่เก็บข้างใน (ใช้ vector เสมอ) และวิธีเปรียบเทียบ greater<int> ซึ่งกลับลำดับให้

#include <bits/stdc++.h>
using namespace std;

int main()
{
    priority_queue<int, vector<int>, greater<int>> lap_times;
    lap_times.push(58);
    lap_times.push(52);
    lap_times.push(61);
    lap_times.push(49);
    while (!lap_times.empty()) {
        cout << lap_times.top() << ' ';
        lap_times.pop();
    }
    cout << '\n';
    return 0;
}

อีกวิธีที่เห็นบ่อยในโค้ดของคนอื่นคือใส่ค่าติดลบเข้าไปใน priority_queue ปกติ push(-x) แล้วตอนดึงออกก็กลับเครื่องหมาย -top() ค่าที่น้อยที่สุดจะกลายเป็นค่าติดลบที่มากที่สุด ใช้ได้กับตัวเลข แต่อ่านยากกว่าและลืมกลับเครื่องหมายได้ง่าย เอกสารนี้ใช้ greater

เก็บ pair

บ่อยครั้งที่ต้องเก็บมากกว่าตัวเลขตัวเดียว เช่น ระดับความหนักของอาการคู่กับหมายเลขบัตรคิว priority_queue ของ pair เทียบ first ก่อน ถ้าเท่ากันค่อยเทียบ second เหมือน sort ของ pair ในบทที่ 16 กติกาของห้องฉุกเฉินคือ หนักกว่าได้ก่อน ถ้าหนักเท่ากัน คนที่มาก่อน (บัตรคิวเลขน้อย) ได้ก่อน

#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int q;
    cin >> q;
    priority_queue<pair<int, int>> waiting;
    int next_ticket = 1;
    for (int i = 0; i < q; i++) {
        char op;
        cin >> op;
        if (op == '+') {
            int level;
            cin >> level;
            waiting.push({level, -next_ticket}); // (1)!
            next_ticket++;
        } else if (waiting.empty()) {
            cout << "nobody\n";
        } else {
            pair<int, int> top = waiting.top();
            waiting.pop();
            cout << "ticket " << -top.second << " level " << top.first << '\n';
        }
    }
    return 0;
}
  1. priority_queue ปกติเอาค่ามากออกก่อนทั้ง first และ second ถ้าเก็บเลขบัตรตรงๆ บัตรเลขมาก (มาทีหลัง) จะได้ก่อน จึงเก็บเป็นค่าติดลบ บัตร 1 กลายเป็น -1 ซึ่งมากกว่า -3 ตอนพิมพ์ค่อยกลับเครื่องหมาย

เร็วแค่ไหน

ข้างใน priority_queue คือ heap ต้นไม้ที่ทุกโหนดมากกว่าหรือเท่ากับลูกของมัน ค่ามากสุดจึงอยู่ที่ยอดเสมอ เก็บอยู่ใน vector ธรรมดาโดยไม่ต้องมี pointer เลย push กับ pop แค่สลับของขึ้นลงตามความสูงของต้นไม้ วิธีทำ heap เองทีละขั้นจะได้เรียนในบทที่ 26

จำลองเหตุการณ์ตามเวลา

โรงเรียนมีการประชุม N รายการ แต่ละรายการมีเวลาเริ่มและเวลาจบเป็นนาที ประชุมที่เวลาซ้อนกันใช้ห้องเดียวกันไม่ได้ แต่ห้องที่ประชุมจบเวลา 600 พอดี ประชุมถัดไปที่เริ่ม 600 ใช้ต่อได้ ต้องมีห้องประชุมอย่างน้อยกี่ห้อง

คิดแบบเดินตามเวลา เรียงการประชุมตามเวลาเริ่ม แล้วจัดทีละรายการ ทุกครั้งที่มีประชุมจะเริ่ม คำถามเดียวที่สำคัญคือ ในห้องที่มีอยู่ ห้องไหนว่างเร็วที่สุด ถ้าห้องนั้นว่างทันก็ใช้ห้องนั้น ถ้าไม่ทันแปลว่าทุกห้องยังไม่ว่าง ต้องเปิดห้องใหม่ คำถาม "ห้องไหนว่างเร็วที่สุด" คือค่าน้อยสุดของเวลาจบ งานของ priority_queue แบบน้อยสุดก่อนพอดี

#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<pair<int, int>> meetings(n);
    for (int i = 0; i < n; i++) {
        cin >> meetings[i].first >> meetings[i].second;
    }
    sort(meetings.begin(), meetings.end()); // (1)!
    priority_queue<int, vector<int>, greater<int>> ends; // (2)!
    int rooms = 0;
    for (auto meeting : meetings) {
        int start = meeting.first;
        int finish = meeting.second;
        if (!ends.empty() && ends.top() <= start) { // (3)!
            ends.pop();
        } else {
            rooms++;
        }
        ends.push(finish);
    }
    cout << rooms << '\n';
    return 0;
}
  1. pair เรียงตาม first ซึ่งคือเวลาเริ่ม
  2. เก็บเวลาจบของทุกห้องที่เปิดแล้ว ห้องละหนึ่งค่า ตัวบนสุดคือห้องที่ว่างเร็วที่สุด
  3. ห้องที่ว่างเร็วที่สุดว่างทันก่อนประชุมนี้เริ่ม ใช้ห้องนั้นต่อ เอาเวลาจบเก่าออก แล้วใส่เวลาจบใหม่ในบรรทัดที่ 25 ถ้าไม่ทัน เปิดห้องใหม่

โจทย์จำลองเหตุการณ์ส่วนใหญ่มีรูปแบบนี้ เก็บ "เหตุการณ์ถัดไปของแต่ละสิ่ง" ไว้ใน heap แล้วหยิบเหตุการณ์ที่เกิดเร็วที่สุดออกมาจัดการทีละอัน แบบฝึกหัดข้อ 9 เป็นร้านซักผ้าที่ต้องใช้ heap สองตัว

รวมรายการที่เรียงแล้ว k รายการ

ร้านสาขา k แห่งส่งยอดขายรายบิลมา แต่ละสาขาเรียงจากน้อยไปมากแล้ว สำนักงานใหญ่อยากได้รายการรวมที่เรียงแล้ว ค่าที่น้อยที่สุดของทั้งหมดต้องเป็นตัวแรกของสาขาใดสาขาหนึ่ง จึงพอที่จะดูแค่ตัวหน้าสุดของทุกสาขา ใส่ตัวหน้าของทุกสาขาลง heap แบบน้อยสุดก่อน หยิบตัวน้อยสุดออกมา แล้วใส่ตัวถัดไปของสาขาเดียวกันเข้าไปแทน

#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int k;
    cin >> k;
    vector<vector<int>> lists(k);
    for (int i = 0; i < k; i++) {
        int length;
        cin >> length;
        lists[i].resize(length);
        for (int j = 0; j < length; j++) {
            cin >> lists[i][j];
        }
    }
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> heads; // (1)!
    vector<int> next_index(k, 0); // (2)!
    for (int i = 0; i < k; i++) {
        if (!lists[i].empty()) {
            heads.push({lists[i][0], i});
            next_index[i] = 1;
        }
    }
    while (!heads.empty()) {
        int value = heads.top().first;
        int list = heads.top().second;
        heads.pop();
        cout << value << ' ';
        if (next_index[list] < (int)lists[list].size()) { // (3)!
            heads.push({lists[list][next_index[list]], list});
            next_index[list]++;
        }
    }
    cout << '\n';
    return 0;
}
  1. เก็บคู่ (ค่า, หมายเลขสาขา) ต้องรู้ว่าค่าที่หยิบออกมาเป็นของสาขาไหน จะได้ใส่ตัวถัดไปของสาขานั้นถูก ชื่อชนิดยาวเพราะต้องเขียน pair<int, int> ซ้ำสามที่
  2. next_index[i] คือตำแหน่งของตัวถัดไปในสาขา i ที่ยังไม่ได้ใส่ลง heap
  3. สาขานี้ยังมีของเหลือ ใส่ตัวถัดไปเข้า heap ถ้าหมดแล้วก็ไม่ใส่ heap จะเล็กลงเรื่อยๆ จนว่าง

heap มีของไม่เกิน k ตัวตลอดเวลา ถ้ารวมทุกสาขามี T ค่า ใช้เวลา O(T log k) ถ้าเอาทุกค่ามาต่อกันแล้ว sort จะเป็น O(T log T) ซึ่งในทางปฏิบัติก็เร็วพอเมื่อทุกค่าอยู่ในหน่วยความจำได้ วิธี heap ได้เปรียบจริงในสองกรณี คือข้อมูลใหญ่เกินกว่าจะโหลดมาเรียงพร้อมกัน เช่น ไฟล์หลายไฟล์ที่อ่านทีละบรรทัด และกรณีที่ต้องการแค่ M ค่าแรก ซึ่งหยุดหลังหยิบออก M ครั้งได้เลย ใช้เวลาแค่ O(k + M log k)

ค่ามัธยฐานแบบไหลเข้า

แอปวัดเวลาวิ่งรับเวลาของนักวิ่งเข้ามาทีละคน และหลังทุกคนต้องแสดงค่ามัธยฐานของเวลาทั้งหมดที่ได้มาแล้ว ถ้ามีจำนวนคู่ ให้ใช้ตัวกลางตัวที่น้อยกว่า วิธีตรงไปตรงมาคือแทรกเวลาใหม่ลงอาเรย์ที่เรียงไว้แล้วอ่านตัวตรงกลาง แต่การแทรกต้องเลื่อนของครึ่งแถวโดยเฉลี่ย O(N) ต่อครั้ง รวม O(N2)

เคล็ดลับคือแบ่งค่าทั้งหมดเป็นสองกอง กองล่างเก็บครึ่งที่น้อย กองบนเก็บครึ่งที่มาก ค่ามัธยฐานคือค่ามากสุดของกองล่าง กองล่างจึงเป็น heap แบบมากสุดก่อน กองบนเป็น heap แบบน้อยสุดก่อน แล้วรักษาขนาดให้กองล่างมีเท่ากับกองบน หรือมากกว่าหนึ่งตัว

#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    priority_queue<int> lower; // (1)!
    priority_queue<int, vector<int>, greater<int>> upper; // (2)!
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        if (lower.empty() || x <= lower.top()) {
            lower.push(x);
        } else {
            upper.push(x);
        }
        if (lower.size() > upper.size() + 1) { // (3)!
            upper.push(lower.top());
            lower.pop();
        } else if (upper.size() > lower.size()) {
            lower.push(upper.top());
            upper.pop();
        }
        cout << lower.top() << '\n';
    }
    return 0;
}
  1. กองล่าง ครึ่งที่น้อย ตัวบนสุดคือตัวที่มากที่สุดในครึ่งนี้
  2. กองบน ครึ่งที่มาก ตัวบนสุดคือตัวที่น้อยที่สุดในครึ่งนี้ ทุกค่าในกองล่างจึงไม่เกินทุกค่าในกองบน
  3. ใส่ค่าใหม่แล้วขนาดอาจเสียสมดุล ย้ายตัวบนสุดข้ามกองหนึ่งตัวก็กลับมาสมดุล และการย้ายตัวบนสุดไม่ทำให้กฎเรื่องกองล่างไม่เกินกองบนเสีย

แบบที่ 1 ลืมว่าค่าเริ่มต้นคือมากสุดก่อน

อยากได้เวลาวิ่งที่เร็วที่สุด K คน ซึ่งคือเวลาน้อยที่สุด แต่ใช้ priority_queue แบบปกติ

#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, k;
    cin >> n >> k;
    priority_queue<int> times;
    for (int i = 0; i < n; i++) {
        int t;
        cin >> t;
        times.push(t);
    }
    cout << "fastest " << k << ":";
    for (int i = 0; i < k; i++) {
        cout << ' ' << times.top();
        times.pop();
    }
    cout << '\n';
    return 0;
}

สรุป

  • priority_queue<T> ให้ค่ามากสุดที่ top() push กับ pop เป็น O(log N) top เป็น O(1)
  • อยากได้น้อยสุดก่อน ใช้ priority_queue<T, vector<T>, greater<T>>
  • เก็บ pair ได้ เทียบ first ก่อนแล้วค่อย second ใช้ค่าติดลบเมื่อเงื่อนไขของสองค่าต่างทิศกัน
  • เช็ก empty() ก่อน top() และ pop() เสมอ
  • งานจำลองตามเวลา เก็บเหตุการณ์ถัดไปไว้ใน heap แล้วหยิบตัวที่เกิดเร็วที่สุด
  • รวม k รายการที่เรียงแล้วใช้ heap ขนาด k ค่ามัธยฐานแบบไหลเข้าใช้ heap สองกอง
