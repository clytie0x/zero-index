20.1 set เก็บของไม่ซ้ำ

ด่านเก็บเงินบนทางด่วนบันทึกเลขทะเบียนรถทุกคันที่ผ่าน วันหนึ่งมีรถผ่านเป็นแสนครั้ง คันเดียวกันผ่านไปกลับหลายรอบ ถ้าอยากรู้ว่ามีรถกี่คัน หรือรถทะเบียนนี้เคยผ่านแล้วหรือยัง เครื่องมือที่เรามีตอนนี้ยังไม่พอดีมือ อาเรย์นับในหน้า 5.3 ใช้ไม่ได้เมื่อค่าใหญ่ถึง 109 การเรียงข้อมูลในบทที่ 16 ใช้ได้ถ้าข้อมูลมาครบก่อนแล้วค่อยตอบ แต่ถ้ารถผ่านไปเรื่อยๆ และมีคนถามแทรกเข้ามาตลอดเวลา จะเรียงใหม่ทุกครั้งก็ไม่ทัน

set (แปลว่าเซต แบบเซตในวิชาคณิตศาสตร์) คือที่เก็บของที่ไม่ยอมให้มีค่าซ้ำ และจัดเรียงค่าจากน้อยไปมากให้ตลอดเวลา เพิ่ม ลบ หรือค้นหาค่าหนึ่งตัวใช้เวลา O(log N) ซึ่งเร็วพอจะทำเป็นแสนครั้งในเสี้ยววินาที

set ตัวแรก

#include <bits/stdc++.h>
using namespace std;

int main()
{
    set<int> plates; // (1)!
    plates.insert(4521); // (2)!
    plates.insert(1088);
    plates.insert(7310);
    plates.insert(1088);
    plates.insert(4521);
    plates.insert(2950);
    cout << "distinct: " << plates.size() << '\n';
    for (int plate : plates) { // (3)!
        cout << plate << ' ';
    }
    cout << '\n';
    return 0;
}
  1. ประกาศ set ที่เก็บ int เขียนชนิดในวงเล็บมุมเหมือน vector<int> ตอนเริ่มยังว่าง
  2. insert ใส่ค่าลงไป ถ้ามีค่านั้นอยู่แล้ว set ไม่ทำอะไรเลย
  3. วนด้วย for แบบเดียวกับ vector ได้ แต่ค่าจะออกมาเรียงจากน้อยไปมากเสมอ ไม่ใช่ตามลำดับที่ใส่

ใส่ไปหกครั้ง เหลือสี่ค่า เพราะ 1088 กับ 4521 ใส่ซ้ำ size() จึงเป็นจำนวนค่าที่ไม่ซ้ำกันพอดี โจทย์ "มีกี่แบบ" หรือ "มีกี่คนที่ไม่ซ้ำ" จึงจบได้ในไม่กี่บรรทัด

set เก็บอะไรก็ได้ที่เปรียบเทียบมากน้อยได้ เช่น set<string> จะเรียงข้อความตามพจนานุกรม หรือ set<pair<int, int>> จะเรียงตาม first ก่อนแล้วค่อย second แบบเดียวกับที่ sort เรียง pair ในบทที่ 16

ถามว่ามีไหมด้วย count และ find

สหกรณ์โรงเรียนมีรายชื่อรหัสสมาชิก N คน แล้วมีคนมาถามว่ารหัสนี้เป็นสมาชิกหรือเปล่า Q ครั้ง count(x) ของ set คืนจำนวนค่า x ที่มีอยู่ ซึ่งเป็นได้แค่ 0 หรือ 1 เพราะ set ไม่เก็บค่าซ้ำ

#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, q;
    cin >> n >> q;
    set<int> members;
    for (int i = 0; i < n; i++) {
        int id;
        cin >> id;
        members.insert(id);
    }
    for (int i = 0; i < q; i++) {
        int id;
        cin >> id;
        if (members.count(id) == 1) {
            cout << id << " yes\n";
        } else {
            cout << id << " no\n";
        }
    }
    return 0;
}

อีกวิธีคือ find(x) ซึ่งคืน iterator ตัวชี้ตำแหน่งแบบเดียวกับที่ lower_bound ของ vector คืนในหน้า 19.2 ถ้าไม่เจอจะคืน end() ซึ่งเป็นตำแหน่งเลยตัวสุดท้ายไปหนึ่งช่อง ไม่ใช่ค่าจริงในเซต อ่านค่าที่ iterator ชี้ด้วย * ข้างหน้า

#include <bits/stdc++.h>
using namespace std;

int main()
{
    set<int> scores = {72, 85, 91, 60};
    auto it = scores.find(85);
    if (it != scores.end()) {
        cout << "found " << *it << '\n';
    }
    it = scores.find(100);
    if (it == scores.end()) {
        cout << "100 not found\n";
    }
    cout << "smallest " << *scores.begin() << '\n';
    cout << "largest " << *scores.rbegin() << '\n';
    return 0;
}

  • auto ให้ compiler ใส่ชนิดของ iterator ให้เองตามหน้า 15.5 ชนิดจริงยาวมาก
  • begin() ชี้ค่าที่น้อยที่สุด rbegin() (reverse begin) ชี้ค่าที่มากที่สุด ทั้งสองตัวใช้ได้เฉพาะตอนที่ set ไม่ว่าง

ถ้าแค่อยากรู้ว่ามีหรือไม่มี ใช้ count อ่านง่ายกว่า ส่วน find ใช้เมื่ออยากได้ตำแหน่งไปทำอย่างอื่นต่อ เช่น ลบตัวนั้นทิ้ง

ลบด้วย erase

ห้องสมุดบันทึกว่าใครอยู่ในห้องบ้าง + id คือมีคนเข้า - id คือมีคนออก หลังทุกเหตุการณ์ บรรณารักษ์อยากรู้ว่าในห้องมีกี่คน และรหัสน้อยสุดกับมากสุดเป็นเท่าไร

#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int q;
    cin >> q;
    set<int> inside;
    for (int i = 0; i < q; i++) {
        char op;
        int id;
        cin >> op >> id;
        if (op == '+') {
            inside.insert(id);
        } else {
            inside.erase(id);
        }
        if (inside.empty()) {
            cout << "empty\n";
        } else {
            cout << inside.size() << " people, ids " << *inside.begin() << " to " << *inside.rbegin() << '\n';
        }
    }
    return 0;
}

ทุกเหตุการณ์ได้ค่าน้อยสุดและมากสุดมาทันที ถ้าใช้ vector ต้องไล่หาใหม่ทุกครั้ง และการลบตรงกลาง vector ก็ต้องเลื่อนของทั้งแถว

เดินใน set ด้วย iterator

set ไม่มีช่องหมายเลข 0 1 2 แบบ vector จะไปตัวถัดไปต้องเลื่อน iterator ทีละก้าว ++it คือไปตัวถัดไป next(it) คืน iterator ของตัวถัดไปโดยไม่เปลี่ยน it และ prev(it) คืนของตัวก่อนหน้า

#include <bits/stdc++.h>
using namespace std;

int main()
{
    set<int> heights = {158, 171, 149, 165};
    for (auto it = heights.begin(); it != heights.end(); ++it) {
        cout << *it << ' ';
    }
    cout << '\n';
    auto last = prev(heights.end());
    cout << "tallest " << *last << '\n';
    cout << "second tallest " << *prev(last) << '\n';
    cout << "second shortest " << *next(heights.begin()) << '\n';
    return 0;
}

tallest 171
second tallest 165
second shortest 158

prev(heights.end()) คือตัวสุดท้ายจริงๆ ได้ค่าเดียวกับ *heights.rbegin() วิธีนี้จะใช้บ่อยมากในหน้า 20.2 ส่วนการกระโดดไปตัวที่ k ตรงๆ ทำไม่ได้ ต้องเดินไปทีละก้าว k ครั้ง

ทำไมเร็ว และเร็วแค่ไหน

ข้างใน set ไม่ได้เก็บเป็นแถวแบบอาเรย์ แต่เก็บเป็นต้นไม้ค้นหาแบบทวิภาค (binary search tree) ทุกโหนดมีค่าที่น้อยกว่าอยู่ทางซ้าย และค่าที่มากกว่าอยู่ทางขวา การหาค่าหนึ่งตัวจึงเริ่มจากยอด แล้วเลือกเลี้ยวซ้ายหรือขวาทีละชั้น เหมือน binary search ในบทที่ 19 ที่ตัดครึ่งไปเรื่อยๆ

เจ็ดค่าใช้การเทียบแค่สามครั้ง set ใน C++ คอยจัดต้นไม้ให้สมดุลเองทุกครั้งที่ใส่หรือลบ ความสูงของต้นไม้จึงอยู่ราว log2 N เสมอ วิธีสร้างต้นไม้แบบนี้และการทำให้สมดุลจะได้เรียนในบทที่ 26 บทนี้ใช้งานอย่างเดียว

ตัวอย่าง ลบตัวซ้ำ

รับตัวเลข N ตัว แล้วพิมพ์ตัวที่ไม่ซ้ำ ตามลำดับที่เจอครั้งแรก ถ้าพิมพ์จาก set ตรงๆ จะได้ลำดับจากน้อยไปมาก ซึ่งไม่ใช่สิ่งที่โจทย์ถาม จึงใช้ set เป็นแค่สมุดจดว่าเคยเห็นแล้ว ส่วนลำดับเก็บไว้ใน vector

#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    set<int> seen;
    vector<int> order;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        if (seen.count(x) == 0) {
            seen.insert(x);
            order.push_back(x);
        }
    }
    for (int x : order) {
        cout << x << ' ';
    }
    cout << '\n';
    return 0;
}

ลองทำให้พัง

แบบที่ 1 ใช้ set ทั้งที่ค่าซ้ำมีความหมาย

อยากได้คะแนนต่ำสุดของห้อง เลยใส่คะแนนทุกคนลง set เพราะ begin() ให้ค่าน้อยสุดมาฟรีๆ แล้วใช้ size() เป็นจำนวนนักเรียนไปด้วย

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int n;
    cin >> n;
    set<int> scores;
    int total = 0;
    for (int i = 0; i < n; i++) {
        int score;
        cin >> score;
        scores.insert(score);
        total += score;
    }
    cout << "students " << scores.size() << '\n';
    cout << "lowest " << *scores.begin() << '\n';
    cout << fixed << setprecision(2);
    cout << "average " << (double)total / scores.size() << '\n';
    return 0;
}

สรุป

  • set<T> เก็บค่าไม่ซ้ำ และเรียงจากน้อยไปมากเสมอ
  • insert erase count find ใช้เวลา O(log N) size กับ empty ใช้ O(1)
  • ค่าน้อยสุด *s.begin() ค่ามากสุด *s.rbegin() หรือ *prev(s.end()) เช็ก empty() ก่อนเสมอ
  • set ไม่มี s[i] และบวกเลขเข้ากับ iterator ไม่ได้ ใช้ next prev แทน
  • ข้อมูลมาครบก่อนแล้วค่อยตอบ sort เร็วกว่า ข้อมูลเปลี่ยนไปเรื่อยๆ ระหว่างตอบ ใช้ set
  • ค่าซ้ำมีความหมายเมื่อไร set จะตัดทิ้งเงียบๆ ทำให้คำตอบผิด
