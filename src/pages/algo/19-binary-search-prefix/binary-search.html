19.1 binary search แบบวนลูป

หน้า 9.4 เขียน binary search ด้วย recursion เพื่อหาว่าเลขที่ต้องการอยู่ช่องไหน โจทย์ค่าย 2 ถามละเอียดกว่านั้น เช่น มีกี่คนที่ได้คะแนนน้อยกว่า 60 หรือคนแรกที่ได้อย่างน้อย 52 คะแนนอยู่ลำดับที่เท่าไร คำถามแบบนี้ต้องตอบถูกทั้งตอนที่มีค่าซ้ำ และตอนที่ค่าที่ถามไม่มีอยู่ในข้อมูลเลย หน้านี้เปลี่ยน binary search เป็นลูป แล้วเขียนตามกฎข้อเดียวที่ทำให้ทุกขอบถูกโดยไม่ต้องเดา

จาก recursion เป็นลูป

binary search ในหน้า 9.4 เรียกตัวเองแค่ครั้งเดียวต่อชั้น สิ่งที่เปลี่ยนระหว่างชั้นมีแค่ low กับ high จึงเปลี่ยนเป็นลูปได้ตรงๆ แทนที่จะเรียกตัวเองด้วยช่วงใหม่ ก็แก้ค่า low หรือ high แล้ววนกลับไปทำซ้ำ

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int n, target;
    cin >> n >> target;
    vector<int> ids(n);
    for (int i = 0; i < n; i++) {
        cin >> ids[i];
    }
    int low = 0, high = n - 1;
    int position = -1;
    while (low <= high) {
        int mid = (low + high) / 2;
        cout << "low=" << low << " high=" << high << " mid=" << mid << " a[mid]=" << ids[mid] << '\n';
        if (ids[mid] == target) {
            position = mid;
            break;
        }
        if (ids[mid] < target) {
            low = mid + 1;
        } else {
            high = mid - 1;
        }
    }
    cout << "position = " << position << '\n';
    return 0;
}

ช่องแรกที่มีค่าอย่างน้อย x

เปลี่ยนคำถามให้มีคำตอบเสมอ คือ ช่องแรกที่ a[i] >= x อยู่ช่องไหน ถ้าไม่มีเลยให้ตอบ n ลองเขียนผลของเงื่อนไข a[i] >= 52 ไว้ใต้แต่ละช่อง

ข้อมูลเรียงจากน้อยไปมาก ผลจึงเป็น "ไม่" ติดกันทางซ้าย แล้วเป็น "ใช่" ติดกันทางขวาเสมอ ไม่มีทางสลับไปมา งานของเราคือหา รอยต่อ ระหว่างสองฝั่ง คำตอบคือช่อง "ใช่" ช่องแรก ถ้าทุกช่องตอบ "ไม่" รอยต่อก็อยู่หลังช่องสุดท้าย คือ n

ใช้ตัวแปร lo กับ hi แล้วรักษาสัญญาข้อเดียวไว้ตลอดลูป

สัญญาของ lo กับ hi

ทุกช่อง ก่อน lo ตอบ "ไม่" ทุกช่อง ตั้งแต่ hi เป็นต้นไปตอบ "ใช่" รอยต่อจึงอยู่ระหว่าง lo ถึง hi เสมอ

เริ่มที่ lo = 0 กับ hi = n สัญญาเป็นจริงทันที เพราะยังไม่มีช่องไหนอยู่ก่อนช่อง 0 และไม่มีช่องไหนอยู่ตั้งแต่ช่อง n แต่ละรอบดูช่อง mid ตรงกลาง

  • ถ้า a[mid] >= x ช่อง mid และทุกช่องทางขวาตอบ "ใช่" ย้าย hi = mid ห้ามเป็น mid - 1 เพราะ mid อาจเป็นช่อง "ใช่" ช่องแรกพอดี
  • ถ้าไม่ ช่อง mid และทุกช่องทางซ้ายตอบ "ไม่" ย้าย lo = mid + 1

วนจน lo เท่ากับ hi ที่ที่รอยต่ออยู่ได้เหลือจุดเดียว นั่นคือคำตอบ

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int n, x;
    cin >> n >> x;
    vector<int> scores(n);
    for (int i = 0; i < n; i++) {
        cin >> scores[i];
    }
    int lo = 0, hi = n; // (1)!
    while (lo < hi) { // (2)!
        int mid = lo + (hi - lo) / 2; // (3)!
        cout << "lo=" << lo << " hi=" << hi << " mid=" << mid << " a[mid]=" << scores[mid] << '\n';
        if (scores[mid] >= x) {
            hi = mid;
        } else {
            lo = mid + 1;
        }
    }
    cout << "first index with score >= " << x << " is " << lo << '\n';
    return 0;
}
  1. hi เริ่มที่ n ไม่ใช่ n - 1 เพื่อให้ตอบ n ได้เมื่อไม่มีช่องไหนถึง x ช่อง n ไม่มีอยู่จริง แต่ไม่เคยถูกอ่าน เพราะ mid น้อยกว่า hi เสมอ
  2. ช่วงที่รอยต่ออยู่ได้คือ lo ถึง hi วนจนเหลือจุดเดียว ไม่ใช่ <= แบบโปรแกรมก่อนหน้า
  3. ได้ค่าเดียวกับ (lo + hi) / 2 แต่ไม่ล้นเมื่อ lo กับ hi ใหญ่ ดูในส่วนลองทำให้พัง

ทำไมลูปจบแน่นอน

ตอนที่ lo < hi ค่า mid อยู่ระหว่าง lo ถึง hi - 1 เสมอ เพราะการหารปัดเศษลง ทั้ง hi = mid และ lo = mid + 1 จึงทำให้ช่วงสั้นลงอย่างน้อยหนึ่งทุกรอบ และจริงๆ แล้วสั้นลงราวครึ่งหนึ่ง ข้อมูล n ตัววนราว log2 n รอบ แสนตัวไม่เกิน 17 รอบ ล้านตัวไม่เกิน 20 รอบ ค้นหนึ่งครั้งจึงเป็น O(log n) เทียบกับไล่ทีละช่องที่เป็น O(n)

นับด้วยรอยต่อ

ช่องแรกที่ถึง x บอกจำนวนช่องที่ น้อยกว่า x ไปในตัว เพราะทุกช่องก่อนหน้ามันตอบ "ไม่" ทั้งหมด ถ้าเปลี่ยนเงื่อนไขเป็น a[mid] > x จะได้ช่องแรกที่ มากกว่า x ซึ่งก็คือจำนวนช่องที่ไม่เกิน x สองค่านี้ลบกันได้จำนวนช่องที่เท่ากับ x พอดี

#include <bits/stdc++.h>
using namespace std;

int first_at_least(const vector<int> &a, int x) // (1)!
{
    int lo = 0, hi = a.size();
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;
        if (a[mid] >= x) {
            hi = mid;
        } else {
            lo = mid + 1;
        }
    }
    return lo;
}

int first_greater(const vector<int> &a, int x)
{
    int lo = 0, hi = a.size();
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;
        if (a[mid] > x) { // (2)!
            hi = mid;
        } else {
            lo = mid + 1;
        }
    }
    return lo;
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, q;
    cin >> n >> q;
    vector<int> scores(n);
    for (int i = 0; i < n; i++) {
        cin >> scores[i];
    }
    for (int k = 0; k < q; k++) {
        int x;
        cin >> x;
        int less = first_at_least(scores, x);
        int at_most = first_greater(scores, x);
        cout << "x=" << x << " less=" << less << " at_most=" << at_most << " equal=" << at_most - less << '\n';
    }
    return 0;
}
  1. ส่ง vector แบบ reference ตามบทที่ 15 จะได้ไม่ต้องคัดลอกข้อมูลแสนตัวทุกครั้งที่เรียก const บอกว่าฟังก์ชันนี้จะไม่แก้ข้อมูล ถ้าเผลอแก้ compiler จะฟ้อง
  2. ต่างจากฟังก์ชันแรกแค่ >= กับ > ตัวเดียว

เร็วขึ้นแค่ไหน

วัดบนโน้ตบุ๊ก AMD Ryzen AI 9 HX 370 กับ Windows 11 และ GCC 14.2 ของ Code::Blocks คะแนน N = 100000 ตัว สุ่มตั้งแต่ 1 ถึง 109 แล้วเรียงไว้ มีคำถาม Q = 100000 ข้อ แต่ละข้อถามว่ามีกี่ตัวที่น้อยกว่า x วิธีตรงไปตรงมาไล่จากช่องซ้ายสุดจนเจอตัวที่ถึง x ซึ่งเฉลี่ยต้องไล่ครึ่งอาเรย์ทุกคำถาม เป็น O(NQ) ส่วน binary search เป็น O(Q log N) รันแต่ละแบบ 5 รอบ ตารางเป็นค่ากลางของ 5 รอบ

แบบที่ 2 mid ล้น

ในหน้านี้ hi ไม่เกิน n จึงไม่มีปัญหา แต่หน้า 19.3 จะค้นบนช่วงของคำตอบที่ใหญ่ถึงพันล้าน ถ้าหาจุดกลางด้วย (lo + hi) / 2 แบบ int

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int lo, hi;
    cin >> lo >> hi;
    cout << (lo + hi) / 2 << '\n';
    cout << lo + (hi - lo) / 2 << '\n';
    return 0;
}

สรุป

  • เปลี่ยนคำถามเป็น "รอยต่อของเงื่อนไข" ที่เป็น "ไม่" ทางซ้ายและ "ใช่" ทางขวา แล้วหาช่อง "ใช่" ช่องแรก
  • เริ่ม lo = 0 กับ hi = n วน while (lo < hi) ถ้า "ใช่" ให้ hi = mid ถ้า "ไม่" ให้ lo = mid + 1 จบแล้วคำตอบคือ lo
  • ช่องแรกที่ >= x คือจำนวนที่น้อยกว่า x ช่องแรกที่ > x คือจำนวนที่ไม่เกิน x สองค่าลบกันได้จำนวนที่เท่ากับ x
  • ค้นหนึ่งครั้งเป็น O(log n) แสนตัวไม่เกิน 17 รอบ
  • ทุกรอบช่วงต้องสั้นลง lo = mid ทำให้ลูปไม่จบ
  • หาจุดกลางด้วย lo + (hi - lo) / 2 หรือใช้ long long เมื่อช่วงใหญ่
