19.4 binary search บนจำนวนจริง

บางโจทย์คำตอบเป็นทศนิยม เช่น ระดับน้ำ เวลาที่ใช้ ความเร็ว ค่าเหล่านี้ไม่มีคำว่า "ค่าถัดไป" แบบจำนวนเต็ม ช่วงที่ค้นจึงไม่มีวันหดจนเหลือค่าเดียวพอดี หน้านี้ใช้ความคิดเดิมจากหน้า 19.3 กับคำตอบที่เป็น double แล้วเปลี่ยนวิธีหยุดลูปเป็นวนจำนวนรอบที่กำหนดไว้ ซึ่งปลอดภัยกว่าการรอให้ช่วงแคบลงเอง

ลองกับรากที่สอง

รากที่สองของ x คือค่า m ที่ m × m = x ใช้คำถามใช่ไม่ใช่ว่า m * m >= x ไหม ยิ่ง m มากยิ่งตอบใช่ จึงเป็นทางเดียว คำตอบอยู่ระหว่าง 0 ถึง x ถ้า x ไม่น้อยกว่า 1 แต่ถ้า x น้อยกว่า 1 เช่น 0.25 รากที่สองคือ 0.5 ซึ่งมากกว่า x จึงให้ hi เป็นค่าที่มากกว่าระหว่าง 1 กับ x

ต่างจากจำนวนเต็มสองจุด ไม่มีบวกหนึ่งหรือลบหนึ่ง เพราะไม่มีค่าถัดไปให้ข้าม ตั้ง lo = mid หรือ hi = mid ตรงๆ และไม่ใช้ while (lo < hi) แต่วนให้ครบ 100 รอบ

#include <bits/stdc++.h>
using namespace std;

int main()
{
    double x;
    cin >> x;
    double lo = 0, hi = max(1.0, x);
    for (int step = 0; step < 100; step++) { // (1)!
        double mid = (lo + hi) / 2;
        if (mid * mid >= x) {
            hi = mid;
        } else {
            lo = mid;
        }
    }
    cout << fixed << setprecision(9); // (2)!
    cout << "binary search: " << hi << '\n';
    cout << "sqrt:          " << sqrt(x) << '\n';
    return 0;
}
  1. วนครบ 100 รอบเสมอ ไม่ว่าช่วงจะแคบแค่ไหนแล้ว เหตุผลอยู่ในหัวข้อถัดไป
  2. fixed กับ setprecision(9) คือทศนิยม 9 ตำแหน่ง แบบเดียวกับ printf("%.9f") ในบทที่ 2 ตั้งครั้งเดียวมีผลกับทุกค่าที่พิมพ์ต่อจากนั้น

ระดับน้ำ

แท็งก์น้ำแบ่งเป็น N ช่องเรียงกัน ช่องละกว้าง 1 เมตร ผนังกั้นสูงมาก พื้นของแต่ละช่องสูงไม่เท่ากัน ช่องที่ i สูง hi เมตร ทุกช่องมีท่อเล็กๆ เชื่อมถึงกันใต้พื้น เทน้ำลงไป V ตารางเมตร (มองเป็นภาพตัดขวางสองมิติ และไม่คิดน้ำที่ค้างในท่อ) น้ำจะไหลถึงกันจนผิวน้ำของทุกช่องที่มีน้ำอยู่ระดับเดียวกัน ถามว่าผิวน้ำอยู่สูงเท่าไร ตอบทศนิยม 4 ตำแหน่ง

ถ้าผิวน้ำอยู่ที่ระดับ L ช่องที่พื้นต่ำกว่า L จะมีน้ำสูง L - hi ช่องที่พื้นสูงกว่าไม่มีน้ำ น้ำทั้งหมดใต้ระดับ L จึงคำนวณได้ในลูปเดียว คำถามใช่ไม่ใช่คือ "ใต้ระดับ L จุน้ำได้อย่างน้อย V ไหม" ยกระดับขึ้น ทุกช่องจุได้เท่าเดิมหรือมากขึ้น จึงเป็นทางเดียว และคำตอบคือ L ที่น้อยที่สุดที่ตอบใช่

#include <bits/stdc++.h>
using namespace std;

double water_below(const vector<int> &floor_height, double level)
{
    double total = 0;
    for (int i = 0; i < (int)floor_height.size(); i++) {
        if (level > floor_height[i]) {
            total += level - floor_height[i];
        }
    }
    return total;
}

int main()
{
    int n;
    double volume;
    cin >> n >> volume;
    vector<int> floor_height(n);
    int highest = 0;
    for (int i = 0; i < n; i++) {
        cin >> floor_height[i];
        highest = max(highest, floor_height[i]);
    }
    double lo = 0, hi = highest + volume; // (1)!
    for (int step = 0; step < 100; step++) {
        double mid = (lo + hi) / 2;
        if (water_below(floor_height, mid) >= volume) {
            hi = mid;
        } else {
            lo = mid;
        }
    }
    cout << fixed << setprecision(4) << hi << '\n';
    return 0;
}
  1. ที่ระดับ highest + volume ช่องไหนก็ตามมีน้ำสูงอย่างน้อย volume แล้ว จึงตอบใช่แน่นอน ส่วนระดับ 0 ไม่มีน้ำเลย

ตรวจด้วยมือ ระดับต้องอยู่ระหว่าง 3 ถึง 4 เพราะที่ระดับ 3 จุได้แค่ 0 + 2 + 0 + 2 + 0 = 4 ตารางเมตร ช่วงนี้มีน้ำสามช่องคือช่องที่พื้นสูง 3 1 1 จึงได้ 3L - 5 = 6 ได้ L = 11/3 = 3.6667 ตรงกัน ถ้าจะคิดด้วยสูตรแบบนี้ ต้องรู้ก่อนว่าระดับอยู่ช่วงไหน ซึ่งต้องเรียงความสูงแล้วไล่หา ส่วน binary search ไม่ต้องรู้อะไรเลยนอกจากวิธีคำนวณน้ำใต้ระดับหนึ่งๆ

ระวังชุดทดสอบที่ปัดเศษตรงครึ่ง

คำตอบที่ได้จาก binary search คลาดจากค่าจริงนิดเดียว ถ้าค่าจริงตกตรงกลางระหว่างสองค่าที่ปัดได้พอดี เช่น 2.00005 กับทศนิยม 4 ตำแหน่ง ความคลาดนิดเดียวนั้นอาจทำให้ปัดไปคนละทาง โจทย์ที่ดีจึงตรวจคำตอบแบบยอมให้คลาดได้เล็กน้อย หรือเลือกชุดทดสอบที่ไม่ตกตรงครึ่ง ตัวตรวจของเว็บนี้เทียบทุกตัวอักษร ชุดทดสอบในแบบฝึกหัดจึงกรองกรณีแบบนั้นออกแล้ว

เร็วขึ้นแค่ไหน

วิธีตรงไปตรงมาคือไล่ระดับขึ้นทีละ 0.0001 เมตรจาก 0 จนกว่าน้ำใต้ระดับจะถึง V จำนวนรอบเท่ากับคำตอบคูณหนึ่งหมื่น วัดบนเครื่องเดียวกับหน้า 19.1 แท็งก์ N = 1000 ช่อง พื้นสูงสุ่มตั้งแต่ 0 ถึง 100 เมตร เทน้ำ V = 50000 ผิวน้ำอยู่ที่ราว 99 เมตร วิธีไล่จึงต้องตรวจราวหนึ่งล้านครั้ง ครั้งละ 1000 ช่อง รันแต่ละแบบ 5 รอบ

สรุป

  • คำตอบเป็นทศนิยมก็ใช้ binary search ได้ ถ้าคำถามใช่ไม่ใช่เป็นทางเดียว
  • ตั้ง lo = mid หรือ hi = mid ไม่มีบวกลบหนึ่ง และวนจำนวนรอบที่กำหนด เช่น 100 รอบ
  • อย่าหยุดด้วย hi - lo > eps เพราะเมื่อค่าใหญ่ double แคบได้ไม่ถึง eps ลูปจะไม่จบ
  • เวลาทั้งหมดคือ 100 คูณเวลาตรวจหนึ่งครั้ง
  • พิมพ์คำตอบด้วย fixed กับ setprecision หรือ printf ตามจำนวนตำแหน่งที่โจทย์กำหนด
