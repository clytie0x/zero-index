29.3 spanning tree และ Kruskal

อำเภอหนึ่งมีหมู่บ้าน N แห่ง และมีแนวถนนที่สร้างได้ M แนว แต่ละแนวมีราคา งบมีจำกัด จึงอยากเลือกสร้างแค่บางเส้น ให้ทุกหมู่บ้านไปมาหากันได้ และราคารวมน้อยที่สุด โจทย์แบบนี้ซ่อนอยู่ในข้อสอบค่าย 2 และ TOI หลายรูปแบบ ทั้งวางท่อ เดินสายไฟ ต่อสายแลน หน้านี้นิยามสิ่งที่ต้องหาให้ชัด คือ minimum spanning tree แล้วหามันด้วย Kruskal ซึ่งเป็น greedy ที่ใช้ union-find จากหน้า 29.1 และพิสูจน์ว่าทำไม greedy นี้ถึงถูก ด้วย exchange argument จากบทที่ 23

spanning tree

ภาพนี้คือหมู่บ้าน 6 แห่ง กับแนวถนน 9 แนว ตัวเลขบนเส้นคือราคาเป็นล้านบาท กราฟที่เส้นเชื่อมมีตัวเลขกำกับแบบนี้เรียกว่า กราฟถ่วงน้ำหนัก (weighted graph) ตัวเลขเรียกว่า น้ำหนัก (weight) แบบเดียวกับกราฟในบทที่ 28

ถ้าเลือกถนนชุดหนึ่งแล้วทุกหมู่บ้านไปถึงกันได้ ชุดที่ถูกที่สุดจะไม่มีวงจรเลย เพราะถ้ามีวงจร ตัดเส้นใดก็ได้บนวงจรนั้นทิ้ง หมู่บ้านก็ยังไปถึงกันได้ผ่านส่วนที่เหลือของวงจร และราคาไม่เพิ่ม ชุดเส้นที่ต่อทุกโหนดถึงกันและไม่มีวงจร คือต้นไม้ที่ครอบคลุมทุกโหนด เรียกว่า spanning tree (ต้นไม้แผ่ทั่ว) และจากหน้า 29.2 ต้นไม้ที่มี N โหนดมีเส้นเชื่อม N - 1 เส้นพอดี กราฟนี้มีหลาย spanning tree เช่นต้นนี้

spanning tree ที่น้ำหนักรวมน้อยที่สุด เรียกว่า minimum spanning tree หรือ MST ของกราฟนี้คือต้นนี้ ราคา 19

ลองทุก spanning tree ไม่ไหว กราฟที่ทุกคู่โหนดมีเส้นเชื่อมกันมี spanning tree ถึง NN-2 ต้น (สูตรของ Cayley) N = 10 ก็มีร้อยล้านต้นแล้ว N = 20 มีราว 2.6 × 1023 ต้น ต้องมีวิธีที่ฉลาดกว่านั้น

Kruskal

Kruskal คิดแบบ greedy ตรงๆ เรียงเส้นเชื่อมจากเบาไปหนัก แล้วไล่หยิบทีละเส้น ถ้าเส้นนั้นไม่ทำให้เกิดวงจรกับเส้นที่หยิบไว้แล้ว ก็หยิบ ถ้าทำให้เกิดวงจรก็ข้าม คำถามว่า "ทำให้เกิดวงจรไหม" คือคำถามของหน้า 29.2 พอดี เส้น a-b ทำให้เกิดวงจรเมื่อ a กับ b อยู่กลุ่มเดียวกันแล้ว ซึ่ง unite ตอบได้พร้อมรวมให้ในตัว

#include <bits/stdc++.h>
using namespace std;

vector<int> parent, group_size;

int find_root(int x)
{
    if (parent[x] == x) {
        return x;
    }
    parent[x] = find_root(parent[x]);
    return parent[x];
}

bool unite(int a, int b)
{
    int root_a = find_root(a);
    int root_b = find_root(b);
    if (root_a == root_b) {
        return false;
    }
    if (group_size[root_a] < group_size[root_b]) {
        swap(root_a, root_b);
    }
    parent[root_b] = root_a;
    group_size[root_a] += group_size[root_b];
    return true;
}

int main()
{
    int n, m;
    cin >> n >> m;
    vector<tuple<int, int, int>> edges(m);
    for (int i = 0; i < m; i++) {
        int u, v, w;
        cin >> u >> v >> w;
        edges[i] = make_tuple(w, u, v); // (1)!
    }
    sort(edges.begin(), edges.end());
    parent.resize(n + 1);
    group_size.assign(n + 1, 1);
    for (int i = 1; i <= n; i++) {
        parent[i] = i;
    }
    long long total = 0; // (2)!
    int used = 0;
    for (auto [w, u, v] : edges) {
        if (unite(u, v)) { // (3)!
            total += w;
            used++;
            cout << "take " << u << "-" << v << " w=" << w << '\n';
        } else {
            cout << "skip " << u << "-" << v << " w=" << w << '\n';
        }
    }
    if (used == n - 1) { // (4)!
        cout << "total = " << total << '\n';
    } else {
        cout << "not connected" << '\n';
    }
    return 0;
}

  2. N เป็นแสนและน้ำหนักถึง 109 ผลรวมถึงราว 1014 เกิน int
  3. รวมได้ แปลว่าสองปลายอยู่คนละกลุ่ม เส้นนี้ไม่ทำให้เกิดวงจร หยิบได้
  4. หยิบได้ N - 1 เส้นแปลว่าต่อครบทุกหมู่บ้าน ถ้าได้น้อยกว่านั้น กราฟเดิมก็ไม่ได้ต่อถึงกันตั้งแต่แรก ไม่มี spanning tree

ทำไมหยิบเส้นที่ถูกที่สุดก่อนถึงถูก

บทที่ 23 สอนว่า greedy ต้องพิสูจน์เสมอ เพราะหลายโจทย์ที่ greedy ดูน่าจะถูกกลับผิด เครื่องมือที่ใช้คือ exchange argument สมมติว่ามีคำตอบที่ดีที่สุดที่ต่างจากของ greedy แล้วแสดงว่าแลกชิ้นส่วนให้เหมือน greedy ได้ทีละชิ้นโดยไม่แย่ลง ใช้กับ Kruskal ได้แบบนี้

ให้ T เป็น MST ต้นหนึ่งที่เหมือนกับเส้นที่ Kruskal หยิบได้มากที่สุด ถ้าทุกเส้นที่ Kruskal หยิบอยู่ใน T ด้วย ก็จบ เพราะทั้งสองมี N - 1 เส้นเท่ากัน จึงเป็นต้นเดียวกัน ถ้าไม่ใช่ ให้ e เป็นเส้นแรกที่ Kruskal หยิบแต่ไม่อยู่ใน T สมมติ e ต่อโหนด a กับ b และตอนที่ Kruskal หยิบ e ให้ A คือกลุ่มของ a ซึ่งไม่มี b อยู่

  1. T เป็นต้นไม้ จึงมีทางเดินจาก a ไป b ในต้น T ทางนี้เริ่มใน A แต่จบนอก A จึงต้องมีเส้น f บนทางนี้ที่ข้ามจากในกลุ่ม A ออกไปนอกกลุ่ม
  2. f ไม่ใช่เส้นที่ Kruskal หยิบไว้แล้ว เพราะเส้นที่หยิบแล้วทุกเส้นอยู่ภายในกลุ่มใดกลุ่มหนึ่งเสมอ ไม่มีเส้นไหนข้ามกลุ่ม
  3. Kruskal ยังไม่เคยพิจารณา f เพราะกลุ่มมีแต่รวมกันใหญ่ขึ้น ถ้าตอนนี้ปลายทั้งสองของ f อยู่คนละกลุ่ม ตอนก่อนหน้าก็ต้องอยู่คนละกลุ่ม ถ้าเคยพิจารณา f ไปแล้ว Kruskal ต้องหยิบมันไว้ ขัดกับข้อ 2
  4. Kruskal พิจารณาจากเบาไปหนัก f ยังไม่ถูกพิจารณา จึงหนักไม่น้อยกว่า e

แลกกัน เอา f ออกจาก T แล้วใส่ e แทน เรียกต้นใหม่ว่า T' เอา f ออก ต้น T ขาดเป็นสองท่อน โดย a กับ b อยู่คนละท่อน เพราะ f อยู่บนทางเดียวที่เชื่อมสองตัวนี้ ใส่ e กลับเข้าไปก็ต่อสองท่อนนั้นเข้าหากัน T' จึงเป็น spanning tree ที่มีน้ำหนักรวมเท่ากับของ T ลบ f บวก e ซึ่งไม่มากกว่า T T' จึงเป็น MST ด้วย แต่เหมือนเส้นของ Kruskal มากกว่า T อีกหนึ่งเส้น ขัดกับที่เลือก T มาให้เหมือนมากที่สุดแล้ว กรณีนี้จึงเกิดไม่ได้ ต้นที่ Kruskal ได้คือ MST

ลองกับตัวเลขจริง ให้ T คือต้นราคา 33 เส้นแรกที่ Kruskal หยิบคือ 4-5 ซึ่งไม่อยู่ใน T ตอนนั้นกลุ่มของ 4 คือ {4} คนเดียว ทางจาก 4 ไป 5 ในต้น T คือ 4-6-5 เส้นที่ข้ามออกจาก {4} คือ 4-6 ราคา 8 แลก 4-6 ออก ใส่ 4-5 ราคา 1 ได้ต้นใหม่ราคา 33 - 8 + 1 = 26 ทำแบบนี้ไปเรื่อยๆ จะไปจบที่ 19 ถ้ามีหลายเส้นน้ำหนักเท่ากัน MST อาจมีได้หลายต้น Kruskal ได้ต้นใดต้นหนึ่ง แต่น้ำหนักรวมเท่ากันเสมอ

เร็วขึ้นแค่ไหน

วิธีตรงไปตรงมาของการถามว่า "a กับ b ไปถึงกันแล้วหรือยัง" ก่อนมี union-find คือ BFS บนเส้นที่หยิบไว้แล้ว ซึ่งเป็นป่าที่มีไม่เกิน N - 1 เส้น ทุกเส้นที่พิจารณาต้อง BFS หนึ่งรอบ O(N) รวมเป็น O(MN) วัดบนเครื่องเดียวกับหน้า 29.1 กราฟสุ่มที่ต่อถึงกันหมด น้ำหนักสุ่ม 1 ถึง 109 จับเวลาตั้งแต่เรียงจนได้คำตอบ รันแต่ละแบบ 5 รอบ

สรุป

  • spanning tree คือชุดเส้นที่ต่อทุกโหนดถึงกันโดยไม่มีวงจร มี N - 1 เส้นพอดี MST คือต้นที่น้ำหนักรวมน้อยที่สุด
  • Kruskal เรียงเส้นจากเบาไปหนัก หยิบเส้นที่สองปลายอยู่คนละกลุ่มของ union-find ข้ามเส้นที่อยู่กลุ่มเดียวกัน ใช้เวลา O(M log M)
  • พิสูจน์ด้วย exchange argument MST ที่ต่างจาก Kruskal แลกเส้น f บนทางระหว่างสองปลายของ e กับเส้น e ได้โดยไม่แพงขึ้น
  • หยิบได้ไม่ครบ N - 1 เส้นแปลว่ากราฟไม่ต่อถึงกัน
  • เก็บน้ำหนักไว้ช่องแรกของ tuple และเก็บผลรวมด้วย long long
