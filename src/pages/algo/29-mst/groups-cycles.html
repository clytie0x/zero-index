29.2 นับกลุ่ม ขนาดกลุ่ม และตรวจวงจร

union-find ในหน้า 29.1 ตอบได้แค่ว่าสองคนอยู่กลุ่มเดียวกันไหม แต่โจทย์มักถามต่อ เช่น ตอนนี้มีกี่กลุ่ม กลุ่มของคนนี้มีกี่คน กลุ่มที่ใหญ่ที่สุดมีกี่คน หรือถนนเส้นนี้ทำให้เกิดทางวนหรือเปล่า คำถามเหล่านี้ตอบได้ด้วยตัวแปรเพิ่มอีกสองสามตัว โดยไม่ทำให้ช้าลง หน้านี้รวมทุกอย่างไว้ในโปรแกรมเดียว แล้วเทียบกับการเดินกราฟด้วย BFS ของบทที่ 27 ว่าเมื่อไรควรใช้อะไร

โจทย์ถนนในหมู่บ้าน

หมู่บ้านมีบ้าน N หลัง ยังไม่มีถนนเลย เทศบาลสร้างถนนทีละเส้น ถนนวิ่งได้สองทาง ระหว่างนั้นมีคนถามเข้ามา คำสั่งมีสามแบบ

  • road a b สร้างถนนระหว่างบ้าน a กับบ้าน b แล้วพิมพ์ว่าตอนนี้บ้านแบ่งเป็นกี่กลุ่มที่ไปมาหากันได้ และกลุ่มใหญ่สุดมีกี่หลัง
  • same a b จากบ้าน a ขับรถไปบ้าน b ได้หรือยัง
  • size a จากบ้าน a ขับไปได้กี่หลัง นับบ้าน a ด้วย

กลุ่มของบ้านที่ไปมาหากันได้ก็คือส่วนเชื่อมต่อ (connected component) ของกราฟในบทที่ 27 ถนนหนึ่งเส้นคือคำสั่ง join ในหน้า 29.1 สิ่งที่ต้องเพิ่มมีสามอย่าง

  • จำนวนกลุ่ม เริ่มที่ N เพราะทุกหลังอยู่กลุ่มของตัวเอง ทุกครั้งที่ unite รวมสองกลุ่มที่ต่างกันได้จริง จำนวนกลุ่มลดลงหนึ่งพอดี ถ้าสองบ้านอยู่กลุ่มเดียวกันอยู่แล้ว จำนวนไม่เปลี่ยน unite จึงคืนค่า bool บอกว่ารวมจริงหรือไม่
  • ขนาดกลุ่ม มีอยู่แล้วใน group_size ที่ใช้ทำ union by size แต่ค่าที่ถูกต้องอยู่ที่หัวหน้าเท่านั้น ต้องอ่าน group_size[find_root(a)]
  • กลุ่มใหญ่สุด กลุ่มมีแต่รวมกันใหญ่ขึ้น ไม่เคยแยก ขนาดใหญ่สุดจึงเปลี่ยนได้เฉพาะตอนรวม เทียบกับขนาดของกลุ่มที่เพิ่งรวมเสร็จก็พอ
#include <bits/stdc++.h>
using namespace std;

vector<int> parent, group_size;
int groups, largest;

int find_root(int x)
{
    if (parent[x] == x) {
        return x;
    }
    parent[x] = find_root(parent[x]);
    return parent[x];
}

bool unite(int a, int b)
{
    int root_a = find_root(a);
    int root_b = find_root(b);
    if (root_a == root_b) {
        return false; // (1)!
    }
    if (group_size[root_a] < group_size[root_b]) {
        swap(root_a, root_b);
    }
    parent[root_b] = root_a;
    group_size[root_a] += group_size[root_b];
    groups--; // (2)!
    largest = max(largest, group_size[root_a]); // (3)!
    return true;
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, q;
    cin >> n >> q;
    parent.resize(n + 1);
    group_size.assign(n + 1, 1);
    for (int i = 1; i <= n; i++) {
        parent[i] = i;
    }
    groups = n;
    largest = 1;
    for (int k = 0; k < q; k++) {
        string op;
        cin >> op;
        if (op == "road") {
            int a, b;
            cin >> a >> b;
            unite(a, b);
            cout << "groups=" << groups << " largest=" << largest << '\n';
        } else if (op == "same") {
            int a, b;
            cin >> a >> b;
            cout << (find_root(a) == find_root(b) ? "yes" : "no") << '\n';
        } else {
            int a;
            cin >> a;
            cout << "size of " << a << " = " << group_size[find_root(a)] << '\n'; // (4)!
        }
    }
    return 0;
}

  2. มาถึงบรรทัดนี้ได้ก็ต่อเมื่อเป็นสองกลุ่มที่ต่างกันจริง จำนวนกลุ่มจึงลดลงหนึ่ง
  3. group_size[root_a] คือขนาดของกลุ่มที่เพิ่งรวมเสร็จ

BFS หรือ union-find

บทที่ 27 หาส่วนเชื่อมต่อด้วย BFS หรือ DFS ได้ใน O(N + M) เมื่อ N คือจำนวนโหนด และ M คือจำนวนเส้นเชื่อม ซึ่งบทที่ 27 เขียนเป็น O(V + E) สองวิธีนี้ไม่ได้แข่งกัน แต่เหมาะกับโจทย์คนละแบบ

โจทย์ใช้เหตุผล
ให้เส้นเชื่อมมาครบก่อน แล้วค่อยถามBFS หรือ DFS ครั้งเดียว หรือ union-find ก็ได้เดินกราฟครั้งเดียวติดป้ายหมายเลขกลุ่มให้ทุกโหนด แล้วตอบทุกคำถามจากป้าย
เส้นเชื่อมทยอยเพิ่ม มีคำถามแทรกunion-findกลุ่มเปลี่ยนหลังทุกเส้น ป้ายที่ติดไว้ใช้ไม่ได้อีก BFS ต้องเดินใหม่ทุกครั้งที่ถาม
ต้องรู้เส้นทาง ระยะทาง หรือลำดับที่ไปถึงBFS หรือ DFSunion-find รู้แค่ว่าอยู่กลุ่มเดียวกันไหม ไม่รู้ว่าไปทางไหน
มีการลบเส้นเชื่อมไม่ใช่ union-find แบบนี้union-find รวมได้อย่างเดียว แยกกลุ่มกลับไม่ได้

ถ้ามีแต่การลบ ให้ทำย้อนหลัง

โจทย์บางข้อเริ่มจากกราฟเต็ม แล้วลบเส้นทีละเส้นพร้อมถามแทรก ถ้าอ่านคำสั่งทั้งหมดได้ก่อนตอบ ให้เริ่มจากกราฟหลังลบครบแล้ว แล้วทำคำสั่งจากท้ายมาหัว การลบเมื่อมองย้อนหลังก็คือการเพิ่ม ซึ่ง union-find ทำได้ เก็บคำตอบไว้ในอาเรย์แล้วพิมพ์กลับด้านตอนจบ วิธีนี้เรียกว่าทำแบบ ออฟไลน์ (offline) คือรู้คำถามทั้งหมดก่อนตอบ

ตรวจวงจรในกราฟไม่มีทิศทาง

เส้นที่ปิดวงจร

ใส่เส้นเชื่อมทีละเส้น ถ้าเส้น a-b ต่อสองโหนดที่อยู่กลุ่มเดียวกันอยู่แล้ว แปลว่ามีทางจาก a ไป b อยู่ก่อน เส้นใหม่จึงปิดทางนั้นเป็นวงจร ถ้าไม่มีเส้นไหนเป็นแบบนี้เลย กราฟก็ไม่มีวงจร

unite ที่คืน false คือสัญญาณนี้พอดี โปรแกรมนี้อ่านเส้นเชื่อมของกราฟ แล้วบอกว่าเส้นไหนปิดวงจรบ้าง

#include <bits/stdc++.h>
using namespace std;

vector<int> parent, group_size;

int find_root(int x)
{
    if (parent[x] == x) {
        return x;
    }
    parent[x] = find_root(parent[x]);
    return parent[x];
}

bool unite(int a, int b)
{
    int root_a = find_root(a);
    int root_b = find_root(b);
    if (root_a == root_b) {
        return false;
    }
    if (group_size[root_a] < group_size[root_b]) {
        swap(root_a, root_b);
    }
    parent[root_b] = root_a;
    group_size[root_a] += group_size[root_b];
    return true;
}

int main()
{
    int n, m;
    cin >> n >> m;
    parent.resize(n + 1);
    group_size.assign(n + 1, 1);
    for (int i = 1; i <= n; i++) {
        parent[i] = i;
    }
    int extra = 0;
    for (int i = 1; i <= m; i++) {
        int a, b;
        cin >> a >> b;
        if (!unite(a, b)) { // (1)!
            cout << "edge " << i << " (" << a << ", " << b << ") closes a cycle" << '\n';
            extra++;
        }
    }
    cout << "edges that close a cycle: " << extra << '\n';
    cout << (extra == 0 ? "forest" : "has a cycle") << '\n';
    return 0;
}

จำนวนเส้นที่ปิดวงจร = M - (N - C)

เมื่อ C คือจำนวนกลุ่มตอนจบ ทุกเส้นที่รวมกลุ่มได้ทำให้จำนวนกลุ่มลดลงหนึ่ง จาก N ลงมาเหลือ C จึงมีเส้นที่รวมได้ N - C เส้นพอดี เส้นที่เหลือปิดวงจรทั้งหมด ตัวอย่างนี้ได้ 7 - (6 - 1) = 2 ผลที่ตามมาอีกข้อคือ กราฟที่มี N โหนด ต่อถึงกันหมด และไม่มีวงจร ต้องมีเส้นเชื่อม N - 1 เส้นพอดี ซึ่งก็คือต้นไม้ที่จะใช้ในหน้า 29.3

เส้นที่ต่อโหนดกับตัวเอง เช่น 2 2 และเส้นที่ซ้ำกับเส้นเดิม ถูกนับเป็นเส้นที่ปิดวงจรด้วย ลองป้อน 3 3 ตามด้วยเส้น 1 2 2 2 1 2 โปรแกรมรายงานเส้นที่ 2 และ 3 ถ้าโจทย์นับแบบอื่นต้องอ่านให้ดี

ใช้กับกราฟมีทิศทางไม่ได้

union-find ไม่รู้จักทิศของเส้น กราฟมีทิศทาง 1→2 1→3 2→3 ไม่มีวงจร เพราะไม่มีทางเดินตามลูกศรกลับมาที่เดิมได้ แต่โปรแกรมนี้รายงานว่าเส้นที่ 3 ปิดวงจร การหาวงจรในกราฟมีทิศทางใช้ Kahn หรือ DFS แบบสามสถานะในหน้า 30.1 ซึ่งมาพร้อมกับ topological sort

  • จำนวนกลุ่มเริ่มที่ N และลดลงหนึ่งทุกครั้งที่ unite รวมได้จริง ให้ unite คืน bool
  • ขนาดกลุ่มอ่านที่หัวหน้า group_size[find_root(x)] กลุ่มใหญ่สุดอัปเดตตอนรวม

  • union-find แยกกลุ่มไม่ได้ ถ้าโจทย์มีแต่การลบและรู้คำสั่งทั้งหมดก่อน ให้ทำย้อนหลัง
