29.4 Prim และเลือกใช้ให้เหมาะกับกราฟ

Kruskal มองทั้งกราฟพร้อมกัน หยิบเส้นที่ถูกที่สุดจากตรงไหนก็ได้ ต้นไม้จึงงอกออกมาเป็นหลายกลุ่มแล้วค่อยรวมกัน อีกวิธีที่ได้ MST เหมือนกันคือ Prim ซึ่งปลูกต้นไม้ต้นเดียวจากโหนดเริ่มต้น แล้วขยายออกไปทีละโหนด โครงของโปรแกรมเหมือน Dijkstra ในบทที่ 28 แทบทุกบรรทัด หน้านี้เขียน Prim สองแบบ แบบใช้ priority_queue สำหรับกราฟทั่วไป กับแบบ O(N2) ที่ไม่ต้องเก็บเส้นเชื่อมเลย สำหรับกราฟที่ทุกคู่โหนดมีเส้นถึงกัน แล้วสรุปว่าโจทย์แบบไหนควรใช้วิธีไหน

ปลูกต้นไม้จากโหนดเดียว

เริ่มจากต้นไม้ที่มีโหนด 1 โหนดเดียว ทุกรอบดูเส้นเชื่อมทุกเส้นที่ปลายข้างหนึ่งอยู่ในต้นไม้ และอีกข้างอยู่นอกต้นไม้ หยิบเส้นที่ถูกที่สุด แล้วดึงโหนดปลายนอกเข้ามาในต้นไม้ ทำจนครบ N โหนด

เหตุผลที่ถูกเหมือนกับ Kruskal ในหน้า 29.3 ข้อพิสูจน์ตรงนั้นใช้แค่ว่า เส้น e ที่หยิบ ถูกที่สุดในบรรดาเส้นที่ข้ามออกจากกลุ่ม A ของโหนดหนึ่ง ถ้า MST ต้นไหนไม่มี e ก็แลกเส้น f ที่ข้ามออกจาก A บนทางเดียวกันกับ e ได้โดยไม่แพงขึ้น สำหรับ Prim กลุ่ม A ก็คือต้นไม้ที่ปลูกมาถึงตอนนั้น และ e คือเส้นที่ถูกที่สุดที่ข้ามออกจาก A ตามนิยามของ Prim พอดี

Prim ด้วย priority_queue

ถ้าทุกรอบไล่ดูเส้นเชื่อมทั้งหมดเพื่อหาเส้นที่ถูกที่สุดที่ข้ามออกไป ต้องไล่ M เส้น N รอบ เป็น O(NM) ใช้ priority_queue แบบหยิบค่าน้อยก่อนจากหน้า 20.5 ช่วยจำเส้นที่ข้ามออกไปแทน ทุกครั้งที่ดึงโหนด v เข้าต้นไม้ ใส่เส้นทุกเส้นของ v ที่ไปหาโหนดนอกต้นไม้ลงไปเป็นคู่ (น้ำหนัก, โหนดปลาย) แล้วหยิบคู่ที่น้ำหนักน้อยที่สุดออกมา

เส้นที่ใส่ไว้นานแล้วอาจไม่ข้ามออกจากต้นไม้แล้ว เพราะโหนดปลายถูกดึงเข้ามาทางเส้นอื่นไปก่อน แทนที่จะตามไปลบออกจากคิว ให้ตรวจตอนหยิบออกมา ถ้าโหนดปลายอยู่ในต้นไม้แล้วก็ทิ้งไป แบบเดียวกับการข้ามระยะเก่าใน Dijkstra

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int n, m;
    cin >> n >> m;
    vector<vector<pair<int, int>>> adj(n + 1);
    for (int i = 0; i < m; i++) {
        int u, v, w;
        cin >> u >> v >> w;
        adj[u].push_back(make_pair(v, w));
        adj[v].push_back(make_pair(u, w));
    }
    vector<bool> in_tree(n + 1, false);
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq; // (1)!
    pq.push(make_pair(0, 1)); // (2)!
    long long total = 0;
    int added = 0;
    while (!pq.empty()) {
        auto [w, v] = pq.top();
        pq.pop();
        if (in_tree[v]) { // (3)!
            cout << "skip " << v << " w=" << w << '\n';
            continue;
        }
        in_tree[v] = true;
        total += w;
        added++;
        cout << "add " << v << " w=" << w << '\n';
        for (auto [to, cost] : adj[v]) {
            if (!in_tree[to]) { // (4)!
                pq.push(make_pair(cost, to));
            }
        }
    }
    if (added == n) {
        cout << "total = " << total << '\n';
    } else {
        cout << "not connected" << '\n';
    }
    return 0;
}
  1. เก็บคู่ (น้ำหนัก, โหนด) ให้ greater เรียงตามน้ำหนักก่อน และหยิบตัวน้อยที่สุดออกมาก่อน
  2. โหนดเริ่มต้นเข้าต้นไม้ด้วยน้ำหนัก 0 ไม่มีเส้นไหนพามันเข้ามา จะเริ่มที่โหนดไหนก็ได้ MST มีน้ำหนักเท่ากัน
  3. คู่เก่าที่โหนดปลายเข้าต้นไม้ไปแล้ว ทิ้งไป ไม่นับน้ำหนัก
  4. ไม่ต้องใส่เส้นที่ไปหาโหนดในต้นไม้ เพราะไม่ได้ข้ามออกไปไหน

ใช้กราฟเดียวกับหน้า 29.3 ได้ 19 เท่ากัน แต่ลำดับที่หยิบต่างจาก Kruskal Prim ดึง 3 เข้ามาก่อนด้วยเส้น 1-3 ราคา 3 แม้เส้น 4-5 ราคา 1 จะถูกกว่า เพราะตอนนั้น 4 กับ 5 ยังไม่ได้ติดกับต้นไม้ที่ปลูกจาก 1 ส่วนบรรทัด skip 2 w=4 คือคู่ของเส้น 1-2 ที่ใส่ไว้ตอนดึง 1 เข้ามา แต่ 2 เข้าต้นไม้ไปก่อนแล้วผ่าน 3 ด้วยราคา 2

ทุกเส้นถูกใส่ลงคิวไม่เกินหนึ่งครั้ง คือตอนดึงปลายข้างแรกเข้าต้นไม้ พอดึงปลายข้างที่สองเข้ามา ปลายข้างแรกอยู่ในต้นไม้แล้วจึงไม่ใส่ซ้ำ คิวจึงมีของไม่เกิน M + 1 ชิ้น ใส่และหยิบแต่ละครั้งเป็น O(log M) ทั้งหมดเป็น O(M log M) เท่ากับ Kruskal ถ้า added ไม่ถึง N แปลว่ากราฟไม่ต่อถึงกัน

กราฟที่ทุกคู่มีเส้นถึงกัน

โรงเรียนมีอาคาร N หลัง อยากเดินสายแลนให้ทุกอาคารต่อถึงกัน สายเดินตามทางเดินที่เป็นตารางตั้งฉาก สายระหว่างอาคารที่พิกัด (x1, y1) กับ (x2, y2) ยาว |x1 - x2| + |y1 - y2| เมตร ทุกคู่อาคารเดินสายถึงกันได้ กราฟจึงมีเส้นเชื่อม N(N - 1)/2 เส้น N = 5000 ก็มีถึง 12.5 ล้านเส้น Kruskal ต้องสร้างและเรียงทุกเส้น Prim ด้วย priority_queue ก็ต้องใส่เส้นลงคิวเป็นล้านๆ ครั้ง

กราฟแบบนี้เรียกว่า กราฟหนาแน่น (dense graph) M ใกล้ N2 ใช้ Prim อีกแบบที่ไม่เก็บเส้นเชื่อมเลย ให้ best[v] คือราคาของเส้นที่ถูกที่สุดที่ต่อโหนด v นอกต้นไม้เข้ากับต้นไม้ ทุกรอบไล่หาโหนดนอกต้นไม้ที่ best น้อยที่สุด O(N) ดึงเข้ามา แล้วคำนวณระยะจากโหนดใหม่ไปหาโหนดนอกต้นไม้ทุกตัว ปรับ best ให้น้อยลงถ้าทำได้ อีก O(N) ทำ N รอบ รวมเป็น O(N2) ซึ่งเท่ากับจำนวนเส้นเชื่อม จึงเร็วที่สุดเท่าที่เป็นไปได้สำหรับกราฟที่ต้องดูทุกเส้น

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int n;
    cin >> n;
    vector<int> x(n), y(n);
    for (int i = 0; i < n; i++) {
        cin >> x[i] >> y[i];
    }
    vector<long long> best(n, LLONG_MAX); // (1)!
    vector<int> link(n, -1); // (2)!
    vector<bool> in_tree(n, false);
    best[0] = 0;
    long long total = 0;
    for (int step = 0; step < n; step++) {
        int v = -1;
        for (int i = 0; i < n; i++) { // (3)!
            if (!in_tree[i] && (v == -1 || best[i] < best[v])) {
                v = i;
            }
        }
        in_tree[v] = true;
        total += best[v];
        if (link[v] != -1) {
            cout << "cable " << link[v] << "-" << v << " length " << best[v] << '\n';
        }
        for (int i = 0; i < n; i++) {
            long long d = abs(x[v] - x[i]) + abs(y[v] - y[i]);
            if (!in_tree[i] && d < best[i]) { // (4)!
                best[i] = d;
                link[i] = v;
            }
        }
    }
    cout << "total = " << total << '\n';
    return 0;
}
  1. ยังไม่มีเส้นไหนต่อถึงต้นไม้ ให้เป็นค่าที่มากที่สุดของ long long ซึ่งไม่มีระยะจริงไหนเกิน
  2. จำว่าค่า best[v] ได้มาจากเส้นที่ต่อกับโหนดไหน เพื่อพิมพ์ว่าเดินสายจากไหนไปไหน ถ้าโจทย์ถามแค่ผลรวม ไม่ต้องมีอาเรย์นี้
  3. หาโหนดนอกต้นไม้ที่ต่อเข้าต้นไม้ได้ถูกที่สุด ลูปนี้แทน priority_queue
  4. เส้นจาก v ไปหา i ถูกกว่าเส้นที่ดีที่สุดเดิมของ i ก็จำเส้นใหม่

อาคารใช้หมายเลข 0 ถึง N - 1 ตามลำดับที่อ่าน อาคาร 0 ที่ (0, 0) เข้าต้นไม้ก่อน อาคาร 2 ที่ (3, 0) ห่าง 3 เมตรเป็นตัวแรกที่ดึงเข้ามา จากนั้นอาคาร 1 ที่ (2, 2) ห่างจากอาคาร 2 ไป 1 + 2 = 3 เมตร ส่วนอาคาร 4 กับ 3 ต่อจากอาคาร 1 ทั้งคู่ เพราะใกล้อาคาร 1 กว่าอาคารอื่นที่อยู่ในต้นไม้แล้ว สายรวม 15 เมตร โปรแกรมเก็บแค่อาเรย์ไม่กี่ตัวขนาด N ไม่ได้เก็บเส้นเชื่อมสักเส้น

เร็วขึ้นแค่ไหน

วัดบนเครื่องเดียวกับหน้า 29.1 รันแต่ละแบบ 5 รอบ จับเวลาตั้งแต่อ่านข้อมูลเสร็จจนได้คำตอบ ตารางแรกเป็นกราฟสุ่มชุดเดียวกับหน้า 29.3 ซึ่ง M ประมาณ 2 ถึง 5 เท่าของ N เรียกว่า กราฟเบาบาง (sparse graph) วิธีตรงไปตรงมาคือ Prim ที่ทุกรอบไล่ดูเส้นทั้ง M เส้นหาเส้นที่ถูกที่สุดที่ข้ามออกจากต้นไม้

สรุป

  • Prim ปลูกต้นไม้จากโหนดเดียว หยิบเส้นที่ถูกที่สุดที่ข้ามออกจากต้นไม้ทุกรอบ ถูกต้องด้วยเหตุผลเดียวกับ Kruskal
  • Prim ด้วย priority_queue แบบ greater ใส่คู่ (น้ำหนัก, โหนด) ตอนหยิบให้ทิ้งคู่ที่โหนดอยู่ในต้นไม้แล้ว ใช้เวลา O(M log M)
  • กราฟที่ทุกคู่มีเส้นถึงกัน ใช้ Prim แบบ O(N2) กับอาเรย์ best ไม่ต้องเก็บเส้นเชื่อม
  • กราฟเบาบางใช้ Kruskal หรือ Prim ก็ได้ กราฟหนาแน่นใช้ Prim แบบ O(N2) โจทย์ที่ต้องดูลำดับการรวมกลุ่มใช้ Kruskal
  • ลืม greater ได้ maximum spanning tree และลืมทิ้งคู่เก่าได้โหนดซ้ำ
