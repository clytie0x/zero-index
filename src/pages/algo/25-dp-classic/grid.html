25.4 DP บนตาราง

โจทย์ที่ให้ตารางมาแล้วให้เดินจากมุมซ้ายบนไปมุมขวาล่าง โดยเดินได้แค่ ขวา กับ ลง เป็น DP ที่เห็นภาพง่ายที่สุด เพราะตารางของโจทย์ก็คือตาราง DP เลย ช่องหนึ่งมาได้จากช่องบนกับช่องซ้ายเท่านั้น เติมทีละแถวจากบนลงล่างก็ได้คำตอบ หน้านี้มีสามแบบที่เจอในค่าย 2 และ TOI บ่อย นับทางเดิน นับทางเดินเมื่อมีสิ่งกีดขวาง และหาเส้นทางที่เก็บของได้มากที่สุดพร้อมพิมพ์เส้นทาง ปิดท้ายด้วยการลดตารางเหลือแถวเดียว

นับทางเดิน

สวนสาธารณะแบ่งเป็นตาราง 4 แถว 5 คอลัมน์ น้องเริ่มที่มุมซ้ายบน จะไปมุมขวาล่าง เดินทีละช่องไปทางขวาหรือลงล่างเท่านั้น มีกี่เส้นทาง

  • นิยามสถานะ ways[r][c] คือจำนวนเส้นทางจากมุมซ้ายบนมาถึงช่องแถว r คอลัมน์ c
  • หาความสัมพันธ์ ก้าวสุดท้ายที่มาถึงช่องนี้ ถ้าไม่มาจากช่องบนก็มาจากช่องซ้าย สองกลุ่มนี้ไม่ซ้ำกัน จึงบวกกัน ways[r - 1][c] + ways[r][c - 1]
  • กรณีฐาน ช่องในแถวบนสุดมาได้ทางเดียวคือเดินขวาล้วน ช่องในคอลัมน์ซ้ายสุดเดินลงล้วน จึงเป็น 1
  • ลำดับการเติมและคำตอบสุดท้าย ช่องบนกับช่องซ้ายต้องเสร็จก่อน เติมทีละแถวจากบนลงล่าง ในแถวซ้ายไปขวา คำตอบอยู่ช่องขวาล่าง
#include <bits/stdc++.h>
using namespace std;

int main()
{
    int rows, cols;
    cin >> rows >> cols;
    vector<vector<long long>> ways(rows, vector<long long>(cols, 0));
    for (int r = 0; r < rows; r++) {
        for (int c = 0; c < cols; c++) {
            if (r == 0 || c == 0) {
                ways[r][c] = 1;
            } else {
                ways[r][c] = ways[r - 1][c] + ways[r][c - 1];
            }
        }
    }
    for (int r = 0; r < rows; r++) {
        for (int c = 0; c < cols; c++) {
            cout << setw(4) << ways[r][c];
        }
        cout << '\n';
    }
    cout << "paths = " << ways[rows - 1][cols - 1] << '\n';
    return 0;
}

มีสิ่งกีดขวาง

สวนมีบ่อน้ำบางช่อง แทนด้วย # ช่องที่เดินได้แทนด้วย . ห้ามเดินผ่านบ่อน้ำ ตารางใหญ่ได้ถึง 1000 × 1000 จำนวนเส้นทางจึงใหญ่เกิน long long ไปมาก โจทย์ให้ตอบเป็นเศษจากการหารด้วย 1000000007 แบบ mod ในบทที่ 24

#include <bits/stdc++.h>
using namespace std;

const long long MOD = 1000000007;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int rows, cols;
    cin >> rows >> cols;
    vector<string> grid(rows);
    for (int r = 0; r < rows; r++) {
        cin >> grid[r];
    }
    vector<vector<long long>> ways(rows, vector<long long>(cols, 0));
    for (int r = 0; r < rows; r++) {
        for (int c = 0; c < cols; c++) {
            if (grid[r][c] == '#') { // (1)!
                ways[r][c] = 0;
            } else if (r == 0 && c == 0) {
                ways[r][c] = 1;
            } else {
                long long from_up = (r > 0 ? ways[r - 1][c] : 0); // (2)!
                long long from_left = (c > 0 ? ways[r][c - 1] : 0);
                ways[r][c] = (from_up + from_left) % MOD; // (3)!
            }
        }
    }
    cout << ways[rows - 1][cols - 1] << '\n';
    return 0;
}
  1. ช่องบ่อน้ำไม่มีทางมาถึง เป็น 0 ช่องถัดไปที่บวกค่านี้ก็เหมือนไม่มีทางมาจากทางนั้น
  2. ช่องในแถวบนสุดไม่มีช่องบน ให้ถือว่ามาจากทางนั้นได้ 0 ทาง คอลัมน์ซ้ายสุดก็เช่นกัน ไม่ต้องเขียนกรณีของขอบแยก
  3. ค่าทั้งสองน้อยกว่า 1000000007 ผลบวกจึงไม่เกิน long long แน่นอน แล้ว mod ทุกช่อง

เก็บของให้ได้มากที่สุด

สวนมะม่วงแบ่งเป็นตาราง แต่ละช่องมีมะม่วงหล่นอยู่จำนวนหนึ่ง น้องเดินจากมุมซ้ายบนไปมุมขวาล่าง ขวาหรือลงเท่านั้น เก็บมะม่วงทุกช่องที่เดินผ่าน รวมช่องแรกกับช่องสุดท้ายด้วย จะเก็บได้มากที่สุดกี่ลูก และต้องเดินทางไหน

เปลี่ยนจากนับเป็นหาค่ามากสุด best[r][c] คือมะม่วงมากที่สุดที่เก็บได้ตั้งแต่ต้นจนถึงช่องนี้ มาจากช่องบนหรือช่องซ้าย เลือกทางที่ได้มากกว่าแล้วบวกมะม่วงของช่องนี้ ช่องในแถวบนสุดกับคอลัมน์ซ้ายสุดมาได้ทางเดียว ต้องแยกกรณี เพราะคราวนี้ไม่มีค่าไหนแทน "ไม่มีทาง" ได้ง่ายๆ แบบ 0 ของการนับ ซึ่งจะเห็นผลในส่วนลองทำให้พัง

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int rows, cols;
    cin >> rows >> cols;
    vector<vector<int>> mango(rows, vector<int>(cols));
    for (int r = 0; r < rows; r++) {
        for (int c = 0; c < cols; c++) {
            cin >> mango[r][c];
        }
    }
    vector<vector<long long>> best(rows, vector<long long>(cols, 0));
    for (int r = 0; r < rows; r++) {
        for (int c = 0; c < cols; c++) {
            if (r == 0 && c == 0) {
                best[r][c] = mango[r][c];
            } else if (r == 0) {
                best[r][c] = best[r][c - 1] + mango[r][c];
            } else if (c == 0) {
                best[r][c] = best[r - 1][c] + mango[r][c];
            } else {
                best[r][c] = max(best[r - 1][c], best[r][c - 1]) + mango[r][c];
            }
        }
    }
    string path;
    int r = rows - 1, c = cols - 1;
    while (r > 0 || c > 0) {
        if (c == 0 || (r > 0 && best[r - 1][c] >= best[r][c - 1])) { // (1)!
            path += 'D';
            r--;
        } else {
            path += 'R';
            c--;
        }
    }
    reverse(path.begin(), path.end());
    cout << "most mangoes = " << best[rows - 1][cols - 1] << '\n';
    cout << "path = " << path << '\n';
    return 0;
}
  1. ย้อนรอยจากช่องขวาล่าง ถ้าค่ามาจากช่องบน ก้าวสุดท้ายคือลง (D) ถ้ามาจากช่องซ้ายคือขวา (R) คอลัมน์ 0 ต้องมาจากบนเสมอ ถ้าสองทางเท่ากันเลือกบน ได้ตัวอักษรจากท้ายไปหน้า จึงกลับลำดับก่อนพิมพ์

เหลือแถวเดียว

ทั้งสามโปรแกรม แถว r ใช้แค่แถว r กับแถว r - 1 แบบ LCS ในหน้า 25.3 แต่คราวนี้ช่องซ้ายเป็นค่าใหม่ ช่องบนเป็นค่าเก่า ที่ตำแหน่งเดียวกัน จึงใช้แถวเดียวได้เลย ก่อนเขียนทับ ways[c] ยังเป็นค่าของแถวบน ส่วน ways[c - 1] ถูกเขียนทับเป็นค่าของแถวนี้ไปแล้ว ตรงกับที่สูตรต้องการพอดี วนซ้ายไปขวาตามปกติ

#include <bits/stdc++.h>
using namespace std;

const long long MOD = 1000000007;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int rows, cols;
    cin >> rows >> cols;
    vector<long long> ways(cols, 0);
    ways[0] = 1;
    for (int r = 0; r < rows; r++) {
        string line;
        cin >> line;
        for (int c = 0; c < cols; c++) {
            if (line[c] == '#') {
                ways[c] = 0;
            } else if (c > 0) {
                ways[c] = (ways[c] + ways[c - 1]) % MOD; // (1)!
            }
        }
    }
    cout << ways[cols - 1] << '\n';
    return 0;
}
  1. ทางซ้ายของเครื่องหมาย = คือแถวนี้ ways[c] ทางขวาคือแถวบน ways[c - 1] คือแถวนี้ ช่อง 0 ไม่มีช่องซ้าย จึงเก็บค่าแถวบนไว้ตามเดิม

แบบที่ 2 ขอบศูนย์กับค่าติดลบ

บางคนชอบทำตารางเริ่มที่ 1 แล้วปล่อยแถว 0 กับคอลัมน์ 0 เป็นศูนย์ไว้เป็นขอบ จะได้ไม่ต้องเขียนกรณีของขอบ วิธีนี้ใช้ได้กับการนับ และกับค่าที่ไม่ติดลบ แต่ลองกับเกมเดินบนกระดานที่บางช่องเป็นกับดัก เดินผ่านแล้วเสียแต้ม

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int rows, cols;
    cin >> rows >> cols;
    vector<vector<long long>> best(rows + 1, vector<long long>(cols + 1, 0));
    for (int r = 1; r <= rows; r++) {
        for (int c = 1; c <= cols; c++) {
            int score;
            cin >> score;
            if (r == 1 && c == 1) {
                best[r][c] = score;
            } else {
                best[r][c] = max(best[r - 1][c], best[r][c - 1]) + score;
            }
        }
    }
    cout << "best score = " << best[rows][cols] << '\n';
    return 0;
}

  • นับทางเดินคือบวกสองช่อง ช่องสิ่งกีดขวางเป็น 0 ตอบเป็น mod ถ้าเลขใหญ่
  • ค่ามากสุดคือเลือกช่องที่ดีกว่าแล้วบวกค่าของช่อง ย้อนรอยจากปลายทางเพื่อพิมพ์เส้นทาง
  • เก็บแถวเดียวได้ วนซ้ายไปขวาตามปกติ เพราะช่องบนคือค่าเก่าที่ตำแหน่งเดียวกัน
  • ระวังขอบ ช่องหลังสิ่งกีดขวางบนขอบไปไม่ถึง และขอบศูนย์ใช้ไม่ได้เมื่อค่าติดลบได้
