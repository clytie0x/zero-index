28.2 Dijkstra ด้วย priority_queue

Dijkstra แบบอาเรย์เสียเวลาเกือบทั้งหมดไปกับการไล่หาจุดที่ใกล้ที่สุด รอบละ V ช่อง โจทย์ค่าย 2 และ TOI ส่วนใหญ่ให้ V ถึงแสน และ E ถึงสองแสน ซึ่ง V2 คือ 1010 ไม่ทันแน่นอน งาน "ใส่ค่าเข้าไปเรื่อยๆ แล้วหยิบค่าน้อยที่สุดออก" คืองานของ priority_queue ในหน้า 20.5 พอดี หน้านี้เปลี่ยนการไล่หาเป็น priority_queue ซึ่งเป็นแบบที่ใช้จริงในการแข่งขัน แล้วดูกับดักสามข้อที่ทำให้คนเสียคะแนนบ่อยที่สุด

เก็บคู่ (ระยะ, จุด) ไว้ในคิว

ใส่คู่ (dist[v], v) ลงใน priority_queue แบบน้อยสุดก่อน ทุกครั้งที่หยิบออก จะได้จุดที่ระยะน้อยที่สุดในเวลา log ของขนาดคิว แทนการไล่ V ช่อง ปัญหาคือเมื่อผ่อนแล้ว dist[v] ลดลง ค่าของ v ที่อยู่ในคิวอยู่แล้วต้องเปลี่ยนตาม แต่ priority_queue แก้ค่าของที่อยู่ข้างในไม่ได้ ทำได้แค่ใส่กับหยิบ

ทางออกคือ ไม่ต้องแก้ ใส่คู่ใหม่ (ระยะใหม่, v) ลงไปอีกตัว ปล่อยคู่เก่าค้างไว้ในคิว คู่ใหม่ระยะน้อยกว่าจึงออกมาก่อนเสมอ พอคู่เก่าออกมาทีหลัง ก็ดูออกได้ทันทีว่ามันเก่า เพราะระยะในคู่มากกว่า dist[v] ปัจจุบัน ให้ข้ามไปเฉยๆ วิธีปล่อยของเก่าค้างแล้วค่อยทิ้งตอนหยิบออกมาแบบนี้เรียกว่า lazy deletion

#include <bits/stdc++.h>
using namespace std;

const long long INF = 1e18;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    cin >> n >> m;
    vector<vector<pair<int, int>>> adj(n + 1);
    for (int i = 0; i < m; i++) {
        int u, v, w;
        cin >> u >> v >> w;
        adj[u].push_back({v, w});
        adj[v].push_back({u, w});
    }
    vector<long long> dist(n + 1, INF);
    priority_queue<pair<long long, int>, vector<pair<long long, int>>, greater<pair<long long, int>>> pq; // (1)!
    dist[1] = 0;
    pq.push({0, 1});
    while (!pq.empty()) {
        auto [d, u] = pq.top();
        pq.pop();
        if (d > dist[u]) { // (2)!
            cout << "skip " << u << " d=" << d << '\n';
            continue;
        }
        cout << "take " << u << " dist=" << d << '\n';
        for (auto [v, w] : adj[u]) {
            if (d + w < dist[v]) {
                dist[v] = d + w;
                pq.push({dist[v], v}); // (3)!
            }
        }
    }
    for (int u = 1; u <= n; u++) {
        cout << (dist[u] == INF ? -1 : dist[u]) << (u == n ? '\n' : ' ');
    }
    return 0;
}
  1. priority_queue แบบน้อยสุดก่อนตามหน้า 20.5 เก็บ pair ช่องแรกคือระยะ จึงเรียงตามระยะ ระยะเป็น long long ช่องแรกของ pair จึงต้องเป็น long long ด้วย
  2. คู่นี้เก่าแล้ว เคยมีคู่ที่ระยะน้อยกว่าของจุดเดียวกันออกไปก่อน ข้ามไปเลย โปรแกรมจริงไม่ต้องพิมพ์ ตัวอย่างนี้พิมพ์ไว้ให้เห็น
  3. ทุกครั้งที่ระยะลดลงให้ใส่คู่ใหม่ จุดเดียวกันจึงอยู่ในคิวได้หลายคู่

ลำดับ take ตรงกับแบบอาเรย์ในหน้า 28.1 ทุกบรรทัด บรรทัด skip คือคู่เก่าที่ค้างอยู่ เช่น ตลาดเคยถูกใส่เป็น (4, 2) ตอนผ่อนจากบ้าน แล้วถูกใส่อีกครั้งเป็น (3, 2) ตอนผ่อนจากวัด คู่ (3, 2) ออกก่อนและถูกใช้ คู่ (4, 2) ออกทีหลังจึงถูกข้าม

ใน for ที่ผ่อนถนน ใช้ d + w ไม่ใช่ dist[u] + w สองค่านี้เท่ากันเสมอหลังผ่านบรรทัดที่ข้ามคู่เก่าแล้ว เขียนแบบไหนก็ได้ แต่หน้า 28.3 จะแสดงว่าถ้าลืมบรรทัดที่ข้าม สองแบบนี้ให้ผลต่างกัน

ความซับซ้อน

นับว่าแต่ละอย่างเกิดขึ้นกี่ครั้ง

  • คู่ใหม่ถูกใส่ทุกครั้งที่ผ่อนสำเร็จ ผ่อนได้ไม่เกินจำนวนครั้งที่ดูถนน คือราว 2E ครั้งเมื่อถนนเป็นสองทาง คิวจึงมีของไม่เกินราว 2E ตัว ใส่และหยิบตัวละ O(log E)
  • แต่ละจุดผ่านบรรทัดที่ข้ามไปได้ครั้งเดียว จึงดูถนนของตัวเองครั้งเดียว รวม O(V + E)
  • E ไม่เกิน V2 ทำให้ log E ไม่เกิน 2 log V จึงเขียนเป็น log V ได้

รวมเป็น O((V + E) log V) ที่ V เป็นแสนและ E เป็นสองแสน ได้ราว 3 × 105 × 17 ประมาณห้าล้านขั้น สบายมาก

เร็วขึ้นแค่ไหน

วัดบนเครื่องเดียวกับหน้า 28.1 คอมไพล์แบบ -O2 กราฟสุ่มด้วย mt19937 น้ำหนัก 1 ถึง 109 ทุกจุดไปถึงกันได้ จับเวลาเฉพาะช่วง Dijkstra แต่ละช่องเป็นค่ากลางของ 5 รอบ

เลือกแบบไหน

โจทย์ที่ E ใกล้ V2 เช่นให้ระยะของทุกคู่มาเป็นตาราง V × V และ V ไม่เกินหลักพัน แบบอาเรย์ทั้งง่ายกว่าและไม่มีทางช้ากว่า V2 นอกนั้นใช้แบบ priority_queue หน้า 28.5 มีตารางเลือกวิธีครบทุกแบบ

ลองทำให้พัง

แบบที่ 1 ไม่ข้ามคู่เก่า

บรรทัด if (d > dist[u]) continue; ดูเหมือนไม่จำเป็น เพราะถ้าลบทิ้ง คู่เก่าที่ออกมาจะผ่อนถนนด้วยระยะที่มากกว่าเดิม ซึ่งไม่มีทางทำให้อะไรดีขึ้น คำตอบจึงยังถูก แต่ทุกครั้งที่คู่เก่าออกมา มันไล่ดูถนน ทุกเส้น ของจุดนั้นซ้ำ ถ้าจุดที่มีถนนเยอะถูกใส่ลงคิวหลายครั้ง งานจะระเบิด

โปรแกรมนี้สร้างกราฟที่ตั้งใจให้เกิดแบบนั้น add คือ lambda แบบหน้า 16.3 แต่แทนที่จะส่งให้ sort เราเก็บมันไว้ในตัวแปรชื่อ add ชนิดของ lambda ไม่มีชื่อให้เขียน จึงประกาศด้วย auto จากหน้า 15.5 จากนั้นเรียก add(1, 2, 5) ได้เหมือนฟังก์ชันธรรมดา ส่วน [&] ทำให้ข้างใน lambda ใช้ตัวแปร adj ของ main ได้ และแก้ที่ตัวจริง ไม่ใช่สำเนา ถ้าเขียนเป็นฟังก์ชันธรรมดานอก main ต้องส่ง adj เข้าไปด้วยทุกครั้ง จุด 1 มีถนนไปจุดป้อน k จุด ทุกจุดป้อนมีถนนไปจุดศูนย์กลางจุดเดียว น้ำหนักตั้งไว้ให้ทุกครั้งที่จุดป้อนถูกเลือก ระยะของศูนย์กลางลดลง ศูนย์กลางจึงถูกใส่ลงคิว k ครั้ง และศูนย์กลางมีถนนไปจุดปลายอีก L จุด ตัวแปร skip_old เลือกว่าจะข้ามคู่เก่าหรือไม่ โปรแกรมนับด้วยว่าดูถนนไปกี่ครั้ง

#include <bits/stdc++.h>
using namespace std;

const long long INF = 1e18;

int main()
{
    int feeders, leaves, skip_old;
    cin >> feeders >> leaves >> skip_old;
    int n = feeders + leaves + 2;
    int hub = feeders + 2;
    vector<vector<pair<int, int>>> adj(n + 1);
    auto add = [&](int u, int v, int w) {
        adj[u].push_back({v, w});
        adj[v].push_back({u, w});
    };
    for (int i = 1; i <= feeders; i++) {
        add(1, 1 + i, i);
        add(1 + i, hub, 3 * feeders - 2 * i);
    }
    for (int j = 1; j <= leaves; j++) {
        add(hub, hub + j, 1);
    }
    clock_t start = clock();
    vector<long long> dist(n + 1, INF);
    priority_queue<pair<long long, int>, vector<pair<long long, int>>, greater<pair<long long, int>>> pq;
    dist[1] = 0;
    pq.push({0, 1});
    long long scanned = 0;
    while (!pq.empty()) {
        auto [d, u] = pq.top();
        pq.pop();
        if (skip_old && d > dist[u]) {
            continue;
        }
        for (auto [v, w] : adj[u]) {
            scanned++;
            if (d + w < dist[v]) {
                dist[v] = d + w;
                pq.push({dist[v], v});
            }
        }
    }
    clock_t end = clock();
    long long total = 0;
    for (int u = 1; u <= n; u++) {
        total += dist[u];
    }
    cout << "n = " << n << " edges scanned = " << scanned << " sum = " << total << '\n';
    cout << "time = " << fixed << setprecision(3) << (double)(end - start) / CLOCKS_PER_SEC << " seconds\n";
    return 0;
}

แบบที่ 2 ลืม greater

ประกาศคิวเป็น priority_queue<pair<long long, int>> pq; เพราะลืมว่าค่าเริ่มต้นคือมากสุดก่อน ตามที่หน้า 20.5 เตือนไว้ ผลแปลกกว่าที่คิด คือกับแผนที่เมืองยัง ได้คำตอบถูก 0 3 1 8 10 13 แต่ take ออกมา 11 ครั้งแทน 6 ครั้ง เพราะหยิบจุดที่ไกลที่สุดก่อน แล้วระยะของมันลดลงทีหลังจึงต้องทำซ้ำ บรรทัดที่ข้ามคู่เก่ากันไม่ให้ผิด แต่กันไม่ให้ทำซ้ำไม่ได้ วัดบนเครื่องเดียวกัน แถวที่เป็นตารางคือกราฟที่แต่ละจุดเป็นช่องในตาราง ต่อกับช่องบนล่างซ้ายขวา น้ำหนักสุ่ม 1 ถึง 9 ซึ่งหน้า 28.3 จะเขียนให้ดู

แบบที่ 3 เก็บระยะใน int

โจทย์บอกว่าน้ำหนักไม่เกิน 109 ซึ่งอยู่ใน int หลายคนจึงเก็บระยะเป็น int ด้วย ลองถนนสามเส้นต่อกัน เส้นละ 109

#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    cin >> n >> m;
    vector<vector<pair<int, int>>> adj(n + 1);
    for (int i = 0; i < m; i++) {
        int u, v, w;
        cin >> u >> v >> w;
        adj[u].push_back({v, w});
        adj[v].push_back({u, w});
    }
    vector<int> dist(n + 1, INT_MAX);
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;
    dist[1] = 0;
    pq.push({0, 1});
    while (!pq.empty()) {
        auto [d, u] = pq.top();
        pq.pop();
        if (d > dist[u]) {
            continue;
        }
        for (auto [v, w] : adj[u]) {
            if (d + w < dist[v]) {
                dist[v] = d + w;
                pq.push({dist[v], v});
            }
        }
    }
    for (int u = 1; u <= n; u++) {
        cout << dist[u] << (u == n ? '\n' : ' ');
    }
    return 0;
}

  • หยิบออกมาแล้วถ้า d > dist[u] คือคู่เก่า ต้องข้าม ไม่งั้นอาจช้าระเบิด
  • ความซับซ้อน O((V + E) log V) ใช้กับกราฟเส้นน้อยที่ V เป็นแสนได้ กราฟที่ E ใกล้ V2 แบบอาเรย์ไม่แพ้
  • ต้องใส่ greater ลืมแล้วคำตอบยังถูกแต่ช้า ตรวจด้วยตัวอย่างเล็กไม่เจอ
  • ระยะรวมเกิน int ได้ง่าย ใช้ long long ทั้งใน dist และในคิว
