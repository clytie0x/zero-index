27.1 กราฟและการเก็บในโปรแกรม

โจทย์ที่พูดถึงหมู่บ้านกับถนน เพื่อนกับเพื่อน สถานีรถไฟฟ้า ทางเดินในเขาวงกต หรือการเทน้ำจากขวดหนึ่งไปอีกขวด ดูต่างกันมาก แต่ถ้าตัดเรื่องราวทิ้ง ทุกข้อเหลือแค่ของหลายชิ้นกับคู่ของที่เชื่อมกัน สิ่งนี้เรียกว่า กราฟ (graph) เรียนวิธีคิดกับกราฟครั้งเดียว ใช้ได้กับโจทย์ทุกแบบที่ว่ามา หน้านี้ทำความรู้จักกราฟ แล้วเก็บกราฟลงในโปรแกรมสองแบบ คือ adjacency matrix กับ adjacency list พร้อมวัดว่าเลือกผิดแบบแล้วเสียอะไร

กราฟคืออะไร

กราฟประกอบด้วยสองอย่าง จุดยอด (vertex) คือของแต่ละชิ้น เช่น บ้านหนึ่งหลัง และ เส้นเชื่อม (edge) คือคู่ของจุดยอดที่เชื่อมกันโดยตรง เช่น ถนนหนึ่งเส้นระหว่างบ้านสองหลัง โจทย์มักเรียกจุดยอดว่า node และใช้ตัวอักษร V แทนจำนวนจุดยอด E แทนจำนวนเส้นเชื่อม หรือใช้ N กับ M ตามข้อมูลเข้า

ภาพนี้คือหมู่บ้านเล็กๆ ที่มีบ้าน 9 หลัง หมายเลข 1 ถึง 9 และถนน 9 เส้น ตัวอย่างในหน้า 27.1 ถึง 27.3 ใช้กราฟนี้ ยกเว้นตัวอย่างกราฟมีทิศและกราฟมีน้ำหนักซึ่งมีภาพของตัวเอง

ตำแหน่งที่วาดไม่มีความหมาย ย้ายบ้าน 9 ไปไว้มุมซ้ายก็ยังเป็นกราฟเดิม สิ่งเดียวที่สำคัญคือใครเชื่อมกับใคร คำที่โจทย์ใช้บ่อยมีดังนี้

คำความหมายในภาพ
เพื่อนบ้าน (neighbor, adjacent)จุดยอดที่มีเส้นเชื่อมกันโดยตรงเพื่อนบ้านของ 2 คือ 1 3 5
ดีกรี (degree)จำนวนเส้นเชื่อมที่ออกจากจุดยอดนั้นดีกรีของ 2 คือ 3 ดีกรีของ 9 คือ 1
เส้นทาง (path)ลำดับจุดยอดที่ติดกันทีละคู่ ความยาวคือจำนวนเส้นเชื่อมที่ผ่าน4 5 6 3 ยาว 3
วงจร (cycle)เส้นทางที่วนกลับมาจุดเริ่ม โดยไม่ใช้เส้นเชื่อมซ้ำ1 2 5 4 1
ส่วนเชื่อมต่อ (connected component)กลุ่มของจุดยอดที่เดินไปหากันได้ทั้งหมดมีสองกลุ่ม คือ 1 ถึง 6 กับ 7 ถึง 9

ต้นไม้ในบทที่ 26 ก็คือกราฟแบบหนึ่ง เป็นกราฟที่ทุกจุดเดินถึงกันหมด และไม่มีวงจรเลย ต้นไม้ N จุดจึงมีเส้นเชื่อม N - 1 เส้นพอดี กลุ่มบ้าน 7 8 9 ในภาพเป็นต้นไม้ ส่วนกลุ่มบ้าน 1 ถึง 6 ไม่ใช่ เพราะมีวงจร ความต่างข้อนี้คือเหตุผลที่บทนี้ต้องมีของใหม่ที่ต้นไม้ไม่ต้องใช้ คือการจำว่าจุดไหนเคยไปแล้ว เพราะในกราฟทั่วไป เดินไปเรื่อยๆ อาจวนกลับมาที่เดิม

กราฟสามแบบที่เจอบ่อย

กราฟไม่มีทิศ (undirected graph) คือแบบในภาพข้างบน ถนนจาก 1 ไป 2 ใช้ขับจาก 2 กลับมา 1 ได้ด้วย ความเป็นเพื่อนก็เป็นแบบนี้

กราฟมีทิศ (directed graph) เส้นเชื่อมมีลูกศร ไปได้ทางเดียว เช่น ถนนเดินรถทางเดียว การติดตามในโซเชียลที่ A ติดตาม B โดย B ไม่ต้องติดตาม A ภาพนี้ไปจาก 1 ถึง 4 ได้ แต่จาก 4 ไปไหนไม่ได้เลย

กราฟมีน้ำหนัก (weighted graph) เส้นเชื่อมมีตัวเลขกำกับ เช่น ความยาวถนนเป็นกิโลเมตร หรือค่าผ่านทาง ในภาพนี้ทางที่ผ่านถนนน้อยเส้นที่สุดจาก 1 ไป 4 มีสองทาง ยาวเท่ากันคือสองเส้น แต่ทาง 1 2 4 ยาว 9 กิโลเมตร สั้นกว่าทาง 1 3 4 ที่ยาว 12 กิโลเมตร

บทนี้ทำงานกับกราฟที่ ไม่มีน้ำหนัก ทุกเส้นนับเป็นหนึ่งก้าวเท่ากัน ยกเว้นหัวข้อเสริม 0-1 BFS ในหน้า 27.5 ที่น้ำหนักเป็นได้แค่ 0 กับ 1 การหาทางที่สั้นที่สุดเมื่อน้ำหนักเป็นตัวเลขอะไรก็ได้ ต้องใช้วิธีของบทที่ 28

adjacency matrix

วิธีแรกคือตาราง N × N ช่องแถว u คอลัมน์ v เป็น true ถ้า u กับ v เชื่อมกัน เรียกว่า adjacency matrix (เมทริกซ์ประชิด) ใช้อาเรย์สองมิติจากบทที่ 7 ได้ตรงๆ

#include <bits/stdc++.h>
using namespace std;

bool road[101][101]; // (1)!

int main()
{
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        road[u][v] = true;
        road[v][u] = true; // (2)!
    }
    for (int u = 1; u <= n; u++) {
        for (int v = 1; v <= n; v++) {
            cout << road[u][v];
        }
        cout << '\n';
    }
    return 0;
}
  1. ประกาศนอก main ทุกช่องเริ่มเป็น false ให้เอง และไม่ติดขีดจำกัดราว 2 MB ของตัวแปรใน main ตามหน้า 5.5 จองไว้ 101 แถวเพราะใช้ช่อง 1 ถึง 100 ไม่ใช้ช่อง 0
  2. ถนนวิ่งได้สองทาง จึงต้องเขียนทั้งช่อง [u][v] และ [v][u] ถ้าเป็นกราฟมีทิศ เขียนแค่ช่องแรก

adjacency list

วิธีที่สองเก็บเฉพาะเพื่อนบ้านที่มีจริง แต่ละจุดยอดมี vector ของตัวเองหนึ่งตัว ใส่หมายเลขเพื่อนบ้านลงไป เรียกว่า adjacency list (รายการประชิด) เป็นรูปแบบเดียวกับรายชื่อสมาชิกชุมนุมในหน้า 16.2 ที่แต่ละแถวยาวไม่เท่ากัน

#include <bits/stdc++.h>
using namespace std;

const int MAXN = 100001;
vector<int> adj[MAXN]; // (1)!

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v); // (2)!
        adj[v].push_back(u);
    }
    for (int u = 1; u <= n; u++) {
        cout << u << " (degree " << adj[u].size() << "):"; // (3)!
        for (int v : adj[u]) {
            cout << ' ' << v;
        }
        cout << '\n';
    }
    return 0;
}
  1. อาเรย์ของ vector แต่ละช่องเป็น vector ว่างหนึ่งตัว adj[u] คือรายชื่อเพื่อนบ้านของ u ตัว vector ว่างกินที่ช่องละ 24 byte ข้อมูลจริงอยู่นอก stack จึงประกาศแสนช่องได้ไม่มีปัญหา
  2. ถนนหนึ่งเส้นเพิ่มเพื่อนบ้านให้ทั้งสองฝั่ง เส้นเชื่อม M เส้นจึงเก็บรวม 2M ตัว ถ้าเป็นกราฟมีทิศ เขียนแค่บรรทัดแรก
  3. ดีกรีคือขนาดของรายการ ได้ทันทีด้วย size()

กราฟมีทิศ

เก็บเฉพาะทิศที่ไปได้ ตัวอย่างนี้ใช้กราฟถนนทางเดียวในภาพก่อนหน้า และนับด้วยว่ามีลูกศรกี่อันชี้เข้าแต่ละจุด เรียกว่า in-degree

#include <bits/stdc++.h>
using namespace std;

const int MAXN = 100001;
vector<int> adj[MAXN];
int in_degree[MAXN];

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int from, to;
        cin >> from >> to;
        adj[from].push_back(to);
        in_degree[to]++;
    }
    for (int u = 1; u <= n; u++) {
        cout << u << " in " << in_degree[u] << " out:";
        for (int v : adj[u]) {
            cout << ' ' << v;
        }
        cout << '\n';
    }
    return 0;
}

กราฟมีน้ำหนัก

แต่ละช่องในรายการเก็บเป็น pair ของเพื่อนบ้านกับน้ำหนัก แล้วแตกออกด้วย structured binding ตามหน้า 16.4

#include <bits/stdc++.h>
using namespace std;

const int MAXN = 100001;
vector<pair<int, int>> adj[MAXN];

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int u, v, km;
        cin >> u >> v >> km;
        adj[u].push_back({v, km});
        adj[v].push_back({u, km});
    }
    for (int u = 1; u <= n; u++) {
        cout << u << ':';
        for (auto [v, km] : adj[u]) {
            cout << ' ' << v << '(' << km << " km)";
        }
        cout << '\n';
    }
    return 0;
}

เรื่องหน่วยความจำตัดสินได้เร็วกว่า โจทย์ค่าย 2 และ TOI ให้ N ถึงแสนหรือสองแสนเป็นปกติ adjacency matrix แบบ bool ของ N = 100000 คือ 1010 byte หรือ 10 GB ขณะที่โจทย์ส่วนใหญ่จำกัดหน่วยความจำไว้ราว 256 MB ส่วนโปรแกรมที่ใช้ adjacency list ในหัวข้อถัดไป กับ N = 100000 และ M = 200000 ใช้หน่วยความจำสูงสุดทั้งโปรแกรมแค่ 13.8 MB หลักเลือกจึงง่าย

  • N ไม่เกินสองสามพัน และอยากถามว่า u กับ v เชื่อมกันไหมบ่อยๆ ใช้ adjacency matrix ได้
  • นอกนั้นใช้ adjacency list เสมอ

เร็วต่างกันแค่ไหน

สถานีรถเมล์ N สถานีเชื่อมกันด้วยถนน M เส้น สถานีที่ i มีคนรอ wi คน หาสถานีที่สถานีข้างเคียงทั้งหมดรวมกันมีคนรอมากที่สุด ถ้าเท่ากันเลือกหมายเลขน้อย งานนี้คือไล่เพื่อนบ้านของทุกสถานีครบหนึ่งรอบพอดี

#include <bits/stdc++.h>
using namespace std;

const int MAXN = 200001;
vector<int> adj[MAXN];
int waiting[MAXN];

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    cin >> n >> m;
    for (int u = 1; u <= n; u++) {
        cin >> waiting[u];
    }
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    int best = 1;
    long long best_sum = -1;
    for (int u = 1; u <= n; u++) {
        long long sum = 0;
        for (int v : adj[u]) {
            sum += waiting[v];
        }
        if (sum > best_sum) {
            best_sum = sum;
            best = u;
        }
    }
    cout << best << ' ' << best_sum << '\n';
    return 0;
}

    }
}

#include <bits/stdc++.h>
using namespace std;

const int MAXN = 100001;
vector<int> adj[MAXN];

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
    }
    for (int u = 1; u <= n; u++) {
        cout << u << " (degree " << adj[u].size() << "):";
        for (int v : adj[u]) {
            cout << ' ' << v;
        }
        cout << '\n';
    }
    return 0;
}

  • กราฟคือจุดยอดกับเส้นเชื่อม มีแบบไม่มีทิศ มีทิศ และมีน้ำหนัก ต้นไม้คือกราฟที่ถึงกันหมดและไม่มีวงจร
  • adjacency matrix ตอบว่า u กับ v เชื่อมกันไหมใน O(1) แต่ใช้ที่ V2 ใช้ได้เมื่อ N ไม่เกินสองสามพัน
  • adjacency list vector<int> adj[MAXN] ใช้ที่ V + 2E และไล่เพื่อนบ้านของทุกจุดได้ใน O(V + E) เป็นแบบที่ใช้เกือบทุกข้อ
  • กราฟไม่มีทิศต้องใส่ทั้ง adj[u].push_back(v) และ adj[v].push_back(u) กราฟมีน้ำหนักเก็บเป็น pair
