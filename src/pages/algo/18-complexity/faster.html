18.5 ลดความซับซ้อนและสองตัวชี้

สี่หน้าที่ผ่านมาสอนให้ดูว่าวิธีไหนช้า หน้านี้ลงมือทำให้เร็วขึ้น โจทย์เดียวกันจะถูกแก้สามรอบ จาก O(N3) เป็น O(N2) แล้วเป็น O(N) โดยแต่ละรอบตัดงานที่ทำซ้ำทิ้ง แล้ววัดเวลาจริงให้เห็นว่าต่างกันแค่ไหน ปิดท้ายด้วยเทคนิค สองตัวชี้ (two pointers) ซึ่งเปลี่ยนการไล่ทุกช่วง O(N2) ให้เหลือ O(N) ในโจทย์อีกกลุ่มหนึ่ง

โจทย์ ช่วงที่กำไรมากที่สุด

ร้านน้ำปั่นจดกำไรของแต่ละวันไว้ N วัน วันที่ขาดทุนจดเป็นค่าติดลบ เจ้าของร้านอยากรู้ว่าถ้าเลือกช่วงวันที่ติดกันได้หนึ่งช่วง ยาวกี่วันก็ได้แต่อย่างน้อยหนึ่งวัน กำไรรวมสูงสุดเป็นเท่าไร ต่างจากแบบฝึกหัดในบทที่ 5 ที่ความยาวช่วงถูกกำหนดมาแล้ว ข้อนี้ต้องลองทุกความยาวด้วย

ถ้ากำไรเป็น -2 6 -1 4 -8 3 2 -5 ช่วง 6 -1 4 ได้ 9 มากที่สุด สังเกตว่าช่วงที่ดีที่สุดยอมรวมวันที่ขาดทุน 1 ไว้ด้วย เพราะวันถัดไปได้กลับมา 4

วิธีที่ 1 ลองทุกช่วง บวกใหม่ทุกครั้ง

#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<int> profit(n);
    for (int i = 0; i < n; i++) {
        cin >> profit[i];
    }
    long long best = profit[0];
    for (int l = 0; l < n; l++) {
        for (int r = l; r < n; r++) {
            long long total = 0;
            for (int k = l; k <= r; k++) {
                total += profit[k];
            }
            best = max(best, total);
        }
    }
    cout << best << '\n';
    return 0;
}

วิธีที่ 3 ใช้ผลรวมสะสม

ให้ P[i] คือผลรวมของ i วันแรกแบบหน้า 5.4 โดย P[0] = 0 ผลรวมของวันที่ l ถึง r (นับช่องจาก 0) คือ P[r + 1] − P[l] ถ้าตรึง r ไว้ อยากให้ผลลบนี้มากที่สุด ก็ต้องเลือก P[l] ที่ น้อยที่สุด ในบรรดา P[0] ถึง P[r] ไม่ต้องลองทุก l เลย แค่จำค่าน้อยสุดที่เคยเห็นมาไว้ในตัวแปรเดียว แล้วอัปเดตไปพร้อมกับ r

#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<int> profit(n);
    for (int i = 0; i < n; i++) {
        cin >> profit[i];
    }
    long long best = profit[0];
    long long prefix = 0, lowest = 0; // (1)!
    for (int r = 0; r < n; r++) {
        prefix += profit[r]; // (2)!
        best = max(best, prefix - lowest); // (3)!
        lowest = min(lowest, prefix); // (4)!
    }
    cout << best << '\n';
    return 0;
}
  1. prefix คือ P ของวันปัจจุบัน lowest คือ P ที่น้อยที่สุดที่เคยเห็น เริ่มที่ P[0] = 0
  2. ตอนนี้ prefix เป็น P[r + 1]
  3. ช่วงที่ดีที่สุดที่จบวันที่ r มีผลรวม P[r + 1] ลบ P ที่น้อยที่สุดก่อนหน้า
  4. อัปเดตทีหลังบรรทัดก่อนหน้า เพื่อไม่ให้ลบตัวเองจนได้ช่วงที่ว่างเปล่า

ทุกครั้งที่ลดความซับซ้อน ให้ถามสองข้อ

มีอะไรที่คำนวณซ้ำแล้วซ้ำอีกไหม ถ้ามี เก็บผลไว้ใช้ต่อ (วิธีที่ 2) และในลูปที่ลองทุกค่า เราต้องการแค่ค่าที่ดีที่สุดหรือเปล่า ถ้าใช่ จำค่าที่ดีที่สุดไว้ในตัวแปรเดียว ไม่ต้องลองใหม่ทุกครั้ง (วิธีที่ 3)

สองตัวชี้

อีกโจทย์หนึ่ง ร้านขนม N ร้านเรียงกันบนถนน ร้านที่ i ขายขนมชิ้นละ ci บาท ทุกราคาเป็นจำนวนเต็มบวก น้องมีเงิน S บาท อยากเดินซื้อร้านละหนึ่งชิ้นจากร้านที่อยู่ติดกันหนึ่งช่วง ซื้อได้มากที่สุดกี่ร้าน หรือก็คือหาช่วงที่ยาวที่สุดที่ผลรวมไม่เกิน S

วิธีตรงไปตรงมาคือลองทุกจุดเริ่ม l แล้วขยาย r ไปจนเงินไม่พอ

#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    long long budget;
    cin >> n >> budget;
    vector<int> cost(n);
    for (int i = 0; i < n; i++) {
        cin >> cost[i];
    }
    int best = 0;
    for (int l = 0; l < n; l++) {
        long long total = 0;
        for (int r = l; r < n; r++) {
            total += cost[r];
            if (total > budget) {
                break;
            }
            best = max(best, r - l + 1);
        }
    }
    cout << best << '\n';
    return 0;
}

มี break ก็จริง แต่ถ้า S ใหญ่จนช่วงยาวเป็นครึ่งหนึ่งของถนน แต่ละ l ต้องขยายไปราว N/2 ร้าน กรณีแย่ที่สุดยังเป็น O(N2)

ข้อสังเกตที่ทำให้เร็วขึ้นคือ ราคาเป็นบวกทุกตัว ถ้าช่วง l ถึง r ใช้เงินไม่เกิน S ช่วงที่เริ่มหลัง l แล้วจบที่ r ก็ใช้เงินน้อยกว่านั้นอีก จึงไม่เกิน S แน่นอน แปลว่าเมื่อ r ขยับไปทางขวา จุดเริ่มที่ใช้ได้ซึ่งอยู่ซ้ายสุดไม่มีวันถอยกลับไปทางซ้าย เราจึงใช้สองตัวชี้ l กับ r ที่เดินไปทางขวาทางเดียวกันทั้งคู่ ขยาย r ทีละร้าน ถ้าเงินเกินก็เลื่อน l ตามมาจนเงินพอ

#include <bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    long long budget;
    cin >> n >> budget;
    vector<int> cost(n);
    for (int i = 0; i < n; i++) {
        cin >> cost[i];
    }
    int best = 0;
    long long total = 0;
    int l = 0;
    for (int r = 0; r < n; r++) {
        total += cost[r];
        while (total > budget) {
            total -= cost[l];
            l++;
        }
        best = max(best, r - l + 1);
    }
    cout << best << '\n';
    return 0;
}

มี while ซ้อนอยู่ใน for ดูเหมือน O(N2) แต่ให้นับว่า l++ ทำได้กี่ครั้ง l เริ่มที่ 0 เพิ่มทีละหนึ่ง ไม่เคยลดลง และไม่เกิน N ทั้งโปรแกรมจึงเลื่อน l รวมกันไม่เกิน N ครั้ง ส่วน r เลื่อน N ครั้ง งานรวมไม่เกิน 2N เป็น O(N) การนับแบบนี้คือดูว่าตัวแปรเดินไปได้ไกลแค่ไหนตลอดทั้งโปรแกรม แทนที่จะคูณรอบของลูปนอกกับลูปใน

วัดแบบเดียวกับตารางข้างบน ราคาสุ่ม 1 ถึง 100 บาท S = 25N ทำให้ช่วงที่ดีที่สุดยาวราวครึ่งหนึ่งของ N

สรุป

  • ลดความซับซ้อนด้วยการตัดงานที่ทำซ้ำ เก็บผลรวมไว้ต่อยอดลด O(N3) เป็น O(N2) จำค่าที่ดีที่สุดไว้ลด O(N2) เป็น O(N)
  • ช่วงกำไรสูงสุดใช้ผลรวมสะสมลบค่าน้อยสุดที่เคยเห็น ทำได้ใน O(N) N สิบล้านใช้ 0.010 วินาทีบนเครื่องที่เขียนเอกสารนี้
  • สองตัวชี้ใช้กับโจทย์หาช่วงที่ ถ้าช่วงใช้ได้ ช่วงข้างในก็ใช้ได้ ตัวชี้ทั้งสองเดินไปทางเดียว รวมไม่เกิน 2N ก้าว
  • นับความซับซ้อนของ while ในสองตัวชี้ ให้นับว่าตัวแปรเดินได้ไกลแค่ไหนทั้งโปรแกรม ไม่ใช่คูณรอบลูป
  • ค่าติดลบทำให้สองตัวชี้แบบผลรวมผิด ตรวจขอบเขตของค่าก่อนใช้
