22.3 นับคู่สลับด้วย merge sort

หน้า 13.3 เห็นแล้วว่าจำนวนครั้งที่ bubble sort สลับ เท่ากับจำนวน คู่ที่ผิดลำดับ คือคู่ i < j ที่ ai > aj คู่แบบนี้เรียกว่า inversion (คู่สลับ) โจทย์ค่าย 2 และ TOI ใช้มันบ่อยในหลายเรื่องราว เช่น ต้องสลับคนที่ยืนติดกันกี่ครั้งแถวจึงจะเรียง หรือกรรมการสองคนจัดอันดับผู้เข้าแข่งขันขัดกันกี่คู่ หน้า 18.2 นับด้วยลูปซ้อนซึ่งเป็น O(N2) แต่โจทย์มักให้ N ถึงสองแสน หน้านี้นับคู่สลับไปพร้อมกับ merge sort ได้ใน O(N log N) โดยเพิ่มโค้ดแค่บรรทัดเดียว

เพิ่มบรรทัดเดียวใน merge sort

#include <bits/stdc++.h>
using namespace std;

long long count_inversions(vector<int> &a, vector<int> &temp, int lo, int hi)
{
    if (hi - lo <= 1) {
        return 0;
    }
    int mid = (lo + hi) / 2;
    long long count = count_inversions(a, temp, lo, mid) + count_inversions(a, temp, mid, hi); // (1)!
    long long cross = 0;
    int i = lo, j = mid, k = lo;
    while (i < mid && j < hi) {
        if (a[i] <= a[j]) { // (2)!
            temp[k] = a[i];
            i++;
        } else {
            temp[k] = a[j];
            j++;
            cross += mid - i; // (3)!
        }
        k++;
    }
    while (i < mid) {
        temp[k] = a[i];
        i++;
        k++;
    }
    while (j < hi) {
        temp[k] = a[j];
        j++;
        k++;
    }
    for (int t = lo; t < hi; t++) {
        a[t] = temp[t];
    }
    cout << "[" << lo << ", " << mid << ") + [" << mid << ", " << hi << ") cross = " << cross << '\n';
    return count + cross;
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<int> height(n);
    for (int i = 0; i < n; i++) {
        cin >> height[i];
    }
    vector<int> temp(n);
    long long swaps = count_inversions(height, temp, 0, n);
    cout << "swaps = " << swaps << '\n';
    return 0;
}

  2. สูงเท่ากันต้องหยิบจากซ้ายก่อน คนสูงเท่ากันไม่ใช่คู่สลับ ดูส่วนลองทำให้พัง

ได้ 14 ตรงกับที่นับด้วยมือ บรรทัดสุดท้ายก่อนคำตอบคือหกคู่ที่ไล่ในตาราง ตอนเทส่วนที่เหลือไม่ต้องนับอะไร ฝั่งขวาที่เหลือไม่มีใครในฝั่งซ้ายเหลือให้จับคู่แล้ว ส่วนฝั่งซ้ายที่เหลือ คู่ของมันกับฝั่งขวาถูกนับไปแล้วตอนที่ฝั่งขวาแต่ละตัวถูกหยิบ

เวลาเท่ากับ merge sort คือ O(N log N) เพราะบรรทัดที่เพิ่มทำงานคงที่ทุกครั้งที่หยิบ คำตอบมีได้ถึง N(N - 1)/2 ซึ่งที่ N เป็นแสนคือราว 5 × 109 เกิน int ตัวนับและค่าที่ฟังก์ชันคืนจึงต้องเป็น long long

นับเงื่อนไขอื่นข้ามครึ่ง

วิธีนี้ใช้ได้กว้างกว่าคู่สลับ ข้อเท็จจริงที่ใช้มีแค่สองข้อ ข้อแรก คู่ข้ามครึ่งไม่เปลี่ยนแม้เรียงแต่ละครึ่งแล้ว ข้อสอง พอสองครึ่งเรียงแล้ว นับคู่ข้ามครึ่งที่ผ่านเงื่อนไขได้ด้วยตัวชี้ที่เดินทางเดียว เช่น ถ้าต้องการคู่ i < j ที่ ai > aj + D ให้ไล่ j ในครึ่งขวาจากน้อยไปมาก แล้วเลื่อนตัวชี้ในครึ่งซ้ายข้ามทุกตัวที่ไม่เกิน aj + D ตัวที่เหลือในครึ่งซ้ายผ่านเงื่อนไขทั้งหมด พอ j ถัดไปมีค่ามากขึ้น ตัวชี้ก็ไม่ต้องถอยกลับ งานจึงเป็น O(n) ต่อการรวมหนึ่งครั้งเหมือนเดิม เงื่อนไขแบบนี้ไม่ตรงกับการเทียบที่ใช้รวม จึงต้องนับเป็นลูปแยก ก่อน รวม แบบฝึกหัดข้อ 6 และข้อ 7 ใช้วิธีนี้

อีกแบบที่เจอบ่อยคือกรรมการสองคนจัดอันดับผู้เข้าแข่งขัน N คน อยากรู้ว่ามีกี่คู่ที่สองคนเห็นไม่ตรงกันว่าใครดีกว่า ให้เรียงผู้เข้าแข่งขันตามอันดับของกรรมการคนแรก แล้วเขียนอันดับของกรรมการคนที่สองตามลำดับนั้น คู่ที่ขัดกันก็คือคู่สลับของลำดับที่ได้

แบบที่ 1 นับด้วย int

เขียนตัวนับและค่าที่คืนเป็น int แล้วทดสอบกับแถวที่เรียงจากสูงไปเตี้ย ซึ่งทุกคู่เป็นคู่สลับ คำตอบที่ถูกคือ N(N - 1)/2 โปรแกรมนี้สร้างแถวเองจาก N

#include <bits/stdc++.h>
using namespace std;

int count_inversions(vector<int> &a, vector<int> &temp, int lo, int hi)
{
    if (hi - lo <= 1) {
        return 0;
    }
    int mid = (lo + hi) / 2;
    int count = count_inversions(a, temp, lo, mid) + count_inversions(a, temp, mid, hi);
    int i = lo, j = mid, k = lo;
    while (i < mid && j < hi) {
        if (a[i] <= a[j]) {
            temp[k] = a[i];
            i++;
        } else {
            temp[k] = a[j];
            j++;
            count += mid - i;
        }
        k++;
    }
    while (i < mid) {
        temp[k] = a[i];
        i++;
        k++;
    }
    while (j < hi) {
        temp[k] = a[j];
        j++;
        k++;
    }
    for (int t = lo; t < hi; t++) {
        a[t] = temp[t];
    }
    return count;
}

int main()
{
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        a[i] = n - i;
    }
    vector<int> temp(n);
    int count = count_inversions(a, temp, 0, n);
    cout << "inversions = " << count << '\n';
    return 0;
}

สรุป

  • คู่สลับคือคู่ i < j ที่ ai > aj เท่ากับจำนวนครั้งที่ต้องสลับตัวที่อยู่ติดกันจนเรียง
  • คู่ทั้งหมดแบ่งเป็นคู่ในครึ่งซ้าย คู่ในครึ่งขวา และคู่ข้ามครึ่ง การเรียงแต่ละครึ่งไม่ทำให้จำนวนคู่ข้ามครึ่งเปลี่ยน
  • ระหว่างรวม ทุกครั้งที่หยิบจากครึ่งขวา บวก mid - i ได้ O(N log N) แทน O(N2) บนเครื่องที่วัด แสนตัวใช้ 0.009 วินาทีแทน 14.5 วินาที
  • ตัวนับต้องเป็น long long และค่าที่เท่ากันต้องหยิบจากซ้ายก่อนด้วย <=
  • เงื่อนไขอื่นที่นับด้วยตัวชี้ทางเดียวบนสองครึ่งที่เรียงแล้วได้ ให้นับเป็นลูปแยกก่อนรวม
