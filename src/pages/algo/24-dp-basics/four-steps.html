24.2 วิธีคิด DP สี่ขั้น

หน้า 24.1 ได้สูตรทอนเหรียญมาจากการคิดว่า "เหรียญแรกเป็นอะไรได้บ้าง" แต่พอเจอโจทย์ใหม่ น้องจะคิดสูตรแบบนั้นเองได้ยังไง หน้านี้แบ่งการคิดเป็นสี่ขั้นที่ทำบนกระดาษก่อนเขียนโค้ดทุกครั้ง ทำครบสี่ขั้นแล้ว โค้ดแทบจะเขียนตัวเอง แล้วต่อด้วยสองเรื่องที่มากับ DP เสมอ คือคำตอบที่ใหญ่จนต้อง mod และการประมาณเวลาจากจำนวนสถานะ

โจทย์บันไดที่มีขั้นพัง

บันไดหน้าอาคารเรียนมี N ขั้น น้องเริ่มยืนที่พื้น ซึ่งนับเป็นขั้น 0 แต่ละก้าวขึ้นได้ 1 หรือ 2 ขั้น ต้องไปหยุดที่ขั้น N พอดี แต่มีบางขั้นพัง ห้ามเหยียบ ถามว่ามีวิธีขึ้นกี่วิธี สองวิธีถือว่าต่างกันถ้าลำดับการก้าวต่างกัน

ลองกับบันได 10 ขั้นที่ขั้น 4 กับ 7 พัง จะข้ามขั้น 4 ได้ต้องก้าว 2 ขั้นจากขั้น 3 ไปขั้น 5 ข้ามขั้น 7 ได้ต้องก้าวจากขั้น 6 ไปขั้น 8 ทุกวิธีจึงต้องผ่านขั้น 3 5 6 และ 8 จากพื้นไปขั้น 3 มี 3 วิธี (1+1+1, 1+2, 2+1) จากขั้น 8 ไปขั้น 10 มี 2 วิธี รวม 3 × 2 = 6 วิธี นับด้วยมือได้เพราะบันไดสั้น ถ้าบันไดยาวแสนขั้นต้องให้โปรแกรมนับ

ขั้นที่ 1 นิยามสถานะ

สถานะ (state) คือปัญหาย่อยหนึ่งข้อ ที่เราจะเก็บคำตอบไว้ในตารางหนึ่งช่อง ขั้นนี้สำคัญที่สุด ต้องเขียนเป็นประโยคเต็มว่าช่อง i เก็บอะไร ข้อนี้

สถานะของโจทย์บันได

ways[i] คือจำนวนวิธีเดินจากพื้นไปหยุดที่ขั้น i โดยไม่เหยียบขั้นพัง

สถานะที่ดีมีสองอย่าง อย่างแรก คำตอบของโจทย์ต้องเป็นช่องใดช่องหนึ่ง หรือหาได้ง่ายจากตาราง ข้อนี้คำตอบคือ ways[N] อย่างที่สอง แต่ละช่องต้องคำนวณได้จากช่องอื่นที่เล็กกว่า โดยไม่ต้องรู้ว่าก่อนหน้านั้นเดินมายังไง ถ้าน้องต้องรู้ประวัติการเดินด้วยถึงจะคำนวณต่อได้ แปลว่าสถานะเก็บข้อมูลไม่พอ ต้องเพิ่มข้อมูลเข้าไปในสถานะ ซึ่งจะเห็นในแบบฝึกหัดข้อท้ายๆ

ขั้นที่ 2 หาความสัมพันธ์

หาว่าช่อง i คำนวณจากช่องไหน วิธีคิดที่ใช้ได้เกือบทุกข้อคือ ดูการตัดสินใจครั้งสุดท้าย จะไปหยุดที่ขั้น i ก้าวสุดท้ายต้องเป็นก้าว 1 ขั้นจากขั้น i - 1 หรือก้าว 2 ขั้นจากขั้น i - 2 อย่างใดอย่างหนึ่ง

ways[i] = 0                          ถ้าขั้น i พัง
ways[i] = ways[i - 1] + ways[i - 2]  ถ้าไม่พัง

บวกกันได้เพราะสองกลุ่มนี้ ไม่ซ้อนกันและครอบคลุมครบ ทุกวิธีที่ไปหยุดขั้น i มีก้าวสุดท้ายแบบเดียว จึงอยู่ในกลุ่มเดียวพอดี ไม่ถูกนับสองครั้ง และไม่มีวิธีไหนตกหล่น โจทย์นับวิธีทุกข้อต้องตรวจสองข้อนี้ ส่วนโจทย์หาค่าน้อยสุดหรือมากสุดแบบหน้า 24.1 ใช้ min หรือ max แทนการบวก ซึ่งไม่เดือดร้อนถ้ากลุ่มซ้อนกัน แต่ยังต้องครอบคลุมครบ

ขั้นที่ 3 กรณีฐาน

ช่องที่สูตรใช้ไม่ได้ ต้องกำหนดค่าเอง ข้อนี้คือขั้น 0 กับขั้น 1 เพราะสูตรจะอ่านขั้นติดลบ

  • ways[0] = 1 ยืนอยู่บนพื้นแล้วมีหนึ่งวิธี คือไม่ต้องก้าว ฟังดูแปลก แต่ตรวจได้ ways[2] ต้องเป็น 2 (1+1 กับ 2) ซึ่งได้จาก ways[1] + ways[0] = 1 + 1 ถ้าให้ ways[0] เป็น 0 จะผิดทันที ดูในส่วนลองทำให้พัง
  • ways[1] มาจากขั้น 0 ได้ทางเดียว จึงเท่ากับ ways[0] ถ้าขั้น 1 ไม่พัง

ขั้นที่ 4 ลำดับการเติมและคำตอบสุดท้าย

ดูจากสูตรว่าช่อง i อ่านช่องไหน แล้วเลือกลำดับที่ช่องเหล่านั้นเสร็จก่อน ข้อนี้อ่าน i - 1 กับ i - 2 จึงเติมจาก 1 ขึ้นไปถึง N แล้วตอบ ways[N] สองขั้นนี้รวมกันเพราะคิดคู่กันเสมอ ตำแหน่งของคำตอบไม่จำเป็นต้องเป็นช่องสุดท้าย หน้า 24.4 มีโจทย์ที่คำตอบคือค่ามากสุดของทุกช่อง

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int n, k;
    cin >> n >> k;
    vector<bool> broken(n + 1, false);
    for (int j = 0; j < k; j++) {
        int step;
        cin >> step;
        broken[step] = true;
    }
    vector<long long> ways(n + 1); // (1)!
    ways[0] = 1; // (2)!
    for (int i = 1; i <= n; i++) { // (3)!
        if (broken[i]) {
            ways[i] = 0;
        } else if (i == 1) { // (4)!
            ways[i] = ways[i - 1];
        } else {
            ways[i] = ways[i - 1] + ways[i - 2];
        }
    }
    for (int i = 0; i <= n; i++) { // (5)!
        cout << ways[i] << (i < n ? ' ' : '\n');
    }
    return 0;
}

  3. ขั้นที่ 4 เติมจากน้อยไปมาก
  4. ขั้นที่ 3 อีกกรณี ขั้น 1 มาจากขั้น 0 ได้ทางเดียว ไม่มีขั้น -1 ให้อ่าน
  5. พิมพ์ทั้งตารางให้ตรวจกับมือได้ โปรแกรมส่ง grader พิมพ์แค่ ways[n]

ความซับซ้อนคือจำนวนสถานะคูณงานต่อสถานะ

DP คำนวณแต่ละสถานะครั้งเดียว เวลาทั้งหมดจึงประมาณได้ก่อนเขียนจากสองตัวเลข

สูตรประมาณเวลา

เวลาของ DP = จำนวนสถานะ × งานที่ทำต่อหนึ่งสถานะ

บันไดข้างบนมี N + 1 สถานะ แต่ละสถานะบวกสองค่า จึงเป็น O(N) ลองทำให้งานต่อสถานะมากขึ้น ถ้าแต่ละก้าวขึ้นได้ 1 ถึง K ขั้น ก้าวสุดท้ายมาได้จาก K ขั้นก่อนหน้า ต้องบวก K ค่าต่อหนึ่งสถานะ

#include <bits/stdc++.h>
using namespace std;

const int MOD = 1000000007;

int main()
{
    int n, k;
    cin >> n >> k;
    vector<int> ways(n + 1);
    ways[0] = 1;
    for (int i = 1; i <= n; i++) {
        long long total = 0;
        for (int j = 1; j <= k && j <= i; j++) {
            total += ways[i - j];
        }
        ways[i] = total % MOD;
    }
    cout << ways[n] << '\n';
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

vector<char> broken;

long long count_ways(int i)
{
    if (broken[i]) {
        return 0;
    }
    if (i <= 1) {
        return 1;
    }
    return count_ways(i - 1) + count_ways(i - 2);
}

int main()
{
    int n, k;
    cin >> n >> k;
    broken = vector<char>(n + 1, false);
    for (int j = 0; j < k; j++) {
        int step;
        cin >> step;
        broken[step] = true;
    }
    cout << count_ways(n) << '\n';
    return 0;
}

const int MOD = 1000000007;

int main()
{
    int n, k;
    cin >> n >> k;
    vector<int> ways(n + 1);
    ways[0] = 1;
    for (int i = 1; i <= n; i++) {
        int total = 0;
        for (int j = 1; j <= k && j <= i; j++) {
            total += ways[i - j];
        }
        ways[i] = total % MOD;
    }
    cout << ways[n] << '\n';
    return 0;
}
