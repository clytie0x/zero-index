24.5 เก็บเส้นทางเพื่อพิมพ์คำตอบจริง

ทุกโปรแกรมที่ผ่านมาตอบได้แค่ตัวเลข เช่น ใช้ 2 เหรียญ หรือได้เงิน 140 บาท แต่โจทย์จำนวนมากถามต่อว่า "ใช้เหรียญอะไรบ้าง" "เข้าบ้านหลังไหน" หรือ "ช่วงวันไหน" ตารางเก็บแค่ค่าที่ดีที่สุดของแต่ละสถานะ ไม่ได้เก็บว่าได้มาจากทางไหน หน้านี้แสดงสองวิธี คือจดทางเลือกที่ชนะไว้ระหว่างเติมตาราง กับเดินย้อนตารางที่เติมเสร็จแล้ว ทั้งสองวิธีเพิ่มเวลาแค่เท่าความยาวของคำตอบ

จดว่าชนะด้วยทางไหน

ตอนเติมช่อง x ของโจทย์ทอนเหรียญ เราลองทุกเหรียญแล้วเก็บค่าที่น้อยที่สุด ถ้าจดไว้ด้วยว่าเหรียญไหนทำให้ได้ค่านั้น ก็รู้ว่าเหรียญสุดท้ายของคำตอบที่ x บาทคือเหรียญอะไร หักเหรียญนั้นออกได้ x - c บาท ซึ่งก็จดไว้แล้วว่าเหรียญสุดท้ายคืออะไร เดินย้อนแบบนี้จนเหลือ 0 บาท

#include <bits/stdc++.h>
using namespace std;

const int INF = 1000000000;

int main()
{
    int n, target;
    cin >> n >> target;
    vector<int> coins(n);
    for (int i = 0; i < n; i++) {
        cin >> coins[i];
    }
    vector<int> fewest(target + 1, INF);
    vector<int> last_coin(target + 1, 0); // (1)!
    fewest[0] = 0;
    for (int x = 1; x <= target; x++) {
        for (int c : coins) {
            if (c <= x && fewest[x - c] + 1 < fewest[x]) {
                fewest[x] = fewest[x - c] + 1;
                last_coin[x] = c; // (2)!
            }
        }
    }
    if (fewest[target] == INF) { // (3)!
        cout << "impossible\n";
        return 0;
    }
    cout << "coins = " << fewest[target] << '\n';
    int x = target;
    while (x > 0) { // (4)!
        cout << last_coin[x] << (x - last_coin[x] > 0 ? ' ' : '\n');
        x -= last_coin[x];
    }
    return 0;
}
  1. อาเรย์ที่สองขนาดเท่าตาราง จดเหรียญสุดท้ายของคำตอบที่ดีที่สุดของแต่ละช่อง อาเรย์แบบนี้มักเรียกว่า parent หรือ choice
  2. จดเฉพาะตอนที่ได้ค่าดีขึ้นจริง จึงเปลี่ยน min เป็น if ที่เทียบด้วย <
  3. เช็กว่าทำได้ก่อนเดินย้อน ดูในส่วนลองทำให้พัง
  4. เดินย้อนจากเป้าหมายลงไป แต่ละรอบหักเหรียญที่จดไว้ จบเมื่อเหลือ 0 บาท

เดินย้อนตารางโดยไม่จดเพิ่ม

บางครั้งไม่ต้องมีอาเรย์ที่สอง เพราะดูจากตารางก็รู้ว่าแต่ละช่องมาจากทางไหน บ้านติดกันในหน้า 24.4 มีแค่สองทาง ถ้า best[i] เท่ากับ best[i - 1] แปลว่าไม่ต้องเข้าหลังที่ i ก็ได้เท่านี้ ข้ามไปดูหลังที่ i - 1 ถ้าไม่เท่า แปลว่าค่านี้ได้มาจากการเข้าหลังที่ i เท่านั้น จดหลังที่ i แล้วกระโดดไปดูหลังที่ i - 2

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int n;
    cin >> n;
    vector<long long> money(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> money[i];
    }
    vector<long long> best(n + 1);
    best[0] = 0;
    best[1] = money[1];
    for (int i = 2; i <= n; i++) {
        best[i] = max(best[i - 1], best[i - 2] + money[i]);
    }
    vector<int> chosen;
    int i = n;
    while (i >= 1) {
        if (i >= 2 && best[i] == best[i - 1]) { // (1)!
            i -= 1;
        } else {
            chosen.push_back(i); // (2)!
            i -= 2;
        }
    }
    reverse(chosen.begin(), chosen.end()); // (3)!
    cout << "total = " << best[n] << '\n';
    cout << "houses =";
    for (int h : chosen) {
        cout << ' ' << h;
    }
    cout << '\n';
    return 0;
}
  1. ไม่เข้าหลังที่ i ก็ได้ค่าเท่ากัน จึงไม่เข้า เช็ก i >= 2 ก่อน เพราะหลังที่ 1 ตาราง best[1] คือ a[1] ถ้าไปถึงหลังที่ 1 ได้แปลว่าต้องเข้า
  2. ค่านี้ได้จากการเข้าหลังที่ i เท่านั้น จด แล้วข้ามหลังที่ i - 1 ซึ่งเข้าไม่ได้
  3. เดินย้อนจึงได้เลขบ้านจากมากไปน้อย กลับลำดับให้เป็นน้อยไปมากก่อนพิมพ์

ช่วงย่อยมากสุดเป็นวันไหน

Kadane ในหน้า 24.4 ตอบว่ากำไรมากสุดคือ 7 แต่เจ้าของร้านอยากรู้ว่าวันไหนถึงวันไหน ช่วงที่จบที่วัน i เริ่มที่ไหน ขึ้นกับว่าวัน i ต่อจากช่วงเดิมหรือเริ่มใหม่ จึงจดวันเริ่มไว้คู่กับทุกช่อง

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int n;
    cin >> n;
    vector<long long> profit(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> profit[i];
    }
    vector<long long> ending(n + 1);
    vector<int> start(n + 1); // (1)!
    ending[1] = profit[1];
    start[1] = 1;
    int best_end = 1;
    for (int i = 2; i <= n; i++) {
        if (ending[i - 1] + profit[i] >= profit[i]) { // (2)!
            ending[i] = ending[i - 1] + profit[i];
            start[i] = start[i - 1];
        } else {
            ending[i] = profit[i];
            start[i] = i;
        }
        if (ending[i] > ending[best_end]) { // (3)!
            best_end = i;
        }
    }
    cout << "best = " << ending[best_end] << '\n';
    cout << "days " << start[best_end] << " to " << best_end << '\n';
    return 0;
}
  1. start[i] คือวันเริ่มของช่วงที่ดีที่สุดที่จบที่วัน i
  2. ต่อจากช่วงเดิม วันเริ่มก็เป็นวันเริ่มเดิม ถ้าเริ่มใหม่ วันเริ่มคือวันนี้ ใช้ >= ให้ต่อช่วงเดิมเมื่อได้เท่ากัน จึงได้ช่วงที่ยาวกว่า
  3. จำวันจบของช่วงที่ดีที่สุดแทนการจำค่า ใช้ > จึงได้ช่วงที่จบเร็วที่สุดเมื่อมีหลายช่วงเท่ากัน

ทำไมไม่เก็บคำตอบทั้งชุดไว้ในทุกช่อง

สรุป

  • ตาราง DP เก็บแค่ค่าที่ดีที่สุด ถ้าต้องการคำตอบจริงต้องรู้ว่าแต่ละช่องได้ค่ามาจากทางไหน
  • วิธีแรก จดทางเลือกที่ชนะลงอาเรย์ที่สองขณะเติมตาราง แล้วเดินย้อนจากช่องคำตอบ
  • วิธีที่สอง เดินย้อนตารางที่เติมเสร็จแล้ว ถามว่าค่าในช่องนี้ได้มาจากทางไหน
  • เดินย้อนได้คำตอบกลับหัว ใช้ reverse ถ้าต้องการลำดับเดิม
  • เมื่อมีหลายคำตอบที่ดีเท่ากัน ลำดับที่ลองและ < กับ <= เป็นตัวเลือกว่าได้แบบไหน
  • เก็บคำตอบทั้งชุดไว้ทุกช่องเป็น O(X2) ทั้งเวลาและหน่วยความจำ จดเหรียญสุดท้ายเพิ่มแค่อาเรย์เดียว
  • เช็กว่าทำได้ก่อนเดินย้อนเสมอ ไม่งั้นลูปไม่จบ
