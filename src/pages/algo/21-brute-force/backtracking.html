21.3 backtracking

วิธีในหน้า 21.2 สร้างคำตอบให้ครบทุกช่องก่อน แล้วค่อยตรวจว่าใช้ได้ไหม ถ้าคำตอบส่วนใหญ่ผิดกฎตั้งแต่ช่องแรกๆ ก็เสียเวลาสร้างส่วนที่เหลือไปเปล่าๆ backtracking (การย้อนรอย) สร้างคำตอบทีละช่อง และเลือกเฉพาะตัวเลือกที่ยังไม่ผิดกฎ ถ้าถึงช่องไหนแล้วไม่มีตัวเลือกเหลือ ก็ถอยกลับไปเปลี่ยนตัวเลือกของช่องก่อนหน้า โครง เลือก ลอง คืนสภาพ ในหน้า 9.5 คือ backtracking อยู่แล้ว หน้านี้เอาไปใช้กับโจทย์ที่มีกฎจริงๆ คือ N-Queens ผลรวมเซตย่อย และซูโดกุ

ต้นไม้การค้นหา

โจทย์ N-Queens คือวางควีน N ตัวบนกระดาน N × N ไม่ให้มีสองตัวไหนกินกันได้ ควีนกินกันได้เมื่ออยู่แถวเดียวกัน คอลัมน์เดียวกัน หรือแนวทแยงเดียวกัน แต่ละแถวจึงมีควีนได้ตัวเดียวพอดี คำตอบหนึ่งแบบคือคอลัมน์ของควีนในแถวที่ 1 ถึง N

backtracking วางทีละแถว แถวละหนึ่งตัว ทุกครั้งที่วางให้เลือกเฉพาะคอลัมน์ที่ไม่โดนควีนแถวก่อนๆ กิน ภาพนี้คือการค้นหาทั้งหมดเมื่อ N = 4 แต่ละกล่องคือสถานะระหว่างทาง ตัวเลขในกล่องคือคอลัมน์ของควีนในแถวที่วางไปแล้วเรียงตามแถว เช่น 14 คือแถว 1 วางคอลัมน์ 1 แถว 2 วางคอลัมน์ 4 เส้นจากกล่องลงไปคือตัวเลือกที่ยังไม่ผิดกฎ ภาพแบบนี้เรียกว่า ต้นไม้การค้นหา (search tree)

กล่องเส้นประคือทางตัน เช่น 13 วางแถว 3 ที่ไหนก็โดนกิน จึงไม่มีเส้นลงไป ต้องถอยกลับไปลอง 14 แทน กล่องสีคือคำตอบที่วางครบ 4 ตัว มีสองแบบ คือ 2413 กับ 3142 ทั้งต้นไม้มี 17 กล่อง เทียบกับการสร้างทุกการเรียงสับเปลี่ยนของคอลัมน์ก่อนแล้วค่อยตรวจ ซึ่งต้องสร้างกระดานเต็ม 4! = 24 กระดาน กิ่งที่ตันตั้งแต่แถวที่ 2 ไม่ถูกสร้างต่อเลย ยิ่ง N ใหญ่ ส่วนที่ไม่ต้องสร้างยิ่งมาก

N-Queens

ก่อนวางควีนที่แถว row คอลัมน์ c ต้องรู้ให้เร็วว่าโดนกินไหม ไม่ต้องไล่ดูควีนทุกตัว ใช้อาเรย์จำไว้สามอัน

  • col_used[c] คอลัมน์ c มีควีนแล้ว
  • diag1_used[row + c] แนวทแยงที่ขึ้นไปทางขวา ทุกช่องบนเส้นเดียวกันมี row + c เท่ากัน มีค่า 0 ถึง 2N - 2
  • diag2_used[row - c + N - 1] แนวทแยงที่ลงไปทางขวา ทุกช่องบนเส้นเดียวกันมี row - c เท่ากัน บวก N - 1 ให้ไม่ติดลบ ใช้เป็นตำแหน่งในอาเรย์ได้
#include <bits/stdc++.h>
using namespace std;

int n;
int queen_col[15];
bool col_used[15], diag1_used[30], diag2_used[30];
long long solutions = 0, nodes = 0;

void print_board()
{
    for (int r = 0; r < n; r++) {
        for (int c = 0; c < n; c++) {
            cout << (queen_col[r] == c ? 'Q' : '.');
        }
        cout << '\n';
    }
}

void place(int row)
{
    nodes++; // (1)!
    if (row == n) {
        if (solutions == 0) {
            print_board();
        }
        solutions++;
        return;
    }
    for (int c = 0; c < n; c++) {
        int d1 = row + c, d2 = row - c + n - 1;
        if (col_used[c] || diag1_used[d1] || diag2_used[d2]) { // (2)!
            continue;
        }
        queen_col[row] = c;
        col_used[c] = diag1_used[d1] = diag2_used[d2] = true; // (3)!
        place(row + 1);
        col_used[c] = diag1_used[d1] = diag2_used[d2] = false; // (4)!
    }
}

int main()
{
    cin >> n;
    place(0);
    cout << "solutions " << solutions << '\n';
    cout << "nodes " << nodes << '\n';
    return 0;
}
  1. นับว่าเรียก place กี่ครั้ง ซึ่งก็คือจำนวนกล่องในต้นไม้การค้นหา ใช้วัดว่าค้นหาหนักแค่ไหน ไม่เกี่ยวกับคำตอบ
  2. ตัดตัวเลือกที่ผิดกฎทิ้งตั้งแต่ตอนนี้ ไม่ปล่อยให้ลงไปสร้างแถวที่เหลือ บรรทัดนี้คือส่วนที่ทำให้ backtracking ต่างจากลองทุกแบบ
  3. เลือก วางควีนแล้วจดว่าคอลัมน์และแนวทแยงทั้งสองเส้นถูกใช้แล้ว
  4. คืนสภาพ หลังลองทุกแบบที่มีควีนตัวนี้แล้ว ปลดออก ให้รอบถัดไปของลูปลองคอลัมน์อื่น ถ้าลืม แถวก่อนหน้าจะเห็นช่องว่างเป็นช่องที่โดนกินทั้งที่ไม่มีควีนแล้ว

โครงของ backtracking

โจทย์ backtracking ทุกข้อเขียนตามโครงเดียวกัน เปลี่ยนแค่ว่าสถานะคืออะไร ตัวเลือกคืออะไร และกฎคืออะไร

search(สถานะตอนนี้)
    ถ้าสร้างครบแล้ว  บันทึกหรือนับคำตอบ แล้วกลับ
    สำหรับทุกตัวเลือก x ของช่องถัดไป
        ถ้า x ผิดกฎ  ข้าม
        เลือก x  (เปลี่ยนสถานะ)
        search(สถานะใหม่)
        คืนสภาพ x  (เปลี่ยนสถานะกลับเหมือนก่อนเลือก)

คำถามที่ต้องตอบก่อนเขียน คือแต่ละชั้นของ recursion เติมช่องไหน (N-Queens เติมทีละแถว) ตัวเลือกมีอะไรบ้าง (คอลัมน์ 0 ถึง N - 1) และเช็กกฎยังไงให้เร็ว (อาเรย์สามอัน)

ผลรวมเซตย่อย

ห้องเรียนมีงบซื้อของขวัญวันครู S บาทพอดี ร้านมีของ n ชิ้นราคาต่างๆ ให้พิมพ์ทุกชุดของของที่ราคารวมเท่ากับ S พอดี โจทย์แบบนี้เรียกว่า ผลรวมเซตย่อย (subset sum) ใช้ bitmask ในหน้า 21.2 ก็ได้ แต่เขียนแบบ backtracking ไล่ทีละชิ้น ชั้นที่ i ตัดสินใจเรื่องชิ้นที่ i ว่าจะเลือกหรือไม่ และเก็บชิ้นที่เลือกไว้ใน vector

#include <bits/stdc++.h>
using namespace std;

int n, target;
vector<int> price;
vector<int> chosen;

void search(int i, int sum)
{
    if (i == n) {
        if (sum == target) {
            for (int k = 0; k < (int)chosen.size(); k++) {
                cout << chosen[k] << (k + 1 < (int)chosen.size() ? ' ' : '\n');
            }
        }
        return;
    }
    chosen.push_back(price[i]); // (1)!
    search(i + 1, sum + price[i]);
    chosen.pop_back(); // (2)!
    search(i + 1, sum); // (3)!
}

int main()
{
    cin >> n >> target;
    price.resize(n);
    for (int i = 0; i < n; i++) {
        cin >> price[i];
    }
    search(0, 0);
    return 0;
}
  1. เลือก ชิ้นที่ i ใส่ท้าย vector แล้วลองทุกแบบของชิ้นที่เหลือ ผลรวมส่งไปเป็นพารามิเตอร์ จึงไม่ต้องคืนสภาพเอง เพราะ sum ของชั้นนี้ไม่ได้เปลี่ยน
  2. คืนสภาพ เอาชิ้นที่ i ออกจาก vector ก่อนลองทางที่ไม่เลือก vector เป็นของที่ทุกชั้นใช้ร่วมกัน ต่างจาก sum
  3. ทางที่สอง ไม่เลือกชิ้นที่ i ทุกชั้นแตกเป็นสองทางแบบสตริงฐานสองในหน้า 9.5 รวม 2n ชุดที่ไปถึงชั้นล่างสุด

ซูโดกุ 4 × 4

ซูโดกุขนาดเล็ก ตาราง 4 × 4 แบ่งเป็นกล่อง 2 × 2 สี่กล่อง ต้องเติมเลข 1 ถึง 4 ให้ทุกแถว ทุกคอลัมน์ และทุกกล่องมีเลขครบไม่ซ้ำ ช่องที่เป็น 0 คือช่องว่าง เติมทีละช่องจากซ้ายไปขวา บนลงล่าง ให้ช่องที่ pos คือแถว pos / 4 คอลัมน์ pos % 4 เหมือนเปลี่ยนตารางเป็นแถวยาวแถวเดียว

#include <bits/stdc++.h>
using namespace std;

int grid[4][4];

bool can_put(int r, int c, int d)
{
    for (int k = 0; k < 4; k++) {
        if (grid[r][k] == d || grid[k][c] == d) {
            return false;
        }
    }
    int top = r / 2 * 2, left = c / 2 * 2; // (1)!
    for (int i = top; i < top + 2; i++) {
        for (int j = left; j < left + 2; j++) {
            if (grid[i][j] == d) {
                return false;
            }
        }
    }
    return true;
}

bool solve(int pos)
{
    if (pos == 16) {
        return true;
    }
    int r = pos / 4, c = pos % 4;
    if (grid[r][c] != 0) {
        return solve(pos + 1); // (2)!
    }
    for (int d = 1; d <= 4; d++) {
        if (can_put(r, c, d)) {
            grid[r][c] = d;
            if (solve(pos + 1)) { // (3)!
                return true;
            }
            grid[r][c] = 0;
        }
    }
    return false;
}

int main()
{
    for (int r = 0; r < 4; r++) {
        for (int c = 0; c < 4; c++) {
            cin >> grid[r][c];
        }
    }
    if (!solve(0)) {
        cout << "no solution\n";
        return 0;
    }
    for (int r = 0; r < 4; r++) {
        for (int c = 0; c < 4; c++) {
            cout << grid[r][c] << (c < 3 ? ' ' : '\n');
        }
    }
    return 0;
}
  1. มุมซ้ายบนของกล่องที่ช่อง (r, c) อยู่ หารแล้วคูณ 2 ปัดลงไปหาเลขคู่ที่ใกล้ที่สุด เช่น r = 3 ได้ 2
  2. ช่องที่มีเลขให้มาแล้วไม่ต้องเลือก ข้ามไปช่องถัดไปเลย
  3. solve คืน true เมื่อเติมครบ พอเจอคำตอบแรกแล้วทุกชั้นคืน true ต่อกันขึ้นไปทันที ไม่ลองตัวเลือกที่เหลือ ตารางจึงค้างค่าของคำตอบไว้ให้พิมพ์ ถ้าโจทย์ต้องการนับทุกคำตอบ ให้ใช้แบบ N-Queens ที่ไม่หยุด

สรุป

  • backtracking สร้างคำตอบทีละช่อง ตัดตัวเลือกที่ผิดกฎทิ้งทันที ถ้าไม่มีตัวเลือกเหลือก็ถอยกลับไปเปลี่ยนช่องก่อนหน้า
  • ต้นไม้การค้นหามีกล่องคือสถานะระหว่างทาง นับจำนวนกล่องด้วยตัวแปร nodes เพื่อดูว่าค้นหาหนักแค่ไหน
  • N-Queens วางทีละแถว ใช้อาเรย์จำคอลัมน์ row + c และ row - c + N - 1 ตรวจได้ใน O(1)
  • ทุกอย่างที่ทุกชั้นใช้ร่วมกัน ทั้ง vector อาเรย์ และตาราง ต้องคืนสภาพหลังลองเสร็จ
  • ต้องการแค่คำตอบเดียว ให้ฟังก์ชันคืน bool แล้วหยุดทันทีที่เจอ ต้องการนับทุกคำตอบ ให้ไล่จนครบ
