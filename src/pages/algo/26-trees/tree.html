26.1 ต้นไม้และการเก็บในโปรแกรม

ข้อมูลทุกแบบที่ผ่านมาเรียงเป็นแถว ไม่ว่าจะเป็นอาเรย์ linked list หรือตาราง แต่ของหลายอย่างในชีวิตจริงเป็นลำดับชั้น ผังโรงเรียนมีผู้อำนวยการอยู่บนสุด ใต้ลงมาเป็นรองผู้อำนวยการ หัวหน้ากลุ่มสาระ แล้วจึงเป็นครู โฟลเดอร์ในเครื่องมีโฟลเดอร์ย่อยซ้อนลงไปเรื่อยๆ สายตระกูลมีพ่อแม่ ลูก และหลาน โครงสร้างแบบนี้เรียกว่า ต้นไม้ (tree) และ set map priority_queue ในบทที่ 20 ก็เป็นต้นไม้อยู่ข้างใน หน้านี้เริ่มจากคำศัพท์ที่โจทย์ใช้ แล้วดูสามวิธีเก็บต้นไม้ในโปรแกรม พร้อมวิธีอ่านต้นไม้จากข้อมูลเข้า ซึ่งเป็นสิ่งแรกที่ต้องทำในโจทย์ต้นไม้ทุกข้อ

ต้นไม้คืออะไร

ต้นไม้ประกอบด้วย โหนด (node) ซึ่งก็คือของแต่ละชิ้น เช่น คนหนึ่งคนในผังโรงเรียน กับ เส้นเชื่อม (edge) ระหว่างโหนดที่เป็นหัวหน้ากับลูกน้องโดยตรง กติกามีสามข้อ

  • มีโหนดเดียวที่อยู่บนสุด เรียกว่า ราก (root)
  • ทุกโหนดที่ไม่ใช่รากมีโหนดเหนือตัวเองโดยตรงหนึ่งโหนดพอดี
  • เริ่มจากโหนดไหนก็ตาม เดินขึ้นไปเรื่อยๆ แล้วจะถึงรากเสมอ ไม่มีทางวนกลับมาที่เดิม

ต้นไม้ในวิชาคอมพิวเตอร์วาดรากไว้ข้างบนและแตกกิ่งลงล่าง กลับหัวกับต้นไม้จริง ภาพนี้คือชมรมหุ่นยนต์ของโรงเรียนที่มีสมาชิก 9 คน หมายเลข 1 ถึง 9 แต่ละเส้นต่อจากหัวหน้าลงไปหาคนที่รายงานตรงต่อคนนั้น

คำศัพท์ที่โจทย์ใช้

โจทย์ต้นไม้เขียนด้วยคำเหล่านี้ทั้งนั้น อ่านตารางคู่กับภาพข้างบน

นับเส้นหรือนับโหนด

เอกสารนี้นับความลึกและความสูงเป็นจำนวนเส้น ต้นไม้ที่มีโหนดเดียวจึงสูง 0 แต่บางโจทย์และบางตำรานับเป็นจำนวนโหนด ต้นไม้โหนดเดียวสูง 1 ต่างกันอยู่หนึ่งเสมอ อ่านโจทย์และตัวอย่างให้แน่ใจก่อนว่าใช้แบบไหน

binary tree

ต้นไม้ที่ทุกโหนดมีลูกได้ไม่เกินสองตัว และแยกชัดว่าตัวไหนเป็นลูกซ้าย (left child) ตัวไหนเป็นลูกขวา (right child) เรียกว่า binary tree (ต้นไม้ทวิภาค) โหนดที่มีลูกตัวเดียวก็ต้องบอกว่าอยู่ทางซ้ายหรือทางขวา สองแบบนี้ถือเป็นคนละต้นไม้กัน ต้นไม้ค้นหาในหน้า 26.4 และ heap ในหน้า 26.5 เป็น binary tree ทั้งคู่

โหนด 1 ไม่มีลูกซ้าย มีแต่ลูกขวาคือ 4 ถ้าย้าย 4 ไปเป็นลูกซ้ายของ 1 จะได้ binary tree อีกต้นหนึ่ง ตัวเลขในวงกลมเป็นแค่หมายเลขประจำโหนด ไม่ได้บอกลำดับอะไร ต่อไปจะเก็บต้นไม้ต้นนี้ในโปรแกรมสองแบบ

แบบที่ 1 node กับ pointer

แบบตรงไปตรงมาที่สุดคือทำเหมือน linked list ในหน้า 14.1 แต่ละ node มี pointer สองตัวไปหาลูกซ้ายกับลูกขวา ขอ node ใหม่ด้วย new และคืนด้วย delete ตามหน้า 15.5

#include <bits/stdc++.h>
using namespace std;

struct Node {
    int value;
    Node *left; // (1)!
    Node *right;
};

Node *make_node(int value)
{
    Node *node = new Node;
    node->value = value;
    node->left = nullptr; // (2)!
    node->right = nullptr;
    return node;
}

int main()
{
    Node *root = make_node(3);
    root->left = make_node(1);
    root->right = make_node(6);
    root->left->right = make_node(4); // (3)!
    root->right->left = make_node(2);
    root->right->right = make_node(5);
    cout << "root " << root->value << '\n';
    cout << "left of root " << root->left->value << '\n';
    cout << "right of root " << root->right->value << '\n';
    cout << "right of " << root->left->value << " is " << root->left->right->value << '\n';
    if (root->left->left == nullptr) {
        cout << root->left->value << " has no left child\n";
    }
    delete root->right->right; // (4)!
    delete root->right->left;
    delete root->left->right;
    delete root->right;
    delete root->left;
    delete root;
    return 0;
}
  1. linked list มี next ตัวเดียว node ของ binary tree มีทางไปต่อสองทาง
  2. nullptr คือยังไม่มีลูกทางนั้น ต้องตั้งเองทุกครั้ง new ไม่ได้ตั้งให้ เหมือนกับดักเรื่อง next ของตู้สุดท้ายในหน้า 14.1
  3. เดินตาม pointer สองทอด จากราก ไปลูกซ้าย แล้วตั้งลูกขวาของโหนดนั้น
  4. คืนลูกก่อนพ่อ ถ้าคืนพ่อก่อน จะอ่าน root->right หลังจากที่ root ถูกคืนไปแล้วไม่ได้ หน้า 26.4 จะเขียนฟังก์ชันคืนทั้งต้นไม้ให้

left of root 1

right of 1 is 4
1 has no left child

แบบนี้เหมาะเมื่อต้นไม้โตขึ้นเรื่อยๆ ระหว่างรัน และไม่รู้ล่วงหน้าว่าจะมีกี่โหนด เช่น ต้นไม้ค้นหาในหน้า 26.4 ที่ใส่ค่าเข้าไปทีละตัว

แบบที่ 2 อาเรย์ลูกซ้ายกับลูกขวา

โจทย์แข่งขันส่วนใหญ่ให้โหนดมีหมายเลข 1 ถึง N อยู่แล้ว เหมือนนักเรียนหมายเลข 1 ถึง N ในหน้า 14.3 จึงใช้หมายเลขเป็น index ได้เลย เก็บลูกซ้ายของโหนด v ไว้ที่ left[v] ลูกขวาที่ right[v] และใช้ 0 แทน "ไม่มีลูก" เหมือนที่ 0 แทน NULL ในหน้า 14.3

ข้อมูลเข้าของโจทย์แบบนี้มักเป็น บรรทัดแรกคือ N แล้วอีก N บรรทัด บรรทัดที่ v บอกลูกซ้ายและลูกขวาของโหนด v โดยไม่บอกว่ารากคือโหนดไหน ต้องหาเอง รากคือโหนดเดียวที่ไม่ได้เป็นลูกของใครเลย

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int n;
    cin >> n;
    vector<int> left(n + 1), right(n + 1); // (1)!
    vector<bool> has_parent(n + 1, false);
    for (int v = 1; v <= n; v++) {
        cin >> left[v] >> right[v];
        has_parent[left[v]] = true; // (2)!
        has_parent[right[v]] = true;
    }
    int root = 0;
    for (int v = 1; v <= n; v++) {
        if (!has_parent[v]) { // (3)!
            root = v;
        }
    }
    cout << "root " << root << '\n';
    for (int v = 1; v <= n; v++) {
        if (left[v] == 0 && right[v] == 0) {
            cout << "leaf " << v << '\n';
        }
    }
    return 0;
}
  1. ขอ n + 1 ช่องแล้วไม่ใช้ช่อง 0 หมายเลขโหนดจะตรงกับ index พอดี
  2. โหนดที่ไม่มีลูกซ้ายจะได้ left[v] เป็น 0 บรรทัดนี้จึงไปตั้ง has_parent[0] ด้วย ไม่เสียหาย เพราะช่อง 0 มีอยู่จริงและเราไม่เคยดูมัน
  3. โหนดที่ไม่มีใครชี้มาหาคือราก ต้นไม้ที่ถูกต้องมีแบบนี้โหนดเดียว

แบบที่ 3 รายชื่อลูกด้วย vector

ต้นไม้ทั่วไปอย่างชมรมหุ่นยนต์ แต่ละโหนดมีลูกได้ไม่จำกัด จะเปิดอาเรย์ left right ไม่ได้แล้ว ให้แต่ละโหนดมี vector ของลูกตัวเอง children[v] ยาวเท่าไรก็ได้ตามจำนวนลูก

รูปแบบข้อมูลเข้าที่พบบ่อยที่สุดของต้นไม้ทั่วไปคือบอกพ่อของแต่ละโหนด บรรทัดแรกคือ N บรรทัดที่สองมี N ตัว ตัวที่ v คือพ่อของโหนด v และรากมีพ่อเป็น 0 ชมรมหุ่นยนต์เขียนได้แบบนี้

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int n;
    cin >> n;
    vector<int> parent(n + 1);
    vector<vector<int>> children(n + 1); // (1)!
    int root = 0;
    for (int v = 1; v <= n; v++) {
        cin >> parent[v];
        if (parent[v] == 0) {
            root = v;
        } else {
            children[parent[v]].push_back(v); // (2)!
        }
    }
    cout << "root " << root << '\n';
    for (int v = 1; v <= n; v++) {
        cout << "node " << v << ':';
        if (children[v].empty()) {
            cout << " leaf";
        }
        for (int c : children[v]) {
            cout << ' ' << c;
        }
        cout << '\n';
    }
    return 0;
}
  1. vector ของ vector แบบตารางในหน้า 16.2 แต่ละแถวยาวไม่เท่ากัน เริ่มต้นทุกแถวว่าง
  2. รู้พ่อของ v แล้ว ก็ใส่ v ลงในรายชื่อลูกของพ่อ อ่านข้อมูลรอบเดียวได้รายชื่อลูกครบทุกโหนด

เก็บทั้ง parent และ children ไว้คู่กันเป็นเรื่องปกติ parent ใช้เดินขึ้นหาราก children ใช้เดินลงหาลูกหลาน binary tree ก็เก็บแบบนี้ได้เหมือนกัน แต่จะเสียข้อมูลว่าลูกตัวไหนอยู่ซ้ายตัวไหนอยู่ขวา

หาความลึกด้วยการเดินขึ้น

มีพ่อของทุกโหนดอยู่แล้ว ความลึกของโหนด v ก็คือจำนวนก้าวที่เดินจาก v ขึ้นไปจนถึงราก

#include <bits/stdc++.h>
using namespace std;

int main()
{
    int n;
    cin >> n;
    vector<int> parent(n + 1);
    for (int v = 1; v <= n; v++) {
        cin >> parent[v];
    }
    int height = 0;
    for (int v = 1; v <= n; v++) {
        int depth = 0;
        for (int u = v; parent[u] != 0; u = parent[u]) { // (1)!
            depth++;
        }
        cout << "depth of " << v << " is " << depth << '\n';
        height = max(height, depth);
    }
    cout << "height " << height << '\n';
    return 0;
}
  1. เดินขึ้นทีละก้าวแบบเดินตาม next ในลิสต์ หยุดเมื่อถึงโหนดที่ไม่มีพ่อ ซึ่งก็คือราก
